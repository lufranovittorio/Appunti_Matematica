\chapter{Vanishing and Embedding Theorems}\label{kodaira}
\chapterstatus{planned}

Positivity of line bundles, vanishing theorems and the characterisation of projective manifolds among compact K\"ahler manifolds.

\noindent\textbf{Prerequisites.} Chapter~\ref{lefschetz-theorems} (Hard Lefschetz and the Hodge--Riemann Relations) and Chapter~\ref{holomorphic-bundles} (Holomorphic Vector Bundles).

\noindent\textbf{Planned contents.}
\begin{itemize}
\item Positive line bundles.
\item The Bochner--Kodaira--Nakano identity.
\item The Kodaira--Nakano vanishing theorem.
\item The Lefschetz hyperplane theorem.
\item The Kodaira embedding theorem: Hodge manifolds are projective.
\end{itemize}
