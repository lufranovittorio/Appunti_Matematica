\chapter{The Standard Conjectures and Motives}\label{standard-conjectures}
\chapterstatus{planned}

The conjectural framework in which the Hodge conjecture lives.

\noindent\textbf{Prerequisites.} Chapter~\ref{intersection-theory} (Algebraic Cycles and Chow Groups), Chapter~\ref{lefschetz-theorems} (Hard Lefschetz and the Hodge--Riemann Relations) and Chapter~\ref{tate-conjecture} (The Tate Conjecture).

\noindent\textbf{Planned contents.}
\begin{itemize}
\item Weil cohomology theories.
\item Grothendieck's standard conjectures.
\item The Hodge conjecture implies the Lefschetz standard conjecture in characteristic zero.
\item Pure motives and Jannsen's semisimplicity theorem.
\item The Bloch--Beilinson conjectures.
\end{itemize}
