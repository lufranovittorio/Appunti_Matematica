\chapter{Algebraic Cycles and Chow Groups}\label{intersection-theory}
\chapterstatus{planned}

Algebraic cycles, rational equivalence and intersection products.

\noindent\textbf{Prerequisites.} Chapter~\ref{divisors} (Divisors and Line Bundles) and Chapter~\ref{schemes} (Schemes).

\noindent\textbf{Planned contents.}
\begin{itemize}
\item Cycles, rational equivalence and Chow groups.
\item Proper pushforward and flat pullback.
\item Intersection with divisors; Chern classes of vector bundles.
\item Intersection products on smooth varieties.
\item Correspondences.
\item Algebraic, homological and numerical equivalence.
\item The Grothendieck--Riemann--Roch theorem (statement).
\end{itemize}
