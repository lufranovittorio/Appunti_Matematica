\chapter{de Rham Cohomology}\label{de-rham}
\chapterstatus{planned}

de Rham cohomology and the de Rham theorem identifying it with singular cohomology with real coefficients.

\noindent\textbf{Prerequisites.} Chapter~\ref{differential-forms} (Differential Forms and Integration), Chapter~\ref{sheaf-cohomology} (Sheaf Cohomology), Chapter~\ref{poincare-duality} (Topological Manifolds and Poincar\'e Duality) and Chapter~\ref{distributions} (Distributions and Sobolev Spaces).

\noindent\textbf{Planned contents.}
\begin{itemize}
\item The de Rham complex and de Rham cohomology.
\item The Poincar\'e lemma and homotopy invariance.
\item The Mayer--Vietoris sequence and cohomology with compact supports.
\item The de Rham theorem via sheaf cohomology; wedge product and cup product.
\item Poincar\'e duality in de Rham cohomology.
\item Currents, and currents of integration over submanifolds.
\end{itemize}
