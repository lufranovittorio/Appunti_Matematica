\section{Introduction}\label{rings:section-introduction}

This chapter develops the basic theory of rings and modules: ideals and
quotients, prime and maximal ideals, factorisation in integral domains,
polynomial rings, modules and exact sequences, tensor products, algebras, and
the structure of finitely generated modules over a principal ideal domain.
Commutative rings were introduced in Section~\ref{number-systems:section-rings-fields};
the deeper theory of commutative rings is in Chapter~\ref{commutative-algebra}.

References are \cite{lang-algebra}, \cite{dummit-foote} and, for modules over
commutative rings, \cite{atiyah-macdonald}.

\noindent\textbf{Prerequisites.} Chapter~\ref{sets} (Set Theory),
Chapter~\ref{number-systems} (Number Systems), Chapter~\ref{categories}
(Categories) and Chapter~\ref{groups} (Groups).

\section{Rings}\label{rings:section-rings}

\begin{definition}\label{rings:definition-ring}
A \emph{ring} is a set $R$ with two operations $+$ and $\cdot$ and elements $0$, $1$
such that $(R, +, 0)$ is an abelian group, multiplication is associative with
neutral element $1$, and both distributive laws $a(b + c) = ab + ac$ and
$(a + b)c = ac + bc$ hold. A \emph{homomorphism} of rings is a map preserving $+$,
$\cdot$ and $1$. A \emph{subring} is a subset containing $0$ and $1$ and closed under
$+$, $-$ and $\cdot$. An element $u$ is a \emph{unit} if it has a two-sided inverse;
the units form a group $R^\times$. A \emph{division ring} is a ring with $1 \neq 0$ in
which every nonzero element is a unit; a commutative division ring is a field.
The rings form a category $\cat{Ring}$, and the commutative rings a full subcategory
$\cat{CRing}$.
\end{definition}

\begin{remark}\label{rings:remark-zero-ring}
The computations of Lemma~\ref{number-systems:lemma-ring-identities} hold in every
ring. If $1 = 0$ then $a = 1 \cdot a = 0 \cdot a = 0$ for all $a$, so $R = \{0\}$ is the
\emph{zero ring}; it is the terminal object of $\cat{Ring}$, and $\ZZ$ is the initial
object (Exercise~\ref{number-systems:exercise-integers-initial}).
\end{remark}

\begin{example}\label{rings:example-rings}
\begin{enumerate}
\item $\ZZ$, $\QQ$, $\RR$, $\CC$, and $\ZZ/n\ZZ$ with the multiplication
$[a][b] = [ab]$, which is well defined since $a - a' \in n\ZZ$ implies
$ab - a'b \in n\ZZ$.
\item For a ring $R$, the $n \times n$ matrices $M_n(R)$ form a ring under matrix
addition and multiplication; it is not commutative for $n \geq 2$ if $1 \neq 0$.
\item For an abelian group $A$, the endomorphisms $\End(A)$ form a ring under
pointwise addition and composition.
\item The product $\prod_i R_i$ of rings with componentwise operations is a
product in $\cat{Ring}$.
\item The \emph{quaternions} $\mathbb{H}$ are the matrices
$\begin{pmatrix} z & -\bar w \\ w & \bar z \end{pmatrix}$ with $z, w \in \CC$. They form a
subring of $M_2(\CC)$, and a division ring: the determinant of such a matrix is
$|z|^2 + |w|^2$, and if it is nonzero the inverse
$\frac{1}{|z|^2 + |w|^2}\begin{pmatrix} \bar z & \bar w \\ -w & z \end{pmatrix}$ is again of this
form. With $i = \begin{pmatrix} i & 0 \\ 0 & -i \end{pmatrix}$,
$j = \begin{pmatrix} 0 & -1 \\ 1 & 0 \end{pmatrix}$ and $k = ij$ we get
$i^2 = j^2 = k^2 = ijk = -1$, so $\mathbb{H}$ is not commutative.
\end{enumerate}
\end{example}

\section{Ideals and quotient rings}\label{rings:section-ideals}

\begin{definition}\label{rings:definition-ideal}
A \emph{left ideal} of a ring $R$ is an additive subgroup $I$ with $ra \in I$ for
$r \in R$, $a \in I$; right ideals are defined similarly, and a (two-sided)
\emph{ideal} is both. For a commutative ring these notions agree. The ideal
generated by a subset $S$ of a commutative ring is
$(S) = \{r_1 s_1 + \cdots + r_n s_n : r_i \in R, s_i \in S\}$; an ideal of the form
$(a)$ is \emph{principal}. For ideals $I$, $J$ of a commutative ring, $I + J$,
$I \cap J$ and the product $IJ$, generated by the products $ab$ with $a \in I$,
$b \in J$, are ideals, and $IJ \subseteq I \cap J$.
\end{definition}

\begin{theorem}\label{rings:theorem-quotient-ring}
Let $I$ be an ideal of $R$. The quotient group $R/I$ is a ring with
$(a + I)(b + I) = ab + I$, and $\pi \colon R \to R/I$ is a surjective ring homomorphism
with kernel $I$. Every ring homomorphism $f \colon R \to S$ with $I \subseteq \ker f$ factors
uniquely as $\bar f \circ \pi$. Kernels of ring homomorphisms are ideals, and
$R/\ker f \cong \im f$. The ideals of $R/I$ correspond to the ideals of $R$ containing
$I$ via $J \mapsto J/I$, and $(R/I)/(J/I) \cong R/J$.
\end{theorem}

\begin{proof}
If $a - a', b - b' \in I$, then $ab - a'b' = a(b - b') + (a - a')b' \in I$, so the
product is well defined; the ring axioms follow from those of $R$. The remaining
statements follow from Theorems~\ref{groups:theorem-quotient}
and~\ref{groups:theorem-isomorphism-theorems} applied to the additive groups,
together with the observation that the maps constructed there preserve products
and that preimages and images of ideals under surjective ring homomorphisms are
ideals.
\end{proof}

\begin{theorem}[Chinese remainder theorem]\label{rings:theorem-chinese-remainder}
Let $R$ be a commutative ring and $I_1, \ldots, I_n$ ideals with $I_i + I_j = R$ for
$i \neq j$. Then $I_1 \cap \cdots \cap I_n = I_1 \cdots I_n$, and the map
$R \to \prod_i R/I_i$ induces an isomorphism $R/(I_1 \cap \cdots \cap I_n) \cong \prod_i R/I_i$.
\end{theorem}

\begin{proof}
For $n = 2$: write $1 = a + b$ with $a \in I_1$, $b \in I_2$. If $x \in I_1 \cap I_2$ then
$x = xa + xb \in I_1I_2$. Given $r_1, r_2 \in R$, the element $r_1 b + r_2 a$ is congruent
to $r_1$ modulo $I_1$ and to $r_2$ modulo $I_2$, so the map is surjective; its kernel
is $I_1 \cap I_2$. For $n > 2$, note that $I_1 + I_2 \cdots I_n = R$: from
$1 = a_j + b_j$ with $a_j \in I_1$, $b_j \in I_j$ we get
$1 = \prod_{j \geq 2}(a_j + b_j) \in I_1 + I_2 \cdots I_n$. Now use induction.
\end{proof}

\begin{example}\label{rings:example-crt-integers}
If $n = p_1^{e_1} \cdots p_k^{e_k}$ with distinct primes $p_i$, then
$\ZZ/n\ZZ \cong \prod_i \ZZ/p_i^{e_i}\ZZ$ as rings, since $(p_i^{e_i}) + (p_j^{e_j}) = \ZZ$ by
Theorem~\ref{number-systems:theorem-bezout}.
\end{example}

\section{Prime and maximal ideals}\label{rings:section-prime-maximal}

In this section all rings are commutative.

\begin{definition}\label{rings:definition-prime-maximal}
An ideal $\mathfrak{p} \neq R$ is \emph{prime} if $ab \in \mathfrak{p}$ implies $a \in \mathfrak{p}$
or $b \in \mathfrak{p}$. An ideal $\mathfrak{m} \neq R$ is \emph{maximal} if there is no ideal $J$
with $\mathfrak{m} \subsetneq J \subsetneq R$.
\end{definition}

\begin{proposition}\label{rings:proposition-prime-maximal-quotients}
An ideal $I$ is prime if and only if $R/I$ is an integral domain, and maximal if and
only if $R/I$ is a field. In particular maximal ideals are prime.
\end{proposition}

\begin{proof}
The first statement is a restatement of the definitions. A commutative ring
$K \neq 0$ is a field if and only if its only ideals are $0$ and $K$: if $a \neq 0$ then
$(a) = K$ means $a$ is a unit. By the correspondence of ideals in
Theorem~\ref{rings:theorem-quotient-ring}, $R/I$ has only the ideals $0$ and $R/I$
exactly when $I$ is maximal.
\end{proof}

\begin{theorem}[Krull]\label{rings:theorem-maximal-ideals-exist}
Every ideal $I \neq R$ is contained in a maximal ideal. In particular every nonzero
ring has a maximal ideal.
\end{theorem}

\begin{proof}
Let $P$ be the set of ideals $J$ with $I \subseteq J \neq R$, ordered by inclusion. It
contains $I$. The union of a nonempty chain in $P$ is an ideal, and it does not
contain $1$, so it lies in $P$. By Zorn's lemma
(Theorem~\ref{sets:theorem-choice-equivalents}) $P$ has a maximal element, which is
a maximal ideal.
\end{proof}

\begin{example}\label{rings:example-ideals-integers}
The ideals of $\ZZ$ are the $n\ZZ$ (Proposition~\ref{groups:proposition-subgroups-integers}).
The prime ideals are $0$ and $p\ZZ$ for $p$ prime, by
Lemma~\ref{number-systems:lemma-euclid}, and the maximal ideals are the $p\ZZ$. Hence
$\ZZ/n\ZZ$ is a field if and only if $n$ is prime; we write $\mathbb{F}_p = \ZZ/p\ZZ$. The
prime ideal $0$ is not maximal.
\end{example}

\begin{definition}\label{rings:definition-characteristic}
The \emph{characteristic} of a ring $R$ is the $n \geq 0$ with $\ker(\ZZ \to R) = n\ZZ$.
\end{definition}

\begin{lemma}\label{rings:lemma-characteristic-domain}
The characteristic of an integral domain is $0$ or a prime number.
\end{lemma}

\begin{proof}
The image of $\ZZ \to R$ is a subring of a domain, hence a domain, so the kernel
is a prime ideal of $\ZZ$.
\end{proof}

\begin{proposition}\label{rings:proposition-fraction-field}
Let $R$ be an integral domain. There is a field $\mathrm{Frac}(R)$ and an injective
homomorphism $R \to \mathrm{Frac}(R)$ such that every injective homomorphism from $R$
to a field $K$ extends uniquely to a homomorphism $\mathrm{Frac}(R) \to K$.
\end{proposition}

\begin{proof}
Construct $\mathrm{Frac}(R)$ as the set of classes $a/b$ of pairs with $b \neq 0$
under $(a, b) \sim (c, d) \iff ad = bc$, with the operations of
Definition~\ref{number-systems:definition-rationals}; the verification of
Theorem~\ref{number-systems:theorem-rationals} only used that $\ZZ$ is an integral
domain. Given $f \colon R \to K$ injective, $f(b) \neq 0$ for $b \neq 0$, and
$a/b \mapsto f(a)f(b)^{-1}$ is the unique extension, as in
Proposition~\ref{number-systems:proposition-rationals-initial}.
\end{proof}

\section{Factorisation in integral domains}\label{rings:section-factorisation}

In this section $R$ is an integral domain.

\begin{definition}\label{rings:definition-irreducible}
Elements $a, b$ are \emph{associates} if $a = ub$ with $u \in R^\times$. A nonzero
nonunit $a$ is \emph{irreducible} if $a = bc$ implies that $b$ or $c$ is a unit, and
\emph{prime} if $a \mid bc$ implies $a \mid b$ or $a \mid c$. The domain $R$ is a
\emph{unique factorisation domain} (UFD) if every nonzero nonunit is a product of
irreducible elements, and this product is unique up to order and associates. It is
a \emph{principal ideal domain} (PID) if every ideal is principal, and a
\emph{Euclidean domain} if there is a map $N \colon R \setminus \{0\} \to \NN$ such that
for all $a$ and $b \neq 0$ there are $q, r$ with $a = qb + r$ and $r = 0$ or $N(r) < N(b)$.
\end{definition}

\begin{lemma}\label{rings:lemma-prime-irreducible}
A nonzero element $p$ is prime if and only if $(p)$ is a prime ideal. Prime elements
are irreducible.
\end{lemma}

\begin{proof}
The first statement is a restatement. If $p$ is prime and $p = bc$, then $p \mid b$ or
$p \mid c$, say $b = pd$; then $p = pdc$, so $dc = 1$ and $c$ is a unit.
\end{proof}

\begin{lemma}\label{rings:lemma-ufd-criterion}
Suppose every nonzero nonunit of $R$ is a product of irreducible elements. Then $R$
is a UFD if and only if every irreducible element is prime.
\end{lemma}

\begin{proof}
If irreducibles are prime, uniqueness follows as in the proof of
Theorem~\ref{number-systems:theorem-fundamental-arithmetic}: a prime dividing a product
of irreducibles divides one of them, hence is associate to it, and we cancel.
Conversely, let $R$ be a UFD, $p$ irreducible and $p \mid bc$, say $bc = pd$. Factoring
$b$, $c$ and $d$ into irreducibles and comparing the two factorisations of $bc$ shows
that $p$ is associate to an irreducible factor of $b$ or of $c$ (if $b$ or $c$ is a unit
or zero the claim is immediate).
\end{proof}

\begin{theorem}\label{rings:theorem-euclidean-pid-ufd}
A Euclidean domain is a PID, and a PID is a UFD.
\end{theorem}

\begin{proof}
Let $R$ be Euclidean and $I \neq 0$ an ideal; choose $b \in I \setminus \{0\}$ with $N(b)$
minimal. For $a \in I$ write $a = qb + r$; then $r \in I$, so $r = 0$ and $I = (b)$.

Let $R$ be a PID. The union of a chain of ideals $(a_1) \subseteq (a_2) \subseteq \cdots$ is
an ideal $(a)$, and $a \in (a_n)$ for some $n$, so the chain is eventually constant.
If some nonzero nonunit $a$ were not a product of irreducibles, then $a$ is not
irreducible, so $a = bc$ with $b, c$ nonunits, one of which, say $a_1$, is not a
product of irreducibles; and $(a) \subsetneq (a_1)$. Repeating gives a strictly increasing
chain of principal ideals, a contradiction. (The repeated choices are made using
the axiom of choice in the form of Theorem~\ref{sets:theorem-recursion} applied to a
choice function.) Finally, if $p$ is irreducible then $(p)$ is maximal: if
$(p) \subseteq (c) \neq R$ then $p = cd$ with $c$ a nonunit, so $d$ is a unit and
$(c) = (p)$. Hence $(p)$ is prime by
Proposition~\ref{rings:proposition-prime-maximal-quotients}, and $R$ is a UFD by
Lemma~\ref{rings:lemma-ufd-criterion}.
\end{proof}

\begin{example}\label{rings:example-gaussian-integers}
The ring $\ZZ[i] = \{a + bi : a, b \in \ZZ\} \subseteq \CC$ of \emph{Gaussian integers} is
Euclidean for $N(z) = |z|^2$. Let $a, b \in \ZZ[i]$, $b \neq 0$, and write
$a/b = u + vi$ with $u, v \in \QQ$. Choose integers $m, n$ with $|u - m| \leq 1/2$ and
$|v - n| \leq 1/2$, and put $q = m + ni$, $r = a - qb \in \ZZ[i]$. Then
$N(r) = |b|^2 |a/b - q|^2 \leq N(b)(1/4 + 1/4) < N(b)$.
\end{example}

\begin{counterexample}\label{rings:counterexample-not-ufd}
The ring $R = \ZZ[\sqrt{-5}] = \{a + b\sqrt{-5} : a, b \in \ZZ\} \subseteq \CC$ is not a UFD. The
norm $N(z) = |z|^2 = a^2 + 5b^2$ is multiplicative and takes values in $\NN$, and the
units are the elements of norm $1$, namely $\pm 1$. No element has norm $2$ or $3$. Hence
$2$ (norm $4$) is irreducible: a factorisation into nonunits would have factors of
norm $2$. But $2$ divides $6 = (1 + \sqrt{-5})(1 - \sqrt{-5})$ without dividing either factor,
since $N(1 \pm \sqrt{-5}) = 6$ is not divisible by $N(2) = 4$. So $2$ is irreducible but not
prime.
\end{counterexample}

\section{Polynomial rings}\label{rings:section-polynomials}

\begin{definition}\label{rings:definition-polynomials}
Let $R$ be a ring. The \emph{polynomial ring} $R[x]$ is the set of sequences
$(a_0, a_1, \ldots)$ in $R$ with only finitely many nonzero terms, with termwise
addition and the product $(a_i)(b_j) = (c_k)$, $c_k = \sum_{i + j = k} a_i b_j$. We
identify $r \in R$ with $(r, 0, 0, \ldots)$ and write $x = (0, 1, 0, \ldots)$, so that
every polynomial is uniquely $f = \sum_{i=0}^n a_i x^i$. If $f \neq 0$, its
\emph{degree} $\deg f$ is the largest $i$ with $a_i \neq 0$, and $a_{\deg f}$ is the
\emph{leading coefficient}; $f$ is \emph{monic} if the leading coefficient is $1$. The
ring $R[x_1, \ldots, x_n]$ is defined inductively as $R[x_1, \ldots, x_{n-1}][x_n]$, and the
\emph{power series ring} $R[[x]]$ is defined like $R[x]$ but with all sequences allowed.
\end{definition}

\begin{proposition}\label{rings:proposition-polynomial-universal}
Let $\varphi \colon R \to S$ be a ring homomorphism and $s \in S$ an element commuting with
all elements of $\varphi(R)$. There is a unique ring homomorphism $R[x] \to S$ extending
$\varphi$ and sending $x$ to $s$, namely $\sum a_i x^i \mapsto \sum \varphi(a_i) s^i$. In particular,
for $R$ commutative, $R[x_1, \ldots, x_n]$ is the free commutative $R$-algebra on
$x_1, \ldots, x_n$: homomorphisms to a commutative ring $S$ extending $\varphi$ correspond
to $n$-tuples of elements of $S$.
\end{proposition}

\begin{proof}
The formula is forced, and it preserves products because $s$ commutes with the
$\varphi(a_i)$: $\bigl(\sum \varphi(a_i)s^i\bigr)\bigl(\sum \varphi(b_j)s^j\bigr) = \sum_k \bigl(\sum_{i+j=k} \varphi(a_i)\varphi(b_j)\bigr)s^k$.
The last statement follows by induction on $n$.
\end{proof}

\begin{notation}\label{rings:notation-evaluation}
For $R$ commutative, $f \in R[x]$ and $a \in R$, we write $f(a)$ for the image of $f$ under
the homomorphism $R[x] \to R$ with $x \mapsto a$. An element $a$ with $f(a) = 0$ is a
\emph{root} of $f$.
\end{notation}

\begin{proposition}\label{rings:proposition-polynomial-domain}
If $R$ is an integral domain, then $\deg(fg) = \deg f + \deg g$ for nonzero $f, g$, so
$R[x]$ is an integral domain and $R[x]^\times = R^\times$.
\end{proposition}

\begin{proof}
The coefficient of $x^{\deg f + \deg g}$ in $fg$ is the product of the leading
coefficients, which is nonzero; all higher coefficients vanish. If $fg = 1$ then
$\deg f = \deg g = 0$.
\end{proof}

\begin{theorem}[Division with remainder]\label{rings:theorem-polynomial-division}
Let $R$ be commutative and $g \in R[x]$ a polynomial whose leading coefficient is a unit.
For every $f \in R[x]$ there are unique $q, r \in R[x]$ with $f = qg + r$ and $r = 0$ or
$\deg r < \deg g$. Consequently, if $k$ is a field then $k[x]$ is Euclidean with
$N = \deg$, hence a PID and a UFD.
\end{theorem}

\begin{proof}
Existence by induction on $\deg f$: if $f = 0$ or $\deg f < \deg g$ take $q = 0$. Otherwise,
with $a$ and $b$ the leading coefficients of $f$ and $g$, the polynomial
$f - ab^{-1}x^{\deg f - \deg g}g$ has smaller degree or is zero. Uniqueness: if
$qg + r = q'g + r'$ with $q \neq q'$, then $(q - q')g = r' - r$ has degree
$\geq \deg g$, because the leading coefficient of $g$ is a unit (so it is not a zero
divisor), which is impossible. The last statement follows from
Theorem~\ref{rings:theorem-euclidean-pid-ufd}.
\end{proof}

\begin{corollary}\label{rings:corollary-roots}
Let $R$ be commutative, $f \in R[x]$ and $a \in R$. Then $f(a) = 0$ if and only if
$(x - a) \mid f$. If $R$ is an integral domain and $f \neq 0$, then $f$ has at most $\deg f$
roots in $R$.
\end{corollary}

\begin{proof}
Divide: $f = q(x - a) + r$ with $r \in R$, and $r = f(a)$. If $R$ is a domain and
$a_1, \ldots, a_m$ are distinct roots, induction shows
$(x - a_1) \cdots (x - a_m) \mid f$: if $f = (x - a_1)g$ then $0 = f(a_j) = (a_j - a_1)g(a_j)$
gives $g(a_j) = 0$ for $j \geq 2$. So $m \leq \deg f$ by
Proposition~\ref{rings:proposition-polynomial-domain}.
\end{proof}

\begin{counterexample}\label{rings:counterexample-many-roots}
The hypothesis that $R$ is a domain is needed: in $\ZZ/8\ZZ$ the polynomial $x^2 - 1$ of
degree $2$ has the four roots $1, 3, 5, 7$.
\end{counterexample}

\begin{proposition}\label{rings:proposition-power-series-units}
Let $R$ be commutative. A power series $f = \sum a_i x^i \in R[[x]]$ is a unit if and only if
$a_0$ is a unit of $R$.
\end{proposition}

\begin{proof}
If $fg = 1$ then $a_0 b_0 = 1$. Conversely, if $a_0$ is a unit define $b_0 = a_0^{-1}$
and recursively $b_k = -a_0^{-1}\sum_{i=1}^k a_i b_{k-i}$; then $\sum_{i+j=k} a_i b_j = 0$ for
$k \geq 1$.
\end{proof}

\begin{definition}\label{rings:definition-content}
Let $R$ be a UFD. A \emph{greatest common divisor} of elements of $R$ is a common
divisor divisible by every common divisor; it exists (take the product of the common
prime factors with their minimal multiplicities) and is unique up to associates. The
\emph{content} $c(f)$ of a nonzero $f \in R[x]$ is a greatest common divisor of its
coefficients, and $f$ is \emph{primitive} if $c(f)$ is a unit.
\end{definition}

\begin{lemma}[Gauss]\label{rings:lemma-gauss}
Let $R$ be a UFD. A product of primitive polynomials is primitive, and
$c(fg) = c(f)c(g)$ up to associates.
\end{lemma}

\begin{proof}
If $f$, $g$ are primitive and a prime $p$ divides all coefficients of $fg$, then the images
of $f$ and $g$ in $(R/(p))[x]$ are nonzero but their product is zero; this is impossible
because $R/(p)$ is a domain, hence so is $(R/(p))[x]$
(Proposition~\ref{rings:proposition-polynomial-domain}). In general write $f = c(f)f_0$ and
$g = c(g)g_0$ with $f_0$, $g_0$ primitive; then $fg = c(f)c(g) f_0 g_0$ with $f_0g_0$
primitive.
\end{proof}

\begin{theorem}\label{rings:theorem-polynomial-ufd}
Let $R$ be a UFD with fraction field $K$.
\begin{enumerate}
\item A primitive polynomial $f \in R[x]$ of positive degree is irreducible in $R[x]$ if
and only if it is irreducible in $K[x]$.
\item $R[x]$ is a UFD, whose prime elements are the prime elements of $R$ and the
primitive polynomials of positive degree that are irreducible in $K[x]$.
\end{enumerate}
Consequently $\ZZ[x_1, \ldots, x_n]$ and $k[x_1, \ldots, x_n]$ for a field $k$ are UFDs.
\end{theorem}

\begin{proof}
Every $h \in K[x]$, $h \neq 0$, can be written $h = a h_0$ with $a \in K^\times$ and
$h_0 \in R[x]$ primitive (clear denominators, then divide by the content), and if also
$h = a' h_0'$ then $h_0' = u h_0$ with $u \in R^\times$: writing $a/a' = b/b'$ with $b, b' \in R$,
$b h_0 = b' h_0'$ and comparing contents gives $b = b'u$.

(1) If $f = gh$ in $K[x]$ with $\deg g, \deg h > 0$, write $g = ag_0$, $h = bh_0$; then
$f = ab\, g_0h_0$ with $g_0h_0$ primitive (Lemma~\ref{rings:lemma-gauss}), so $ab \in R^\times$ by
the uniqueness just shown, and $f = (ab\, g_0)h_0$ is a factorisation in $R[x]$ into
polynomials of positive degree. Conversely a factorisation in $R[x]$ of a primitive
polynomial into nonunits has both factors of positive degree (a constant factor would
divide the content), hence is a factorisation in $K[x]$.

(2) A prime $p$ of $R$ is prime in $R[x]$ since $R[x]/(p) \cong (R/(p))[x]$ is a domain. Let
$f$ be primitive of positive degree and irreducible in $K[x]$, and suppose $f \mid gh$ in
$R[x]$ with $g, h \neq 0$. Since $K[x]$ is a PID, $f$ is prime in $K[x]$, so say $g = fq$ with
$q \in K[x]$. Write $q = aq_0$ with $a = c/c'$, $c, c' \in R$ and $q_0 \in R[x]$ primitive. Then
$c'g = c\,fq_0$, and comparing contents using Lemma~\ref{rings:lemma-gauss} gives
$c = c'c(g)u$ with $u \in R^\times$, so $a \in R$ and $q \in R[x]$. Hence $f$ is prime in $R[x]$.

Every irreducible element of $R[x]$ is of one of these two kinds: an irreducible of degree
$0$ is irreducible in $R$, hence prime in $R$; an irreducible $f$ of positive degree is
primitive (otherwise $c(f)$ would be a nonunit factor), hence irreducible in $K[x]$ by (1).
Every nonzero nonunit of $R[x]$ is a product of irreducibles: write $f = c(f) f_0$, factor
$c(f)$ in $R$, and factor $f_0$ in $K[x]$ into irreducibles, which by the first paragraph
and (1) can be taken primitive in $R[x]$, the remaining constant being a unit of $R$ by
Lemma~\ref{rings:lemma-gauss}. By Lemma~\ref{rings:lemma-ufd-criterion}, $R[x]$ is a UFD.
The last statement follows by induction from $\ZZ$ and $k$ being UFDs.
\end{proof}

\begin{proposition}[Eisenstein's criterion]\label{rings:proposition-eisenstein}
Let $R$ be a UFD with fraction field $K$, $f = a_nx^n + \cdots + a_0 \in R[x]$ with $n \geq 1$,
and $p$ a prime of $R$ with $p \nmid a_n$, $p \mid a_i$ for $i < n$ and $p^2 \nmid a_0$. Then $f$ is
irreducible in $K[x]$.
\end{proposition}

\begin{proof}
By Theorem~\ref{rings:theorem-polynomial-ufd}(1) applied to $f/c(f)$ (note $p \nmid c(f)$) it
suffices to exclude $f = gh$ in $R[x]$ with $\deg g, \deg h \geq 1$. Reducing modulo $p$
gives $\bar g \bar h = \bar a_n x^n$ in $(R/(p))[x]$, which embeds in $L[x]$ for
$L = \mathrm{Frac}(R/(p))$. By unique factorisation in $L[x]$, $\bar g = bx^k$ and
$\bar h = cx^{n-k}$ with $k = \deg g$ and $n - k = \deg h$ both positive. So $p$ divides the
constant terms of $g$ and $h$, and $p^2 \mid a_0$, a contradiction.
\end{proof}

\begin{counterexample}\label{rings:counterexample-ufd-not-pid}
A UFD need not be a PID: in $\ZZ[x]$ the ideal $(2, x)$ is not principal. If
$(2, x) = (f)$, then $f \mid 2$, so $f = \pm 1$ or $f = \pm 2$. But $2 \nmid x$, and
$1 \notin (2, x)$ since every element of $(2, x)$ has even constant term.
\end{counterexample}

\section{Modules}\label{rings:section-modules}

\begin{definition}\label{rings:definition-module}
Let $R$ be a ring. A (left) \emph{$R$-module} is an abelian group $M$ with a map
$R \times M \to M$, $(r, m) \mapsto rm$, such that $r(m + m') = rm + rm'$,
$(r + s)m = rm + sm$, $(rs)m = r(sm)$ and $1m = m$; equivalently, a ring homomorphism
$R \to \End(M)$. A \emph{homomorphism} (or $R$-linear map) is an additive map with
$f(rm) = rf(m)$. \emph{Submodules} are subgroups closed under the action. The set
$\Hom_R(M, N)$ of homomorphisms is an abelian group, and an $R$-module if $R$ is
commutative. The small $R$-modules form a category $R\text{-}\cat{Mod}$.
\end{definition}

\begin{example}\label{rings:example-modules}
\begin{enumerate}
\item $R$ is a module over itself, and its submodules are its left ideals.
\item A $\ZZ$-module is the same thing as an abelian group, by Lemma~\ref{groups:lemma-powers}.
\item A module over a field $k$ is a \emph{vector space}.
\item A $k[x]$-module is a $k$-vector space $V$ together with a $k$-linear map
$T \colon V \to V$ (the action of $x$), by Proposition~\ref{rings:proposition-polynomial-universal}.
\end{enumerate}
\end{example}

\begin{proposition}\label{rings:proposition-module-basics}
Quotients $M/N$ by submodules are modules, and the isomorphism and correspondence
theorems of Theorem~\ref{groups:theorem-isomorphism-theorems} hold for modules. For a
family $(M_i)_{i \in I}$, the product $\prod_i M_i$ and the \emph{direct sum}
$\bigoplus_i M_i = \{(m_i) \in \prod_i M_i : m_i = 0 \text{ for almost all } i\}$ are a product and a
coproduct in $R\text{-}\cat{Mod}$. The category $R\text{-}\cat{Mod}$ is additive
(Definition~\ref{categories:definition-additive}), has kernels and cokernels, and has
all limits and colimits.
\end{proposition}

\begin{proof}
The action on $M/N$ is $r(m + N) = rm + N$, well defined since $N$ is a submodule; the
maps in the isomorphism theorems are $R$-linear. The universal properties of $\prod$ and
$\bigoplus$ are verified componentwise; for $\bigoplus$, a family $f_i \colon M_i \to N$ gives
$(m_i) \mapsto \sum_i f_i(m_i)$, a finite sum. Hom sets are abelian groups with bilinear
composition, $0$ is a zero object, and finite products exist, so the category is
additive. The kernel of $f$ is $\ker f \to M$ and the cokernel is $N \to N/\im f$.
Equalisers of $f, g$ are kernels of $f - g$, so all limits exist by
Theorem~\ref{categories:theorem-limits-products-equalisers}, and dually all colimits
exist.
\end{proof}

\begin{definition}\label{rings:definition-free-module}
For a set $S$, the \emph{free module} $R^{(S)} = \bigoplus_{s \in S} R$ has basis vectors
$e_s$. A family $(m_s)_{s \in S}$ in $M$ \emph{generates} $M$ if every element is a finite
$R$-linear combination of the $m_s$, is \emph{linearly independent} if
$\sum r_s m_s = 0$ implies all $r_s = 0$, and is a \emph{basis} if both hold. $M$ is
\emph{finitely generated} if it has a finite generating family.
\end{definition}

\begin{proposition}\label{rings:proposition-free-module}
For every module $M$ and family $(m_s)_{s \in S}$ in $M$ there is a unique homomorphism
$R^{(S)} \to M$ with $e_s \mapsto m_s$. It is surjective iff the family generates, injective
iff it is linearly independent. Every module is a quotient of a free module, and a
finitely generated module is a quotient of some $R^n$.
\end{proposition}

\begin{proof}
The map is $(r_s) \mapsto \sum r_s m_s$. The other statements are immediate; take the family
of all elements of $M$ (or a finite generating family).
\end{proof}

\begin{definition}\label{rings:definition-exact}
A sequence of homomorphisms $\cdots \to M_{i-1} \xrightarrow{f_{i-1}} M_i \xrightarrow{f_i} M_{i+1} \to \cdots$ is
\emph{exact} at $M_i$ if $\im f_{i-1} = \ker f_i$, and exact if it is exact everywhere. A
\emph{short exact sequence} is an exact sequence $0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$.
\end{definition}

\begin{proposition}[Splitting lemma]\label{rings:proposition-splitting}
For a short exact sequence $0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$ the following are
equivalent:
\begin{enumerate}
\item there is $r \colon B \to A$ with $rf = \id_A$;
\item there is $s \colon C \to B$ with $gs = \id_C$;
\item there is an isomorphism $h \colon B \to A \oplus C$ with $hf(a) = (a, 0)$ and
$g = \mathrm{pr}_C \circ h$.
\end{enumerate}
The sequence is then called \emph{split}. If $C$ is free, the sequence splits.
\end{proposition}

\begin{proof}
(1) $\Rightarrow$ (3): $h = (r, g)$ is injective (if $r(b) = 0$ and $g(b) = 0$ then
$b = f(a)$ and $a = rf(a) = 0$) and surjective (given $(a, c)$, pick $b$ with $g(b) = c$; then
$b - fr(b) + f(a)$ maps to $(a, c)$). (2) $\Rightarrow$ (3): the map
$A \oplus C \to B$, $(a, c) \mapsto f(a) + s(c)$, is bijective by a similar check; take its
inverse. (3) $\Rightarrow$ (1), (2): take $r = \mathrm{pr}_A \circ h$ and $s = h^{-1} \circ \iota_C$.
If $C$ has a basis $(c_s)$, choose preimages $b_s$ and define $s(c_s) = b_s$ using
Proposition~\ref{rings:proposition-free-module}.
\end{proof}

\begin{counterexample}\label{rings:counterexample-non-split}
The sequence $0 \to \ZZ \xrightarrow{2} \ZZ \to \ZZ/2\ZZ \to 0$ of abelian groups is exact but not split,
since $\ZZ$ has no element of order $2$ and so cannot contain a copy of $\ZZ/2\ZZ$.
\end{counterexample}

\section{Vector spaces and rank}\label{rings:section-rank}

\begin{lemma}[Exchange lemma]\label{rings:lemma-exchange}
Let $k$ be a field and $V$ a vector space generated by $v_1, \ldots, v_n$. Then any
$n + 1$ vectors $w_1, \ldots, w_{n+1}$ in $V$ are linearly dependent.
\end{lemma}

\begin{proof}
Induction on $n$. For $n = 0$, $V = 0$ and $w_1 = 0$. For $n \geq 1$ write
$w_j = \sum_{i=1}^n a_{ij} v_i$. If $a_{nj} = 0$ for all $j$, then $w_1, \ldots, w_n$ lie in the
span of $v_1, \ldots, v_{n-1}$ and are dependent by induction. Otherwise, renumbering, let
$a_{n,n+1} \neq 0$ and put $w'_j = w_j - a_{nj}a_{n,n+1}^{-1} w_{n+1}$ for $j \leq n$. These lie in the
span of $v_1, \ldots, v_{n-1}$, so by induction $\sum_{j \leq n} c_j w'_j = 0$ with some $c_j \neq 0$.
Then $\sum_{j \leq n} c_j w_j - \bigl(\sum_{j \leq n} c_j a_{nj} a_{n,n+1}^{-1}\bigr) w_{n+1} = 0$ is a
nontrivial relation.
\end{proof}

\begin{theorem}\label{rings:theorem-rank}
Let $R$ be a nonzero commutative ring. If $R^m \cong R^n$ as $R$-modules, then $m = n$.
In particular any two bases of a finitely generated vector space have the same number
of elements.
\end{theorem}

\begin{proof}
For a field, a basis of $n$ elements generates, so by Lemma~\ref{rings:lemma-exchange}
every linearly independent family has at most $n$ elements; hence $k^m \cong k^n$ implies
$m \leq n$ and $n \leq m$. For general $R$, choose a maximal ideal $\mathfrak{m}$
(Theorem~\ref{rings:theorem-maximal-ideals-exist}). An isomorphism $R^m \cong R^n$ maps
$\mathfrak{m}R^m$ onto $\mathfrak{m}R^n$, so it induces an isomorphism
$(R/\mathfrak{m})^m \cong R^m/\mathfrak{m}R^m \cong R^n/\mathfrak{m}R^n \cong (R/\mathfrak{m})^n$ of vector
spaces over the field $R/\mathfrak{m}$.
\end{proof}

\begin{definition}\label{rings:definition-rank}
For $R$ nonzero commutative, the \emph{rank} of a free module $R^n$ is $n$.
\end{definition}

\begin{counterexample}\label{rings:counterexample-rank-noncommutative}
Theorem~\ref{rings:theorem-rank} fails for noncommutative rings. Let $k$ be a field,
$V = k^{(\NN)}$ and $R = \End_k(V)$. Splitting $\NN$ into even and odd numbers gives
$V \cong V \oplus V$. As a left $R$-module (by post-composition),
$R = \Hom_k(V, V) \cong \Hom_k(V \oplus V, V) \cong \Hom_k(V, V) \oplus \Hom_k(V, V) = R^2$.
\end{counterexample}

\section{Tensor products}\label{rings:section-tensor}

In this section $R$ is a commutative ring.

\begin{definition}\label{rings:definition-tensor}
Let $M$, $N$, $P$ be $R$-modules. A map $\beta \colon M \times N \to P$ is \emph{bilinear} if it is
$R$-linear in each variable. A \emph{tensor product} of $M$ and $N$ is a module
$M \otimes_R N$ with a bilinear map $\otimes \colon M \times N \to M \otimes_R N$,
$(m, n) \mapsto m \otimes n$, such that every bilinear $\beta \colon M \times N \to P$ factors as
$\beta = \tilde\beta \circ \otimes$ for a unique linear $\tilde\beta$.
\end{definition}

\begin{proposition}\label{rings:proposition-tensor-exists}
Tensor products exist and are unique up to unique isomorphism. The elements
$m \otimes n$ generate $M \otimes_R N$.
\end{proposition}

\begin{proof}
Uniqueness is the usual argument for universal properties. Let $F = R^{(M \times N)}$ and
$Q \subseteq F$ the submodule generated by the elements
$e_{(m+m', n)} - e_{(m,n)} - e_{(m',n)}$, $e_{(m, n+n')} - e_{(m,n)} - e_{(m,n')}$,
$e_{(rm, n)} - re_{(m,n)}$ and $e_{(m, rn)} - re_{(m,n)}$. Put $M \otimes_R N = F/Q$ and
$m \otimes n = e_{(m,n)} + Q$. A bilinear $\beta$ induces a linear map $F \to P$,
$e_{(m,n)} \mapsto \beta(m,n)$, vanishing on $Q$, hence a unique $\tilde\beta$. The elements
$m \otimes n$ generate since the $e_{(m,n)}$ generate $F$.
\end{proof}

\begin{proposition}\label{rings:proposition-tensor-properties}
There are natural isomorphisms $R \otimes_R M \cong M$, $M \otimes N \cong N \otimes M$,
$(M \otimes N) \otimes P \cong M \otimes (N \otimes P)$ and
$(\bigoplus_i M_i) \otimes N \cong \bigoplus_i (M_i \otimes N)$. In particular
$R^m \otimes R^n \cong R^{mn}$. For an ideal $I$, $(R/I) \otimes_R M \cong M/IM$.
\end{proposition}

\begin{proof}
Each isomorphism is obtained by constructing maps in both directions from bilinear
maps and checking that the composites fix generators $m \otimes n$. For instance
$(r, m) \mapsto rm$ induces $R \otimes M \to M$, inverse to $m \mapsto 1 \otimes m$. For
associativity, for fixed $p$ the map $(m, n) \mapsto m \otimes (n \otimes p)$ is bilinear, giving
$(m \otimes n) \mapsto m \otimes (n \otimes p)$; this is bilinear in $(m \otimes n, p)$, and an inverse
is constructed symmetrically. For direct sums, $((m_i), n) \mapsto (m_i \otimes n)$ is bilinear,
and the inverse is induced by the maps $M_i \otimes N \to (\bigoplus M_i) \otimes N$. Finally,
$(\bar r, m) \mapsto \overline{rm}$ induces $R/I \otimes M \to M/IM$, with inverse
$\bar m \mapsto \bar 1 \otimes m$, well defined since $1 \otimes am = \bar a \otimes m = 0$ for $a \in I$.
\end{proof}

\begin{theorem}[Tensor--hom adjunction]\label{rings:theorem-tensor-hom}
For $R$-modules $M$, $N$, $P$ there is a bijection, natural in all three variables,
\[
\Hom_R(M \otimes_R N, P) \cong \Hom_R(M, \Hom_R(N, P)), \qquad f \mapsto (m \mapsto (n \mapsto f(m \otimes n))).
\]
So $- \otimes_R N$ is left adjoint to $\Hom_R(N, -)$. Consequently $- \otimes_R N$ preserves
colimits, and for every exact sequence $A \to B \to C \to 0$ the sequence
$A \otimes N \to B \otimes N \to C \otimes N \to 0$ is exact.
\end{theorem}

\begin{proof}
Both sides are in bijection with bilinear maps $M \times N \to P$, naturally. The functor
$- \otimes_R N$ preserves colimits by Theorem~\ref{categories:theorem-adjoints-preserve-limits}.
A sequence $A \xrightarrow{u} B \xrightarrow{v} C \to 0$ is exact if and only if $v$ is a cokernel of $u$,
that is, a coequaliser of $u$ and $0$; this is preserved.
\end{proof}

\begin{example}\label{rings:example-tensor-cyclic}
$\ZZ/m\ZZ \otimes_\ZZ \ZZ/n\ZZ \cong \ZZ/d\ZZ$ with $d = \gcd(m, n)$, since by
Proposition~\ref{rings:proposition-tensor-properties} it is
$(\ZZ/n\ZZ)/m(\ZZ/n\ZZ) = \ZZ/(m, n) = \ZZ/d\ZZ$. In particular
$\ZZ/2\ZZ \otimes \ZZ/3\ZZ = 0$, and $\QQ \otimes_\ZZ \ZZ/n\ZZ = 0$ since $x \otimes \bar 1 = nx/n \otimes \bar 1 = x/n \otimes \bar n = 0$.
\end{example}

\begin{counterexample}\label{rings:counterexample-tensor-not-left-exact}
Tensoring does not preserve injectivity. The map $2 \colon \ZZ \to \ZZ$ is injective, but
tensoring with $\ZZ/2\ZZ$ gives the zero map $\ZZ/2\ZZ \to \ZZ/2\ZZ$. Similarly
$\Hom_R(N, -)$ does not preserve surjectivity: $\ZZ \to \ZZ/2\ZZ$ is surjective but
$\Hom(\ZZ/2\ZZ, \ZZ) = 0 \to \Hom(\ZZ/2\ZZ, \ZZ/2\ZZ) \cong \ZZ/2\ZZ$ is not.
\end{counterexample}

\begin{definition}\label{rings:definition-flat}
An $R$-module $N$ is \emph{flat} if $- \otimes_R N$ preserves injective homomorphisms.
Free modules are flat, by Proposition~\ref{rings:proposition-tensor-properties}.
\end{definition}

\begin{proposition}[Extension of scalars]\label{rings:proposition-extension-scalars}
Let $\varphi \colon R \to S$ be a homomorphism of commutative rings. For an $R$-module $M$,
$S \otimes_R M$ is an $S$-module via $s(s' \otimes m) = ss' \otimes m$, and for every $S$-module $N$
(an $R$-module via $\varphi$) there is a natural bijection
$\Hom_S(S \otimes_R M, N) \cong \Hom_R(M, N)$, $f \mapsto (m \mapsto f(1 \otimes m))$.
\end{proposition}

\begin{proof}
For $s \in S$, multiplication by $s$ on $S$ is $R$-linear, so it induces an endomorphism of
$S \otimes_R M$; this gives the module structure. Given an $R$-linear $g \colon M \to N$, the map
$(s, m) \mapsto sg(m)$ is $R$-bilinear and induces an $S$-linear map $S \otimes M \to N$; the two
constructions are inverse because $1 \otimes m$ generate $S \otimes M$ as an $S$-module.
\end{proof}

\section{Algebras and graded rings}\label{rings:section-algebras}

\begin{definition}\label{rings:definition-algebra}
Let $R$ be commutative. An \emph{$R$-algebra} is a ring $A$ with a ring homomorphism
$R \to A$ whose image lies in the center of $A$. Homomorphisms of $R$-algebras are ring
homomorphisms compatible with the maps from $R$. A \emph{graded ring} is a ring
$A = \bigoplus_{n \in \ZZ} A_n$ (as abelian group) with $A_m A_n \subseteq A_{m+n}$; elements of $A_n$ are
\emph{homogeneous of degree $n$}. A \emph{graded module} over $A$ is a module
$M = \bigoplus_n M_n$ with $A_m M_n \subseteq M_{m+n}$. An ideal of a graded ring is \emph{homogeneous} if
it is generated by homogeneous elements.
\end{definition}

\begin{proposition}\label{rings:proposition-tensor-algebras}
Let $A$ and $B$ be commutative $R$-algebras. There is a unique ring structure on
$A \otimes_R B$ with $(a \otimes b)(a' \otimes b') = aa' \otimes bb'$, and with the maps
$a \mapsto a \otimes 1$ and $b \mapsto 1 \otimes b$ it is a coproduct of $A$ and $B$ in the category of
commutative $R$-algebras.
\end{proposition}

\begin{proof}
For fixed $(a', b')$ the map $(a, b) \mapsto aa' \otimes bb'$ is bilinear, inducing
$\mu_{a',b'} \colon A \otimes B \to A \otimes B$, and $(a', b') \mapsto \mu_{a',b'}$ is bilinear, inducing a
bilinear multiplication on $A \otimes B$ with the required formula. Associativity,
commutativity and the unit $1 \otimes 1$ hold on generators, hence everywhere by
bilinearity. Given algebra homomorphisms $f \colon A \to C$, $g \colon B \to C$ to a commutative
algebra, the bilinear map $(a, b) \mapsto f(a)g(b)$ induces a linear map
$A \otimes B \to C$, which is multiplicative on generators and hence a ring homomorphism;
it is unique since the elements $a \otimes b = (a \otimes 1)(1 \otimes b)$ generate.
\end{proof}

\begin{example}\label{rings:example-graded}
The polynomial ring $R[x_1, \ldots, x_n]$ is graded by total degree, with $A_d$ the span of
the monomials of degree $d$. By Proposition~\ref{rings:proposition-tensor-algebras} and
Proposition~\ref{rings:proposition-polynomial-universal},
$R[x] \otimes_R R[y] \cong R[x, y]$. Graded rings of this kind are the homogeneous coordinate
rings of projective varieties (Chapter~\ref{varieties}).
\end{example}

\begin{lemma}\label{rings:lemma-homogeneous-ideal}
An ideal $I$ of a graded commutative ring $A$ is homogeneous if and only if, for every
$f \in I$, all homogeneous components of $f$ lie in $I$.
\end{lemma}

\begin{proof}
If $I$ is generated by homogeneous $g_j$ and $f = \sum_j a_j g_j \in I$, then the degree-$n$
component of $f$ is $\sum_j (a_j)_{n - \deg g_j} g_j \in I$. Conversely, if $I$ contains the
components of its elements, it is generated by these components.
\end{proof}

\section{Modules over a principal ideal domain}\label{rings:section-pid}

In this section $R$ is a principal ideal domain.

\begin{proposition}\label{rings:proposition-submodules-free}
Every submodule of $R^n$ is free of rank at most $n$.
\end{proposition}

\begin{proof}
The proof of Proposition~\ref{groups:proposition-subgroups-free} applies verbatim, using
that the image of a submodule under the last coordinate is an ideal $(d)$ of $R$.
\end{proof}

\begin{theorem}[Smith normal form]\label{rings:theorem-smith}
Let $M$ be an $m \times n$ matrix over $R$. There are invertible matrices $P$, $Q$ over $R$
such that $PMQ$ is diagonal with entries $d_1, \ldots, d_r, 0, \ldots, 0$, where
$d_i \neq 0$ and $d_1 \mid d_2 \mid \cdots \mid d_r$.
\end{theorem}

\begin{proof}
Since $R$ is a UFD (Theorem~\ref{rings:theorem-euclidean-pid-ufd}), each nonzero $a \in R$ has a
\emph{length} $\ell(a)$, the number of prime factors counted with multiplicity; if $d \mid a$
and $d$ is not associate to $a$, then $\ell(d) < \ell(a)$. Besides row and column swaps and
adding multiples of one row (column) to another, we use the following operation: for
entries $a, b$ in the same row, in columns $i$ and $j$, write $d = \gcd(a, b) = xa + yb$
(possible since $(a, b) = (d)$) and replace columns $i, j$ by
$x\,\mathrm{col}_i + y\,\mathrm{col}_j$ and $-(b/d)\mathrm{col}_i + (a/d)\mathrm{col}_j$. This is
multiplication by an invertible matrix (its $2 \times 2$ block has determinant $1$), and it
replaces $a, b$ by $d, 0$. The same operation exists for rows.

Now argue as in Theorem~\ref{groups:theorem-smith}, with $\ell$ in place of the absolute
value. Among all matrices obtained from $M$ by these operations choose one whose
nonzero entry of least length is as short as possible, and move it to position $(1,1)$.
If it did not divide an entry in the first row or column, the operation above would
produce a proper divisor, of smaller length. So it divides these entries, and we clear
the first row and column. If it did not divide some other entry, adding that row to the
first row and applying the operation would again produce a shorter entry. Apply
induction to the remaining block.
\end{proof}

\begin{theorem}[Structure theorem]\label{rings:theorem-structure-pid}
Let $M$ be a finitely generated $R$-module. There are $r \geq 0$ and nonzero nonunits
$d_1 \mid d_2 \mid \cdots \mid d_k$ in $R$ with
\[
M \cong R^r \oplus R/(d_1) \oplus \cdots \oplus R/(d_k),
\]
and $r$ and the ideals $(d_1), \ldots, (d_k)$ are uniquely determined by $M$.
\end{theorem}

\begin{proof}
Existence is proved exactly as in Theorem~\ref{groups:theorem-structure-abelian}, using
Proposition~\ref{rings:proposition-submodules-free} and Theorem~\ref{rings:theorem-smith}.
For uniqueness, the torsion submodule $T = \{m : am = 0 \text{ for some } a \neq 0\}$ is
$\bigoplus R/(d_i)$, and $M/T \cong R^r$, so $r$ is determined by Theorem~\ref{rings:theorem-rank}.
Let $p$ be a prime of $R$ and $j \geq 0$. For $d \neq 0$ with $v = v_p(d)$, the $R/(p)$-vector
space $p^j(R/(d))/p^{j+1}(R/(d))$ is one-dimensional if $v \geq j + 1$ and zero otherwise:
indeed $p^j(R/(d)) = (p^{\min(j, v)})/(d)$. Hence $\dim_{R/(p)} p^jT/p^{j+1}T$ is the number of
$i$ with $p^{j+1} \mid d_i$, which by Theorem~\ref{rings:theorem-rank} depends only on $T$. As in
the proof of Theorem~\ref{groups:theorem-structure-abelian}, these numbers determine $k$
and the $p$-adic valuations of all $d_i$, hence the $d_i$ up to units.
\end{proof}

\begin{corollary}\label{rings:corollary-primary-decomposition}
Every finitely generated torsion $R$-module is a direct sum of modules
$R/(p^e)$ with $p$ prime and $e \geq 1$, and the multiset of the ideals $(p^e)$ is
uniquely determined.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{rings:theorem-chinese-remainder} to $R/(d_i)$ with the prime factorisation
of $d_i$; uniqueness follows from the uniqueness in Theorem~\ref{rings:theorem-structure-pid},
since the $d_i$ are recovered from the prime powers by taking, for each $p$, the largest
powers for $d_k$, the next largest for $d_{k-1}$, and so on.
\end{proof}

\section{Exercises}\label{rings:section-exercises}

\begin{exercise}\label{rings:exercise-matrix-ring-simple}
Let $k$ be a field. Show that $M_n(k)$ has no two-sided ideals other than $0$ and
$M_n(k)$, although it has many left ideals for $n \geq 2$.
\end{exercise}

\begin{exercise}\label{rings:exercise-gaussian-quotient}
Show that $\ZZ[x]/(x^2 + 1) \cong \ZZ[i]$, and that $\RR[x]/(x^2 + 1) \cong \CC$.
\end{exercise}

\begin{exercise}\label{rings:exercise-units-mod-n}
Show that $(\ZZ/n\ZZ)^\times$ consists of the classes of integers coprime to $n$, and
deduce Fermat's little theorem: $a^{p-1} \equiv 1 \pmod p$ for $p$ prime and $p \nmid a$.
\end{exercise}

\begin{solution}
If $\gcd(a, n) = 1$ then $ax + ny = 1$ by Theorem~\ref{number-systems:theorem-bezout}, so
$[a][x] = 1$. If $d = \gcd(a, n) > 1$ then $[a][n/d] = 0$ with $[n/d] \neq 0$, so $[a]$ is a zero
divisor and not a unit. For Fermat, apply Theorem~\ref{groups:theorem-lagrange} to the group
$\mathbb{F}_p^\times$ of order $p - 1$.
\end{solution}

\end{exercise}

\begin{exercise}\label{rings:exercise-jordan-preview}

\end{exercise}
