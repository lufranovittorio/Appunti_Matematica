\chapter{Cohomology Operations}\label{cohomology-operations}
\chapterstatus{complete}

\section{Introduction}\label{cohomology-operations:section-introduction}

A cohomology operation is a natural transformation between cohomology functors, such as $x \mapsto x^2$. Because cohomology is
represented by Eilenberg--MacLane spaces, operations are the same as cohomology classes of these spaces
(Proposition~\ref{cohomology-operations:proposition-operations-representable}). The Steenrod squares are the most important
operations with $\mathbb{F}_2$-coefficients; they refine the cup product and give obstructions that cup products cannot see, such as
the theorem that a map of Hopf invariant one $S^{2n-1} \to S^n$ exists only if $n$ is a power of $2$
(Corollary~\ref{cohomology-operations:corollary-hopf-invariant}). For the Hodge conjecture, operations show that some integral
cohomology classes of smooth projective varieties cannot be classes of algebraic cycles
(Remark~\ref{cohomology-operations:remark-atiyah-hirzebruch}). This chapter is partly a survey: the existence theorems are quoted
from \cite{hatcher-at}, \cite{steenrod-epstein} and \cite{mosher-tangora}, and what follows from them is proved.

\noindent\textbf{Prerequisites.} Chapter~\ref{poincare-duality} (Topological Manifolds and Poincar\'e Duality) and Chapter~\ref{homotopy}
(Homotopy and Covering Spaces).

\noindent\textbf{On a first reading.} Later chapters use this one only for the Stiefel--Whitney classes and Wu's formula in Chapter~\ref{characteristic-classes}; it can be
skipped until then.

\section{Eilenberg--MacLane spaces}\label{cohomology-operations:section-eilenberg-maclane}

\begin{definition}\label{cohomology-operations:definition-eilenberg-maclane}
Let $G$ be a group and $n \geq 1$, with $G$ abelian if $n \geq 2$. An \emph{Eilenberg--MacLane space} $K(G, n)$ is a connected CW complex with $\pi_n(K(G, n)) \cong G$ and
$\pi_i(K(G, n)) = 0$ for $i \neq n$.
\end{definition}

\begin{theorem}\label{cohomology-operations:theorem-eilenberg-maclane-exist}
For every $n \geq 1$ and every group $G$ (abelian if $n \geq 2$) there is an Eilenberg--MacLane space $K(G, n)$, which can be chosen with a single $0$-cell and no other cells
of dimension less than $n$; any two are homotopy equivalent.
\end{theorem}

For the proof, which attaches cells to a wedge of $n$-spheres to realise a presentation of $G$ and then kills the higher homotopy groups, see \cite[Section 4.2]{hatcher-at}.

\begin{example}\label{cohomology-operations:example-eilenberg-maclane}
\begin{enumerate}
\item $S^1$ is a $K(\ZZ, 1)$, since its universal covering $\RR$ is contractible and $\pi_i(S^1) \cong \pi_i(\RR) = 0$ for $i \geq 2$
(Proposition~\ref{homotopy:proposition-covering-higher}). More generally, a connected CW complex with contractible universal covering is a $K(\pi_1, 1)$; this applies
to the tori $T^n$ and to the surfaces $\Sigma_g$ with $g \geq 1$.
\item $\RR\PP^\infty = \bigcup_n \RR\PP^n$ is a $K(\ZZ/2\ZZ, 1)$ and $\CC\PP^\infty$ is a $K(\ZZ, 2)$: they are the quotients of the contractible space
$S^\infty = \bigcup_n S^n$ by $\{\pm 1\}$ and by the circle group, and the long exact sequence of the fibration $S^1 \to S^\infty \to \CC\PP^\infty$ gives
$\pi_2(\CC\PP^\infty) \cong \pi_1(S^1)$ \cite[Section 4.2]{hatcher-at}.
\end{enumerate}
\end{example}

\begin{theorem}[Representability]\label{cohomology-operations:theorem-representability}
Let $G$ be abelian and $n \geq 1$. There is a class $\iota_n \in H^n(K(G, n); G)$ such that for every CW complex $X$ the map $\langle X, K(G, n) \rangle \to H^n(X; G)$,
$[f] \mapsto f^*(\iota_n)$, is a bijection, natural in $X$.
\end{theorem}

This is \cite[Theorem 4.57]{hatcher-at}. The class $\iota_n$ corresponds under $H^n(K(G, n); G) \cong \Hom(H_n(K(G, n)), G) \cong \Hom(\pi_n(K(G, n)), G) = \Hom(G, G)$
(universal coefficients and the Hurewicz theorem) to the identity.

\section{Cohomology operations}\label{cohomology-operations:section-operations}

\begin{definition}\label{cohomology-operations:definition-operation}
A \emph{cohomology operation} of type $(G, m; H, n)$ is a family of maps of sets $\theta_X \colon H^m(X; G) \to H^n(X; H)$, one for each space $X$, such that
$f^*\theta_Y = \theta_X f^*$ for every map $f \colon X \to Y$. It is \emph{additive} if every $\theta_X$ is a homomorphism.
\end{definition}

\begin{proposition}\label{cohomology-operations:proposition-operations-representable}
Restricted to pointed CW complexes, operations of type $(G, m; H, n)$ correspond bijectively to elements of $H^n(K(G, m); H)$, via $\theta \mapsto \theta(\iota_m)$.
\end{proposition}

\begin{proof}
On the category whose objects are pointed CW complexes and whose morphisms are pointed homotopy classes of maps, $H^m(-; G)$ is represented by $(K(G, m), \iota_m)$, by
Theorem~\ref{cohomology-operations:theorem-representability} and Definition~\ref{categories:definition-representable}, and $H^n(-; H)$ is a functor to $\Sets$ by
homotopy invariance. Operations are natural transformations between them, and the Yoneda lemma (Theorem~\ref{categories:theorem-yoneda}) identifies these with
$H^n(K(G, m); H)$. Explicitly, $u \in H^n(K(G, m); H)$ gives $\theta(x) = f_x^*(u)$, where $f_x \colon X \to K(G, m)$ is the map with $f_x^*(\iota_m) = x$.
\end{proof}

\begin{example}\label{cohomology-operations:example-operations}
\begin{enumerate}
\item For a commutative ring $R$, $x \mapsto x^k$ is an operation of type $(R, m; R, km)$, corresponding to $\iota_m^k$; it is not additive in general.
\item A homomorphism $G \to H$ induces an additive operation of type $(G, m; H, m)$, for example the reduction $\rho \colon H^m(X; \ZZ) \to H^m(X; \ZZ/p\ZZ)$.
\item Since $H^n(K(G, m); H) = 0$ for $0 < n < m$ ($K(G, m)$ may be chosen with no cells in dimensions $0 < i < m$), there are no nonzero operations lowering degree, except
constant ones for $n = 0$.
\end{enumerate}
\end{example}

\begin{proposition}[Bockstein homomorphisms]\label{cohomology-operations:proposition-bockstein}
Let $p$ be a prime. The short exact sequences $0 \to \ZZ \xrightarrow{p} \ZZ \to \ZZ/p \to 0$ and $0 \to \ZZ/p \to \ZZ/p^2 \to \ZZ/p \to 0$ induce connecting homomorphisms
$\tilde\beta \colon H^n(X; \ZZ/p) \to H^{n+1}(X; \ZZ)$ and $\beta \colon H^n(X; \ZZ/p) \to H^{n+1}(X; \ZZ/p)$, the \emph{Bockstein} operations. They satisfy $\beta = \rho\tilde\beta$,
$\beta\beta = 0$, the image of $\tilde\beta$ is the $p$-torsion subgroup $\{y : py = 0\}$ of $H^{n+1}(X; \ZZ)$, and the kernel of $\tilde\beta$ is the image of the reduction
$\rho \colon H^n(X; \ZZ) \to H^n(X; \ZZ/p)$.
\end{proposition}

\begin{proof}
Since each $C_n(X)$ is free, $\Hom(C_n(X), -)$ is exact, so a short exact sequence of coefficient groups gives a short exact sequence of cochain complexes and a long exact
sequence (Theorem~\ref{homological-algebra:theorem-long-exact-sequence}):
$H^n(X; \ZZ) \xrightarrow{p} H^n(X; \ZZ) \xrightarrow{\rho} H^n(X; \ZZ/p) \xrightarrow{\tilde\beta} H^{n+1}(X; \ZZ) \xrightarrow{p} H^{n+1}(X; \ZZ)$. Exactness gives the descriptions of the image
and the kernel of $\tilde\beta$. The reduction $\ZZ \to \ZZ/p^2$ gives a morphism from the first short exact sequence to the second, and naturality of connecting homomorphisms
gives $\beta = \rho\tilde\beta$. Finally $\beta\beta = \rho\tilde\beta\rho\tilde\beta = 0$ because $\tilde\beta\rho = 0$.
\end{proof}

\section{Steenrod squares}\label{cohomology-operations:section-steenrod-squares}

\begin{theorem}[Steenrod]\label{cohomology-operations:theorem-steenrod-squares}
There are unique additive operations $\mathrm{Sq}^i \colon H^n(X; \mathbb{F}_2) \to H^{n+i}(X; \mathbb{F}_2)$, $i \geq 0$, defined for all spaces and pairs and natural, such that:
\begin{enumerate}
\item $\mathrm{Sq}^0 = \id$;
\item $\mathrm{Sq}^n(x) = x^2$ if $|x| = n$, and $\mathrm{Sq}^i(x) = 0$ if $i > |x|$;
\item (Cartan formula) $\mathrm{Sq}^i(xy) = \sum_{j+k=i}\mathrm{Sq}^j(x)\mathrm{Sq}^k(y)$;
\item they commute with the suspension isomorphisms $\tilde{H}^n(X) \cong \tilde{H}^{n+1}(SX)$ and with the connecting homomorphisms of pairs.
\end{enumerate}
Moreover $\mathrm{Sq}^1$ is the Bockstein $\beta$ for $p = 2$, and for $a < 2b$ the \emph{Adem relations}
\[
\mathrm{Sq}^a\mathrm{Sq}^b = \sum_{c=0}^{\lfloor a/2 \rfloor}\binom{b - c - 1}{a - 2c}\mathrm{Sq}^{a+b-c}\mathrm{Sq}^c
\]
hold, with binomial coefficients taken modulo $2$.
\end{theorem}

See \cite{steenrod-epstein} and \cite[Section 4.L]{hatcher-at}. Writing $\mathrm{Sq} = \sum_i \mathrm{Sq}^i$ (a finite sum on each class), the Cartan formula says that
$\mathrm{Sq} \colon H^*(X; \mathbb{F}_2) \to H^*(X; \mathbb{F}_2)$ is a ring homomorphism. There are analogous \emph{reduced power operations}
$P^i \colon H^n(X; \mathbb{F}_p) \to H^{n+2i(p-1)}(X; \mathbb{F}_p)$ for odd primes $p$, with $P^i(x) = x^p$ for $|x| = 2i$.

\begin{proposition}\label{cohomology-operations:proposition-squares-projective}
In $H^*(\RR\PP^\infty; \mathbb{F}_2) = \mathbb{F}_2[\alpha]$, $\mathrm{Sq}^i(\alpha^n) = \binom{n}{i}\alpha^{n+i}$. In $H^*(\CC\PP^\infty; \mathbb{F}_2) = \mathbb{F}_2[x]$ with $|x| = 2$,
$\mathrm{Sq}^{2i}(x^n) = \binom{n}{i}x^{n+i}$ and $\mathrm{Sq}^{2i+1} = 0$.
\end{proposition}

\begin{proof}
By axioms (1) and (2), $\mathrm{Sq}(\alpha) = \alpha + \alpha^2$. Since $\mathrm{Sq}$ is a ring homomorphism,
$\mathrm{Sq}(\alpha^n) = (\alpha + \alpha^2)^n = \alpha^n(1 + \alpha)^n = \sum_i \binom{n}{i}\alpha^{n+i}$; compare degrees. For $\CC\PP^\infty$, $\mathrm{Sq}^1(x) \in H^3 = 0$, so
$\mathrm{Sq}(x) = x + x^2$ and the same computation applies, all odd squares vanishing for degree reasons. The rings are those of
Theorem~\ref{singular-cohomology:theorem-projective-spaces}.
\end{proof}

\begin{lemma}\label{cohomology-operations:lemma-decomposable}
If $i \geq 1$ is not a power of $2$, then $\mathrm{Sq}^i$ is \emph{decomposable}: it is a sum of composites $\mathrm{Sq}^a\mathrm{Sq}^b$ with $0 < a, b < i$.
\end{lemma}

\begin{proof}
Choose $k$ with $2^k < i < 2^{k+1}$ and let $a = i - 2^k$, so $0 < a < 2^k$ and $a < 2 \cdot 2^k$. In $\mathbb{F}_2[t]$ we have $(1 + t)^{2^k} = 1 + t^{2^k}$ (the Frobenius map is a
ring homomorphism), hence $(1 + t)^{2^k - 1} = (1 + t^{2^k})/(1 + t) = 1 + t + \cdots + t^{2^k - 1}$, so $\binom{2^k - 1}{a}$ is odd. The Adem relation for $\mathrm{Sq}^a\mathrm{Sq}^{2^k}$
has $c = 0$ term $\binom{2^k - 1}{a}\mathrm{Sq}^i = \mathrm{Sq}^i$, so $\mathrm{Sq}^i = \mathrm{Sq}^a\mathrm{Sq}^{2^k} + \sum_{c \geq 1}\binom{2^k - c - 1}{a - 2c}\mathrm{Sq}^{i-c}\mathrm{Sq}^c$, and all indices
lie strictly between $0$ and $i$.
\end{proof}

\begin{corollary}[Hopf invariant one]\label{cohomology-operations:corollary-hopf-invariant}
Let $n \geq 2$ and $f \colon S^{2n-1} \to S^n$, and let $X_f = S^n \cup_f D^{2n}$, with $H^n(X_f; \ZZ) = \ZZ a$ and $H^{2n}(X_f; \ZZ) = \ZZ b$. Define the \emph{Hopf invariant} $h(f)$ by
$a^2 = h(f)b$. If $h(f)$ is odd, then $n$ is a power of $2$.
\end{corollary}

\begin{proof}
$X_f$ has cells in dimensions $0$, $n$, $2n$, so by cellular cohomology its integral cohomology is free of rank one in these degrees and zero elsewhere, and the reductions
$\bar{a}, \bar{b}$ generate the $\mathbb{F}_2$-cohomology. Reduction is a ring homomorphism, so $\mathrm{Sq}^n(\bar a) = \bar a^2 = h(f)\bar b \neq 0$. If $n$ were not a power of $2$,
Lemma~\ref{cohomology-operations:lemma-decomposable} would write $\mathrm{Sq}^n(\bar a)$ as a sum of terms $\mathrm{Sq}^a\mathrm{Sq}^b(\bar a)$ with $0 < b < n$, and
$\mathrm{Sq}^b(\bar a) \in H^{n+b}(X_f; \mathbb{F}_2) = 0$.
\end{proof}

\begin{remark}\label{cohomology-operations:remark-adams}
The Hopf maps $S^3 \to S^2$, $S^7 \to S^4$ and $S^{15} \to S^8$ have Hopf invariant one. Adams proved, using secondary cohomology operations, that maps of Hopf invariant
one exist only for $n = 1, 2, 4, 8$ \cite{adams-1960}. Consequences include the facts that $\RR^n$ admits a bilinear
multiplication without zero divisors only for $n = 1, 2, 4, 8$, and that $S^{n-1}$ is an $H$-space only for $n = 1, 2, 4, 8$.
\end{remark}

\section{Cohomology of Eilenberg--MacLane spaces}\label{cohomology-operations:section-cohomology-km}

\begin{definition}\label{cohomology-operations:definition-admissible}
A sequence $I = (i_1, \ldots, i_r)$ of positive integers is \emph{admissible} if $i_j \geq 2i_{j+1}$ for all $j$. Its \emph{degree} is $\sum_j i_j$, its \emph{excess} is
$e(I) = \sum_j(i_j - 2i_{j+1}) = i_1 - i_2 - \cdots - i_r$ (with $i_{r+1} = 0$), and $\mathrm{Sq}^I = \mathrm{Sq}^{i_1} \cdots \mathrm{Sq}^{i_r}$. The empty sequence is admissible with
excess $0$.
\end{definition}

\begin{theorem}[Serre]\label{cohomology-operations:theorem-serre}
\begin{enumerate}
\item The \emph{Steenrod algebra} $\mathcal{A}_2$, the algebra of operations $H^*(-; \mathbb{F}_2) \to H^*(-; \mathbb{F}_2)$ generated by the $\mathrm{Sq}^i$ modulo the Adem relations, has
the admissible monomials $\mathrm{Sq}^I$ as an $\mathbb{F}_2$-basis.
\item $H^*(K(\ZZ/2, n); \mathbb{F}_2)$ is the polynomial algebra on the classes $\mathrm{Sq}^I\iota_n$ with $I$ admissible and $e(I) < n$.
\item $H^*(K(\ZZ, n); \mathbb{F}_2)$ is the polynomial algebra on the classes $\mathrm{Sq}^I\iota_n$ with $I$ admissible, $e(I) < n$ and $i_r \neq 1$.
\end{enumerate}
\end{theorem}

This is \cite{serre-1953}; see also \cite{mosher-tangora}.

\begin{example}\label{cohomology-operations:example-k-z-3}
For $n = 3$, the admissible sequences of excess less than $3$ not ending in $1$ are $\emptyset$, $(2)$, $(4, 2)$, $(8, 4, 2)$, \ldots. So
$H^*(K(\ZZ, 3); \mathbb{F}_2) = \mathbb{F}_2[\iota_3, \mathrm{Sq}^2\iota_3, \mathrm{Sq}^4\mathrm{Sq}^2\iota_3, \ldots]$ with generators in degrees $3, 5, 9, 17, \ldots$. In low degrees:
$H^3 = \langle \iota_3 \rangle$, $H^4 = 0$, $H^5 = \langle \mathrm{Sq}^2\iota_3 \rangle$, $H^6 = \langle \iota_3^2 \rangle$, where $\iota_3^2 = \mathrm{Sq}^3\iota_3 = \mathrm{Sq}^1\mathrm{Sq}^2\iota_3$ by the Adem
relation $\mathrm{Sq}^1\mathrm{Sq}^2 = \mathrm{Sq}^3$. Since $\mathrm{Sq}^1 = \rho\tilde\beta$ is nonzero on $\mathrm{Sq}^2\iota_3$, the integral class $\tilde\beta\mathrm{Sq}^2\iota_3 \in H^6(K(\ZZ, 3); \ZZ)$ is a nonzero
$2$-torsion class. In general $x \mapsto \tilde\beta\mathrm{Sq}^2\rho(x)$ is an integral operation raising degree by $3$, defined in every degree, whose reduction modulo $2$ is
$\rho\tilde\beta\mathrm{Sq}^2\rho(x) = \mathrm{Sq}^1\mathrm{Sq}^2\rho(x) = \mathrm{Sq}^3\rho(x)$; it appears in Remark~\ref{cohomology-operations:remark-atiyah-hirzebruch}.
\end{example}

\section{Complex cobordism and algebraic cycles}\label{cohomology-operations:section-cobordism}

\begin{remark}\label{cohomology-operations:remark-cobordism}
\emph{Complex cobordism} $MU^*$ is a generalised cohomology theory: it satisfies all Eilenberg--Steenrod axioms except the dimension axiom. Its coefficient ring is
$MU^* = MU^*(\mathrm{pt}) \cong \ZZ[x_1, x_2, \ldots]$ with $x_i$ in degree $-2i$, generated geometrically by classes of complex projective spaces and hypersurfaces
\cite{stong-cobordism}, and Quillen showed that it carries the universal one-dimensional commutative formal group law \cite{quillen-1969}. For a compact complex manifold
$X$ there is a natural map $MU^*(X) \to H^*(X; \ZZ)$, and a closed complex submanifold $Z \subseteq X$ of codimension $c$ defines a class in $MU^{2c}(X)$ whose image is the cohomology class
of $Z$ (Chapter~\ref{poincare-duality}). See \cite{ravenel-green} for the homotopy-theoretic side and \cite{thom-1954} for the origins of cobordism.
\end{remark}

\begin{remark}\label{cohomology-operations:remark-atiyah-hirzebruch}
Atiyah and Hirzebruch showed that on a smooth projective complex variety $X$ the integral cohomology class of every algebraic cycle lies in the image of $MU^*(X) \to H^*(X; \ZZ)$,
using resolution of singularities, and that cohomology operations of odd degree vanish on this image; in particular $\mathrm{Sq}^3\rho(x) = 0$ for the class $x$ of a codimension-$2$
cycle. They constructed smooth projective varieties with torsion classes $x \in H^4(X; \ZZ)$ with $\mathrm{Sq}^3\rho(x) \neq 0$. Such a class is a Hodge class (torsion classes are
Hodge classes trivially), but it is not the class of an algebraic cycle, which disproves the Hodge conjecture with integral coefficients \cite{atiyah-hirzebruch-1962}. Totaro
refined this by showing that the cycle class map factors as $CH^*(X) \to MU^*(X) \otimes_{MU^*} \ZZ \to H^*(X; \ZZ)$ \cite{totaro-1997}. These results are developed in
Chapter~\ref{failures}.
\end{remark}

\section{Exercises}\label{cohomology-operations:section-exercises}

\begin{exercise}\label{cohomology-operations:exercise-adem-small}
Deduce from the Adem relations that $\mathrm{Sq}^1\mathrm{Sq}^1 = 0$, $\mathrm{Sq}^1\mathrm{Sq}^2 = \mathrm{Sq}^3$ and $\mathrm{Sq}^2\mathrm{Sq}^2 = \mathrm{Sq}^3\mathrm{Sq}^1$.
\end{exercise}

\begin{solution}
For $(a, b) = (1, 1)$ the only term is $c = 0$ with coefficient $\binom{0}{1} = 0$. For $(1, 2)$: $\binom{1}{1}\mathrm{Sq}^3\mathrm{Sq}^0 = \mathrm{Sq}^3$. For $(2, 2)$: $c = 0$ gives $\binom{1}{2} = 0$ and
$c = 1$ gives $\binom{0}{0}\mathrm{Sq}^3\mathrm{Sq}^1$.
\end{solution}

\begin{exercise}\label{cohomology-operations:exercise-polynomial-degree-three}
Show that there is no space $X$ with $H^*(X; \mathbb{F}_2) \cong \mathbb{F}_2[x]$ and $|x| = 3$.
\end{exercise}

\begin{solution}
We would have $x^2 = \mathrm{Sq}^3x = \mathrm{Sq}^1\mathrm{Sq}^2x$, and $\mathrm{Sq}^2x \in H^5(X; \mathbb{F}_2) = 0$, so $x^2 = 0$, contradicting that $H^*$ is a polynomial ring. (More generally, the argument of Corollary~\ref{cohomology-operations:corollary-hopf-invariant} shows that
$|x|$ must be a power of $2$, and by Adams' theorem (Remark~\ref{cohomology-operations:remark-adams}) only $|x| = 1, 2, 4, 8$ can occur.)
\end{solution}

\begin{exercise}\label{cohomology-operations:exercise-bockstein-rp}
Show that $\beta(\alpha^n) = n\alpha^{n+1}$ in $H^*(\RR\PP^\infty; \mathbb{F}_2)$, directly from the cellular cochain complex, and compare with
Proposition~\ref{cohomology-operations:proposition-squares-projective}.
\end{exercise}

\begin{exercise}\label{cohomology-operations:exercise-cp2-attaching}
Show that the attaching map $S^3 \to S^2$ of the $4$-cell of $\CC\PP^2$ is not null-homotopic, using $\mathrm{Sq}^2$ or the cup product, and deduce that $\pi_3(S^2) \neq 0$.
\end{exercise}

\begin{solution}
If it were null-homotopic, $\CC\PP^2$ would be homotopy equivalent to $S^2 \vee S^4$ (attaching along homotopic maps gives homotopy equivalent spaces \cite[Proposition 0.18]{hatcher-at}),
whose cup squares in degree $2$ vanish; but $\mathrm{Sq}^2(x) = x^2 \neq 0$ in $H^4(\CC\PP^2; \mathbb{F}_2)$ by Theorem~\ref{singular-cohomology:theorem-projective-spaces}.
\end{solution}
