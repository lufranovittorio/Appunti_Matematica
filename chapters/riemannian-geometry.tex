\section{Introduction}\label{riemannian-geometry:section-introduction}

A Riemannian metric turns a manifold into a metric space, produces a canonical connection (Theorem~\ref{riemannian-geometry:theorem-levi-civita})
and, together with an orientation, defines the Hodge star, the codifferential and the Laplacian
(Definition~\ref{riemannian-geometry:definition-laplacian}). Harmonic forms are the kernel of that Laplacian, and the Hodge theorem
(Chapter~\ref{hodge-theorem}) says that on a compact manifold they represent de Rham cohomology. Along the way we develop the Riemannian measure and the volume form, which make surface integrals and the divergence theorem instances of integration on
manifolds; the local and global theory of geodesics: covariant derivatives along curves,
the Gauss lemma, the fact that geodesics are locally the shortest curves, and the Hopf--Rinow theorem, all with complete proofs; parallel transport and holonomy, with the holonomy principle, and the
theorems of Ambrose--Singer, de Rham and Berger; Jacobi fields, conjugate points and the first comparison theorems; the first and second variation of energy, with Jacobi's theorem that
geodesics stop minimising after a conjugate point, a complete proof of the Bonnet--Myers theorem, the theorems of Weinstein and Synge on manifolds of positive curvature, and the splitting theorem; the geometry of submanifolds, with the
first and second fundamental forms, the Weingarten equations and the equations of Gauss and Codazzi; the classical global theorems on surfaces: Gauss--Bonnet for
regions with boundary, the curvature of conformal metrics, and the theorems of Minding, Hadamard, Liebmann and Hilbert; minimal surfaces, with the first and second variation of area, the
area-minimising property of minimal graphs and Bernstein's theorem; and isometries and Killing fields, with Bochner's theorem, the theorem of Killing
and Hopf classifying the spaces of constant curvature, and the statements of Cheng's maximal diameter theorem and of the sphere theorem. References:
\cite{do-carmo-riemannian}, \cite{lee-riemannian}, \cite{kobayashi-nomizu}, \cite{joyce-special-holonomy}, \cite{warner-foundations}, \cite[Chapter 0]{griffiths-harris}.

\noindent\textbf{Prerequisites.} Chapter~\ref{connections} (Connections and Curvature).

\noindent\textbf{Conventions.} $(M, g)$ is a Riemannian manifold of dimension $n$, and we write $\langle v, w\rangle = g(v, w)$ and $|v| = \langle v, v\rangle^{1/2}$. Curves are piecewise smooth unless stated otherwise. In local
coordinates we use the summation convention: an index repeated once up and once down is summed over.

\section{Metrics and the Levi-Civita connection}\label{riemannian-geometry:section-levi-civita}

\begin{definition}\label{riemannian-geometry:definition-metric}
A \emph{Riemannian metric} on a smooth manifold $M$ is a smooth section $g$ of $\mathrm{Sym}^2T^*M$ that is an inner product on each tangent space; $(M, g)$ is a \emph{Riemannian manifold}. The
\emph{length} of a piecewise smooth curve is $L(\gamma) = \int_a^b|\dot\gamma(t)|\,dt$ with $|v| = g(v, v)^{1/2}$, and $d(p, q)$ is the infimum of the lengths of curves from $p$ to $q$. An \emph{isometry} is a
diffeomorphism preserving $g$.
\end{definition}

\begin{proposition}\label{riemannian-geometry:proposition-metric-exists}
Every smooth manifold admits a Riemannian metric, and $d$ is a metric inducing the topology of $M$.
\end{proposition}

\begin{proof}
Patch the Euclidean metrics of charts with a partition of unity (Theorem~\ref{smooth-manifolds:theorem-partitions-of-unity}); a convex combination of inner products is an inner product. Symmetry and the
triangle inequality for $d$ are clear, and $d(p, q) > 0$ for $p \neq q$ because in a chart around $p$ the metric is comparable to the Euclidean one on a compact neighbourhood, so curves leaving a small ball
have length bounded below. The same comparison shows that the $d$-balls form a basis of the topology.
\end{proof}

\begin{example}\label{riemannian-geometry:example-metrics}
The Euclidean metric on $\RR^n$; the induced metric on a submanifold, for instance the round metric on $S^n \subseteq \RR^{n+1}$; the hyperbolic metric $(dx^2 + dy^2)/y^2$ on the upper half plane; a
left-invariant metric on a Lie group, obtained from any inner product on $\mathfrak{g}$, which is bi-invariant when $G$ is compact (average over Haar measure,
Theorem~\ref{lie-groups:theorem-haar}); and the Fubini--Study metric on $\CC\PP^n$ (Chapter~\ref{kahler}).
\end{example}

\begin{proposition}\label{riemannian-geometry:proposition-quotient-metric}
Let a discrete group $G$ act freely and properly on $M$ by isometries of $g$. Then $M/G$ carries a unique Riemannian metric for which $\pi \colon M \to M/G$ is a local isometry.
\end{proposition}

\begin{proof}
By Theorem~\ref{smooth-manifolds:theorem-smooth-quotient}, $\pi$ is a local diffeomorphism. For $x \in M$ define the metric on $T_{\pi(x)}(M/G)$ by transporting $g_x$ through $d_x\pi$. This does not depend on the choice of $x$
in its orbit: $\pi \circ h = \pi$ for $h \in G$, so $d_{hx}\pi \circ d_xh = d_x\pi$, and $d_xh$ is an isometry $T_xM \to T_{hx}M$. On an open set $\pi(U)$ over which $\pi$ has the smooth inverse $(\pi|_U)^{-1}$ the metric is the pullback
of $g$ by this inverse, hence smooth. Uniqueness is clear.
\end{proof}

\begin{example}\label{riemannian-geometry:example-quotient-metrics}
The \emph{flat torus} $\RR^n/\ZZ^n$ and the \emph{flat Klein bottle} $\RR^2/G$ (Example~\ref{topological-manifolds:example-covering-actions}), since translations and the maps $(x, y) \mapsto (x + \frac12, -y)$ are Euclidean isometries; and
the round metric on $\RR\PP^n = S^n/\{\pm1\}$, as $-\mathrm{id}$ is an isometry of $S^n$.
\end{example}

\begin{theorem}[Fundamental theorem of Riemannian geometry]\label{riemannian-geometry:theorem-levi-civita}
On a Riemannian manifold there is a unique connection $\nabla$ on $TM$ that is metric ($Xg(Y, Z) = g(\nabla_XY, Z) + g(Y, \nabla_XZ)$) and torsion-free ($\nabla_XY - \nabla_YX = [X, Y]$): the
\emph{Levi-Civita connection}. It is determined by the Koszul formula
\[
2g(\nabla_XY, Z) = Xg(Y, Z) + Yg(X, Z) - Zg(X, Y) + g([X, Y], Z) - g([X, Z], Y) - g([Y, Z], X).
\]
In coordinates, $\nabla_{\partial_i}\partial_j = \sum_k\Gamma^k_{ij}\partial_k$ with $\Gamma^k_{ij} = \frac12\sum_lg^{kl}(\partial_ig_{jl} + \partial_jg_{il} - \partial_lg_{ij})$.
\end{theorem}

\begin{proof}
If $\nabla$ is metric and torsion-free, expanding the three terms $Xg(Y, Z)$, $Yg(X, Z)$, $-Zg(X, Y)$ with the metric property and using torsion-freeness to replace $\nabla_XY - \nabla_YX$ by $[X, Y]$ gives the
Koszul formula; since $g$ is nondegenerate, the formula determines $\nabla_XY$, which proves uniqueness. Conversely, define $\nabla_XY$ by the Koszul formula. The right-hand side is $\RR$-bilinear,
$C^\infty$-linear in $X$ and $Z$ and satisfies the Leibniz rule in $Y$ (the terms with derivatives of $f$ in $Y \mapsto fY$ combine to $2(Xf)g(Y, Z)$), so $\nabla$ is a connection; exchanging $X$ and $Y$ shows it
is torsion-free, and adding the formulas for $g(\nabla_XY, Z)$ and $g(Y, \nabla_XZ)$ gives $Xg(Y, Z)$. The coordinate formula follows by taking $X, Y, Z$ coordinate fields, whose brackets vanish.
\end{proof}

In coordinates, torsion-freeness says $\Gamma^k_{ij} = \Gamma^k_{ji}$, and the metric property for coordinate fields says
\[
\partial_ig_{jk} = \Gamma^l_{ij}g_{lk} + \Gamma^l_{ik}g_{jl}. \tag{$*$}
\]

\begin{definition}\label{riemannian-geometry:definition-geodesic}
A \emph{geodesic} is a curve with $\nabla_{\dot\gamma}\dot\gamma = 0$, where the velocity $\dot\gamma$ is differentiated along $\gamma$ as in
Remark~\ref{connections:remark-along-curves} (it need not extend to a vector field on $M$; see
Remark~\ref{riemannian-geometry:remark-derivative-along-curves}); in coordinates, $\ddot\gamma^k + \sum_{i,j}\Gamma^k_{ij}\dot\gamma^i\dot\gamma^j = 0$. By
Theorem~\ref{real-analysis:theorem-picard-lindelof} there is a unique maximal geodesic with given initial position and velocity, depending smoothly on them; the \emph{exponential map}
$\exp_p(v) = \gamma_v(1)$ is defined on a neighbourhood of $0 \in T_pM$.
\end{definition}

\section{Volume and integration}\label{riemannian-geometry:section-volume}

A metric measures the lengths of tangent vectors, hence the volumes of small parallelepipeds, and so it determines how to integrate functions over the manifold.
For a surface in $\RR^3$ this is the classical surface integral $\iint f\,dA$; the point of this section is that it is an instance of the integration of forms of
Chapter~\ref{differential-forms}, and that the same formulas work for submanifolds of any dimension. In a chart $(U, \varphi)$ with coordinates $x^1, \ldots, x^n$ we write
$g_{ij} = g(\partial_i, \partial_j)$ and $\det g$ for the determinant of the matrix $(g_{ij})$, a smooth positive function on $U$; $\lambda$ is Lebesgue measure.

\begin{lemma}\label{riemannian-geometry:lemma-metric-change-of-chart}
Let $y^1, \ldots, y^n$ be the coordinates of a second chart, with metric coefficients $\tilde g_{ab} = g(\partial/\partial y^a, \partial/\partial y^b)$. Then on the
intersection of the two chart domains
\[
\tilde g_{ab} = \sum_{i,j}\frac{\partial x^i}{\partial y^a}\frac{\partial x^j}{\partial y^b}\,g_{ij}, \qquad
\sqrt{\det\tilde g} = \Bigl|\det\Bigl(\frac{\partial x^i}{\partial y^a}\Bigr)\Bigr|\sqrt{\det g}.
\]
\end{lemma}

\begin{proof}
By Proposition~\ref{smooth-manifolds:proposition-change-of-coordinates}, $\partial/\partial y^a = \sum_i(\partial x^i/\partial y^a)\,\partial/\partial x^i$; expanding
$g$ bilinearly gives the first formula. In matrices it reads $\tilde g = P^TgP$ with $P = (\partial x^i/\partial y^a)$, and taking determinants,
$\det\tilde g = (\det P)^2\det g$.
\end{proof}

\begin{proposition}[Riemannian measure]\label{riemannian-geometry:proposition-riemannian-measure}
Let $(M, g)$ be a Riemannian manifold. There is a unique measure $\mu_g$ on the Borel sets of $M$ such that for every chart $(U, \varphi)$ and every Borel set
$A \subseteq U$,
\[
\mu_g(A) = \int_{\varphi(A)}\sqrt{\det g} \circ \varphi^{-1}\,d\lambda .
\]
Consequently $\int_Af\,d\mu_g = \int_{\varphi(A)}(f\sqrt{\det g}) \circ \varphi^{-1}\,d\lambda$ for every measurable $f \geq 0$ and every integrable $f$ on $A$. We write $dV$ for
$\mu_g$, and $dA$ and $ds$ in dimensions $2$ and $1$ (the \emph{volume}, the \emph{area} and the \emph{arc length}), and $\mathrm{vol}(M) = \mu_g(M)$.
\end{proposition}

\begin{proof}
For a chart $(U, \varphi)$ the formula defines a measure $\mu_U$ on the Borel subsets of $U$, the image under the homeomorphism $\varphi^{-1}$ of the measure with density
$\sqrt{\det g} \circ \varphi^{-1}$ on $\varphi(U)$. Two such measures agree where both are defined: let $A$ be a Borel subset of $U \cap V$ for a second chart
$(V, \psi)$ with coordinates $y$, and let $\tau = \varphi \circ \psi^{-1}$, whose Jacobian matrix is $(\partial x^i/\partial y^a)$. By
Lemma~\ref{riemannian-geometry:lemma-metric-change-of-chart}, $\sqrt{\det\tilde g} \circ \psi^{-1} = (\sqrt{\det g} \circ \varphi^{-1} \circ \tau)\,|\det D\tau|$, so by
the change of variables formula (Theorem~\ref{measure-theory:theorem-change-of-variables}) applied to $\tau \colon \psi(A) \to \varphi(A)$,
\[
\int_{\psi(A)}\sqrt{\det\tilde g} \circ \psi^{-1}\,d\lambda = \int_{\psi(A)}(\sqrt{\det g} \circ \varphi^{-1} \circ \tau)\,|\det D\tau|\,d\lambda
= \int_{\varphi(A)}\sqrt{\det g} \circ \varphi^{-1}\,d\lambda .
\]
Cover $M$ by countably many chart domains $U_1, U_2, \ldots$ ($M$ is second countable), let $B_1 = U_1$ and $B_k = U_k \setminus (U_1 \cup \cdots \cup U_{k-1})$, and put
$\mu_g(A) = \sum_k\mu_{U_k}(A \cap B_k)$. This is a measure, being a countable sum of measures. If $A$ is a Borel subset of a chart domain $U$, the
compatibility just proved gives $\mu_{U_k}(A \cap B_k) = \mu_U(A \cap B_k)$, so $\mu_g(A) = \sum_k\mu_U(A \cap B_k) = \mu_U(A)$. Conversely, any measure with the
stated property assigns to $A$ the value $\sum_k\mu_{U_k}(A \cap B_k)$, which proves uniqueness. The formula for integrals follows from the one for sets, first
for simple functions, then by monotone convergence (Theorem~\ref{measure-theory:theorem-monotone-convergence}) for $f \geq 0$, and by linearity for integrable $f$.
\end{proof}

\begin{proposition}[Riemannian volume form]\label{riemannian-geometry:proposition-volume-form}
Let $(M, g)$ be an oriented Riemannian $n$-manifold. There is a unique $n$-form $\mathrm{vol}_g$ with $\mathrm{vol}_g(e_1, \ldots, e_n) = 1$ for every positively
oriented orthonormal basis $(e_1, \ldots, e_n)$ of a tangent space. In a positively oriented chart
\[
\mathrm{vol}_g = \sqrt{\det g}\;dx^1 \wedge \cdots \wedge dx^n,
\]
and $\int_Mf\,\mathrm{vol}_g = \int_Mf\,d\mu_g$ for every continuous function $f$ with compact support.
\end{proposition}

\begin{proof}
Let $p$ lie in a positively oriented chart, and let $(e_1, \ldots, e_n)$ be a positively oriented orthonormal basis of $T_pM$. Write $e_a = \sum_iP^i_a\partial_i$ with an
invertible matrix $P = (P^i_a)$. Then $\delta_{ab} = \langle e_a, e_b\rangle = (P^TgP)_{ab}$, so $(\det P)^2\det g = 1$, and $\det P > 0$ because both bases are
positively oriented; hence $\det P = 1/\sqrt{\det g}$ and
\[
\bigl(\sqrt{\det g}\;dx^1 \wedge \cdots \wedge dx^n\bigr)(e_1, \ldots, e_n) = \sqrt{\det g}\,\det\bigl(dx^i(e_a)\bigr) = \sqrt{\det g}\,\det P = 1 .
\]
Since $\Lambda^nT^*_pM$ is one-dimensional, an $n$-form at $p$ is determined by its value on one basis. So on each positively oriented chart domain the smooth form
$\sqrt{\det g}\,dx^1 \wedge \cdots \wedge dx^n$ is the only form with the required property; the forms of two charts therefore agree on the overlap, and together they
define the smooth $n$-form $\mathrm{vol}_g$, which is unique. If $f$ is supported in a positively oriented chart, both integrals equal
$\int(f\sqrt{\det g}) \circ \varphi^{-1}\,d\lambda$, by Definition~\ref{differential-forms:definition-integral} and
Proposition~\ref{riemannian-geometry:proposition-riemannian-measure}; in general write $f = \sum_i\psi_if$ with a partition of unity subordinate to positively oriented
charts and use linearity of both sides.
\end{proof}

For a surface $\mathrm{vol}_g$ is the \emph{area form} $dA$. By Proposition~\ref{differential-forms:proposition-orientation}, a non-orientable manifold, such as the
M\"obius band or $\RR\PP^2$, has no nowhere vanishing $n$-form, hence no volume form; but it still has a Riemannian measure, so areas and integrals of functions make sense
on it. The integral of a function needs a metric, the integral of an $n$-form needs an orientation, and the volume form converts one into the other.

\begin{proposition}[Volume of parametrised submanifolds]\label{riemannian-geometry:proposition-parametrised-volume}
Let $M \subseteq \RR^N$ be a $k$-dimensional submanifold with the induced metric, and $x \colon \Omega \to M$ a parametrisation
(Proposition~\ref{smooth-manifolds:proposition-submanifold-criteria}(3)), with $x_i = \partial x/\partial u^i$. Then for every measurable $f \geq 0$ on $x(\Omega)$,
\[
\int_{x(\Omega)}f\,d\mu = \int_\Omega f(x(u))\,\mathcal{J}(u)\,du, \qquad
\mathcal{J} = \sqrt{\det\bigl(\langle x_i, x_j\rangle\bigr)} = \Bigl(\sum_I\Bigl(\det\frac{\partial x^I}{\partial u}\Bigr)^2\Bigr)^{1/2},
\]
where $I$ runs over the subsets $\{i_1 < \cdots < i_k\}$ of $\{1, \ldots, N\}$, and $\partial x^I/\partial u$ is the $k \times k$ matrix formed by the rows
$i_1, \ldots, i_k$ of the Jacobian matrix $Dx$. In particular:
\begin{enumerate}
\item for a curve ($k = 1$), $\mathcal{J} = |x^{\prime}(t)|$, and $\int f\,ds = \int f(x(t))\,|x^{\prime}(t)|\,dt$;
\item for a surface in $\RR^3$, with $E = |x_u|^2$, $F = \langle x_u, x_v\rangle$ and $G = |x_v|^2$, $\mathcal{J} = \sqrt{EG - F^2} = |x_u \times x_v|$;
\item for a hypersurface ($k = N - 1$) with unit normal $\nu$, $\mathcal{J} = |\det(\nu, x_1, \ldots, x_k)|$;
\item for the graph of a smooth function $\phi \colon \Omega \to \RR$, parametrised by $x(u) = (u, \phi(u))$, $\mathcal{J} = \sqrt{1 + |\nabla\phi|^2}$.
\end{enumerate}
\end{proposition}

\begin{proof}
The inverse $x^{-1}$ is a chart of $M$ (Lemma~\ref{smooth-manifolds:lemma-submanifold-structure}), whose coordinate vector field $\partial/\partial u^i$ is $x_i$;
so $g_{ij} = \langle x_i, x_j\rangle$ and the formula is Proposition~\ref{riemannian-geometry:proposition-riemannian-measure}. For the second expression of
$\mathcal{J}$, Proposition~\ref{multilinear-algebra:proposition-induced-inner-product} gives $\det(\langle x_i, x_j\rangle) = |x_1 \wedge \cdots \wedge x_k|^2$ in
$\Lambda^k\RR^N$. Writing $x_j = \sum_i(\partial x^i/\partial u^j)e_i$ and expanding by multilinearity, the terms with a repeated $e_i$ vanish, and collecting the
terms with the same set of indices $I$ (the Leibniz formula for the determinant) gives $x_1 \wedge \cdots \wedge x_k = \sum_I\det(\partial x^I/\partial u)\,e_I$; as the
$e_I$ are orthonormal, $|x_1 \wedge \cdots \wedge x_k|^2 = \sum_I\det(\partial x^I/\partial u)^2$.

(1) is the case of a single vector. (2) $EG - F^2 = |x_u|^2|x_v|^2 - \langle x_u, x_v\rangle^2 = |x_u \times x_v|^2$ (Lagrange's identity); equivalently, the three
$2 \times 2$ minors of $Dx$ are, up to sign, the components of $x_u \times x_v$. (3) The matrix $P$ with columns $\nu, x_1, \ldots, x_k$ satisfies
$P^TP = \bigl(\begin{smallmatrix} 1 & 0\\ 0 & (\langle x_i, x_j\rangle)\end{smallmatrix}\bigr)$, since $|\nu| = 1$ and $\langle\nu, x_i\rangle = 0$; so
$(\det P)^2 = \det(\langle x_i, x_j\rangle)$. (4) Here $x_i = (e_i, \partial_i\phi)$, so $\langle x_i, x_j\rangle = \delta_{ij} + \partial_i\phi\,\partial_j\phi$: this is
the matrix $I + ww^T$ with $w = \nabla\phi$, which is the identity on $w^\perp$ and multiplies $w$ by $1 + |w|^2$, so its determinant is $1 + |w|^2$.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-surface-integrals}
In classical surface theory \cite[Chapter 2]{do-carmo-curves-surfaces} the area of a region $R$ contained in the image of a parametrisation $x$, and the integral of a
function over it, are defined by
\[
\iint_Rf\,dA = \iint_{x^{-1}(R)}f(x(u, v))\,|x_u \times x_v|\,du\,dv,
\]
and one checks by hand that this does not depend on the parametrisation. By Proposition~\ref{riemannian-geometry:proposition-parametrised-volume}(2) this is the
integral with respect to the Riemannian measure of the induced metric: surface integration is integration on a manifold, and nothing new. The independence of the
parametrisation is Lemma~\ref{riemannian-geometry:lemma-metric-change-of-chart} together with the change of variables formula, and a region that is not contained in
one parametrisation is cut into pieces, or treated with a partition of unity, as in Definition~\ref{differential-forms:definition-integral}. When the surface is oriented,
$\iint_Rf\,dA = \int_Rf\,\mathrm{vol}_S$ is the integral of the $2$-form $f\,\mathrm{vol}_S$ (Proposition~\ref{riemannian-geometry:proposition-volume-form}). The same
holds in every dimension and codimension: the length of a curve (Definition~\ref{riemannian-geometry:definition-metric}) is its one-dimensional volume, by (1), and the
$k$-dimensional volume of a piece of a $k$-dimensional submanifold of $\RR^N$ is the integral of the square root of the sum of the squares of the $k \times k$ minors.
\end{remark}

For hypersurfaces of $\RR^{n+1}$ the volume form has a simple expression through the unit normal, and it turns integrals of $n$-forms into flux integrals. We write
$\Omega = dx^1 \wedge \cdots \wedge dx^{n+1}$ for the standard volume form of $\RR^{n+1}$, and, for a vector field $F$ defined along $M$ (for instance the unit normal
$\nu$), $(\iota_F\Omega)|_M$ for the $n$-form on $M$ whose value at $p$ is the restriction of $\iota_{F(p)}\Omega_p$ to $T_pM$.

\begin{proposition}[Volume form of a hypersurface; flux]\label{riemannian-geometry:proposition-flux}
Let $M \subseteq \RR^{n+1}$ be a hypersurface oriented by a unit normal field $\nu$ (Proposition~\ref{differential-forms:proposition-orientation-normal}). Then
\[
\mathrm{vol}_M = (\iota_\nu\Omega)|_M = \Bigl(\sum_{i=1}^{n+1}(-1)^{i-1}\nu^i\,dx^1 \wedge \cdots \wedge \widehat{dx^i} \wedge \cdots \wedge dx^{n+1}\Bigr)\Big|_M,
\]
and for every vector field $F$ along $M$, $(\iota_F\Omega)|_M = \langle F, \nu\rangle\,\mathrm{vol}_M$. In $\RR^3$, for a surface $S$:
$dA = (\nu^1\,dy \wedge dz + \nu^2\,dz \wedge dx + \nu^3\,dx \wedge dy)|_S$, and the \emph{flux} of $F$ through $S$ is
\[
\int_SF^1\,dy \wedge dz + F^2\,dz \wedge dx + F^3\,dx \wedge dy = \iint_S\langle F, \nu\rangle\,dA .
\]
\end{proposition}

\begin{proof}
By Proposition~\ref{differential-forms:proposition-orientation-normal} the orientation of $M$ is that of the form
$(v_1, \ldots, v_n) \mapsto \det(\nu, v_1, \ldots, v_n) = \Omega(\nu, v_1, \ldots, v_n) = (\iota_\nu\Omega)(v_1, \ldots, v_n)$. If $(e_1, \ldots, e_n)$ is a positively
oriented orthonormal basis of $T_pM$, then $(\nu(p), e_1, \ldots, e_n)$ is an orthonormal basis of $\RR^{n+1}$, positive by the same proposition, so its determinant is
$1$. By the uniqueness in Proposition~\ref{riemannian-geometry:proposition-volume-form}, $(\iota_\nu\Omega)|_M = \mathrm{vol}_M$. The coordinate expression is
$\iota_\nu(dx^1 \wedge \cdots \wedge dx^{n+1}) = \sum_i(-1)^{i-1}dx^i(\nu)\,dx^1 \wedge \cdots \wedge \widehat{dx^i} \wedge \cdots \wedge dx^{n+1}$, which follows from
Proposition~\ref{multilinear-algebra:proposition-interior-antiderivation} by induction on the number of factors. For $F$, write $F = \langle F, \nu\rangle\nu + F^\top$
with $F^\top$ tangent to $M$. For $v_1, \ldots, v_n \in T_pM$ the $n + 1$ vectors $F^\top, v_1, \ldots, v_n$ lie in the $n$-dimensional space $T_pM$, hence are
linearly dependent, and $\Omega(F^\top, v_1, \ldots, v_n) = 0$; so $(\iota_F\Omega)|_M = \langle F, \nu\rangle(\iota_\nu\Omega)|_M$. In $\RR^3$,
$\iota_\nu(dx \wedge dy \wedge dz) = \nu^1\,dy \wedge dz - \nu^2\,dx \wedge dz + \nu^3\,dx \wedge dy$ and $-dx \wedge dz = dz \wedge dx$; the flux formula is the integral
of $(\iota_F\Omega)|_S = \langle F, \nu\rangle\,dA$, by Proposition~\ref{riemannian-geometry:proposition-volume-form}.
\end{proof}

\begin{corollary}[Divergence theorem]\label{riemannian-geometry:corollary-divergence}
Let $D \subseteq \RR^{n+1}$ be a compact subset that is a smooth $(n+1)$-manifold with boundary (Definition~\ref{differential-forms:definition-manifold-with-boundary}),
with the charts given by charts of $\RR^{n+1}$, for instance a closed ball. Let $\nu$ be the outward unit normal along $\partial D$, and $F$ a smooth vector field on a
neighbourhood of $D$, with divergence $\operatorname{div}F = \sum_i\partial_iF^i$. Then
\[
\int_D\operatorname{div}F\,d\lambda = \int_{\partial D}\langle F, \nu\rangle\,dA .
\]
\end{corollary}

\begin{proof}
By Proposition~\ref{multilinear-algebra:proposition-interior-antiderivation},
$\iota_F\Omega = \sum_i(-1)^{i-1}F^i\,dx^1 \wedge \cdots \wedge \widehat{dx^i} \wedge \cdots \wedge dx^{n+1}$, and
$d(\iota_F\Omega) = \sum_i\partial_iF^i\,\Omega = (\operatorname{div}F)\,\Omega$, as in the proof of Theorem~\ref{differential-forms:theorem-stokes}. Stokes' theorem gives
$\int_D(\operatorname{div}F)\,\Omega = \int_{\partial D}\iota_F\Omega$, where $D$ carries the standard orientation of $\RR^{n+1}$, so that the left side is
$\int_D\operatorname{div}F\,d\lambda$, and $\partial D$ the induced orientation. We check that the induced orientation is the one defined by $\nu$; then the right side is
$\int_{\partial D}\langle F, \nu\rangle\,dA$ by Proposition~\ref{riemannian-geometry:proposition-flux}. In a positively oriented chart in which $D$ is
$\{x^{n+1} \geq 0\}$, the vector $w = -\partial_{n+1}$ points outward, and $\det(w, \partial_1, \ldots, \partial_n) = (-1)^{n+1}$ with respect to the chart; by
Definition~\ref{differential-forms:definition-manifold-with-boundary}, $(x^1, \ldots, x^n)$ is positively oriented on $\partial D$ exactly when $(-1)^{n+1} = 1$. So a
basis $(v_1, \ldots, v_n)$ of $T_p\partial D$ is positive for the induced orientation if and only if $(w, v_1, \ldots, v_n)$ is a positive basis of $\RR^{n+1}$. The same
holds with $\nu$ in place of $w$, because $w = c\nu + t$ with $c > 0$ and $t$ tangent to $\partial D$, so that $\det(w, v_1, \ldots, v_n) = c\det(\nu, v_1, \ldots, v_n)$.
By Proposition~\ref{differential-forms:proposition-orientation-normal} this is the orientation defined by $\nu$.
\end{proof}

\begin{example}\label{riemannian-geometry:example-areas}
\begin{enumerate}
\item \emph{The sphere.} The map $x(\theta, \varphi) = r(\sin\theta\cos\varphi, \sin\theta\sin\varphi, \cos\theta)$ on $(0, \pi) \times (0, 2\pi)$ parametrises the sphere of
radius $r$ minus a closed half great circle. That half circle has area $0$: without its two endpoints it is the segment $\varphi = 0$ for the same formula on
$(0, \pi) \times (-\pi, \pi)$, a Lebesgue null set, and points have measure $0$. Here $E = r^2$, $F = 0$, $G = r^2\sin^2\theta$, so $dA = r^2\sin\theta\,d\theta\,d\varphi$
and the area is $\int_0^{2\pi}\int_0^\pi r^2\sin\theta\,d\theta\,d\varphi = 4\pi r^2$.
\item \emph{The torus of revolution.} For $0 < b < a$, $x(u, v) = ((a + b\cos u)\cos v, (a + b\cos u)\sin v, b\sin u)$ has $E = b^2$, $F = 0$ and
$G = (a + b\cos u)^2$, so the area is $\int_0^{2\pi}\int_0^{2\pi}b(a + b\cos u)\,du\,dv = 4\pi^2ab$: the length $2\pi b$ of the rotated circle times the length $2\pi a$
of the path of its centre.
\item \emph{Graphs.} The upper hemisphere is the graph of $\phi = \sqrt{1 - u^2 - v^2}$ over the open unit disc, with $1 + |\nabla\phi|^2 = 1/(1 - u^2 - v^2)$. In polar
coordinates its area is $\int_0^{2\pi}\int_0^1\rho(1 - \rho^2)^{-1/2}\,d\rho\,d\theta = 2\pi$, half of $4\pi$.
\item \emph{Spheres and balls.} For $F(x) = x$ on $\RR^{n+1}$, $\operatorname{div}F = n + 1$, and on the unit sphere $\nu(x) = x$ and $\langle F, \nu\rangle = 1$. By
Corollary~\ref{riemannian-geometry:corollary-divergence}, $\mathrm{vol}(S^n) = (n + 1)\,\lambda(B^{n+1})$: for instance $2\pi = 2 \cdot \pi$ and
$4\pi = 3 \cdot \frac{4\pi}{3}$.
\end{enumerate}
\end{example}

\section{Covariant derivatives along curves}\label{riemannian-geometry:section-along-curves}

A \emph{vector field along} a smooth curve $\gamma \colon I \to M$ is a smooth map $V \colon I \to TM$ with $V(t) \in T_{\gamma(t)}M$; for instance the velocity $\dot\gamma$, or $Y \circ \gamma$ for a vector field $Y$ on $M$.

\begin{proposition}\label{riemannian-geometry:proposition-covariant-derivative-curves}
There is a unique operator $D_t$ sending vector fields along $\gamma$ to vector fields along $\gamma$ such that $D_t$ is $\RR$-linear, $D_t(fV) = \dot fV + fD_tV$ for smooth $f \colon I \to \RR$, and
$D_t(Y \circ \gamma) = \nabla_{\dot\gamma}Y$ for every vector field $Y$ defined near $\gamma(I)$. In coordinates, for $V = V^k\partial_k$,
\[
D_tV = \big(\dot V^k + \Gamma^k_{ij}(\gamma)\dot\gamma^iV^j\big)\partial_k.
\]
Moreover $\frac{d}{dt}\langle V, W\rangle = \langle D_tV, W\rangle + \langle V, D_tW\rangle$. A vector field along $\gamma$ is parallel in the sense of Definition~\ref{connections:definition-parallel} if and only if $D_tV = 0$.
\end{proposition}

\begin{proof}
In a chart, $V = V^k(\partial_k \circ \gamma)$, and the axioms force $D_tV = \dot V^k\partial_k + V^k\nabla_{\dot\gamma}\partial_k = (\dot V^k + \dot\gamma^iV^j\Gamma^k_{ij})\partial_k$, since $\nabla_{\dot\gamma}\partial_j = \dot\gamma^i\Gamma^k_{ij}\partial_k$. This
proves uniqueness. Conversely, the formula satisfies the axioms in each chart (for $Y = Y^k\partial_k$, $\frac{d}{dt}Y^k(\gamma) = \dot\gamma^i\partial_iY^k$ matches the coordinate expression of $\nabla_{\dot\gamma}Y$), and by uniqueness the
local definitions agree on overlaps. For the metric property, differentiate $\langle V, W\rangle = g_{jk}(\gamma)V^jW^k$: the derivative of $g_{jk}(\gamma)$ is $\dot\gamma^i\partial_ig_{jk}$, and by $(*)$ the result equals
$\langle D_tV, W\rangle + \langle V, D_tW\rangle$. The last statement is the local expression of $\nabla_{\dot\gamma}s = 0$ in the coordinate frame.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-derivative-along-curves}
In the language of Chapter~\ref{connections}, $D_t$ is the pullback connection $\gamma^*\nabla$ on the bundle $\gamma^*TM$ over $I$, differentiated in the direction of
the vector field $d/dt$ on $I$. Indeed $(\gamma^*\nabla)_{d/dt}$ is $\RR$-linear, satisfies the Leibniz rule for functions of $t$, and sends $Y \circ \gamma$ to
$(\nabla Y)(\dot\gamma) = \nabla_{\dot\gamma}Y$ (Proposition~\ref{connections:proposition-pullback-tensor}(1)), so it equals $D_t$ by the uniqueness in
Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves}. This is the meaning of $\nabla_{\dot\gamma}\dot\gamma$ in
Definition~\ref{riemannian-geometry:definition-geodesic}: it is $D_t\dot\gamma$. It could not be defined as $\nabla_XX$ for a vector field $X$, because the velocity
need not be the restriction of a vector field on $M$: the curve $\gamma(t) = (t^2, 0)$ passes through each point $(t^2, 0)$ twice, with the opposite velocities $(\pm2t, 0)$
(Remark~\ref{connections:remark-along-curves}). When $\dot\gamma$ does extend near $\gamma(t_0)$ to a vector field $X$, that is $X(\gamma(t)) = \dot\gamma(t)$ for $t$ near
$t_0$, then $D_t\dot\gamma = D_t(X \circ \gamma) = \nabla_{\dot\gamma}X = (\nabla_XX)(\gamma(t))$ there, whichever extension is chosen. For example, a unit speed great circle
$c$ on $S^2$ is the integral curve of the rotation field $X(x) = a \times x$ about its axis $a$, and $D_t\dot c = (\nabla_XX)(c)$.
\end{remark}

\begin{lemma}[Symmetry lemma]\label{riemannian-geometry:lemma-symmetry}
Let $f \colon (-\varepsilon, \varepsilon) \times (a, b) \to M$ be smooth, and write $D_s$, $D_t$ for the covariant derivatives along the curves $s \mapsto f(s, t)$ and $t \mapsto f(s, t)$. Then $D_s\partial_tf = D_t\partial_sf$.
\end{lemma}

\begin{proof}
In coordinates both sides equal $(\partial_s\partial_tf^k + \Gamma^k_{ij}\partial_sf^i\partial_tf^j)\partial_k$, by the symmetry $\Gamma^k_{ij} = \Gamma^k_{ji}$.
\end{proof}

\begin{proposition}\label{riemannian-geometry:proposition-geodesic-speed}
A curve $\gamma$ is a geodesic (Definition~\ref{riemannian-geometry:definition-geodesic}) if and only if $D_t\dot\gamma = 0$. Geodesics have constant speed, and parallel transport along any curve is a linear isometry.
\end{proposition}

\begin{proof}
The coordinate expression of $D_t\dot\gamma$ is the geodesic equation. If $D_t\dot\gamma = 0$ then $\frac{d}{dt}|\dot\gamma|^2 = 2\langle D_t\dot\gamma, \dot\gamma\rangle = 0$. If $V$ and $W$ are parallel, $\langle V, W\rangle$ is constant by
Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves}.
\end{proof}

\section{The exponential map and minimising geodesics}\label{riemannian-geometry:section-exponential}

Geodesics are the projections to $M$ of the integral curves of a vector field on $TM$, the \emph{geodesic spray}, which in the coordinates $(x, v)$ of $TM$ is $v^k\partial_{x^k} - \Gamma^k_{ij}(x)v^iv^j\partial_{v^k}$. By
Theorem~\ref{smooth-manifolds:theorem-flow}, the set $\mathcal{E} \subseteq TM$ of vectors $v$ for which $\gamma_v$ is defined on $[0, 1]$ is open, and $\exp \colon \mathcal{E} \to M$ is smooth. Since $\gamma_{cv}(t) = \gamma_v(ct)$, each
$\mathcal{E}_p = \mathcal{E} \cap T_pM$ is star-shaped about $0$, and $\exp_p(tv) = \gamma_v(t)$.

\begin{proposition}\label{riemannian-geometry:proposition-exponential}
$d_0\exp_p = \id$, so $\exp_p$ is a diffeomorphism from a neighbourhood of $0$ onto a neighbourhood of $p$, giving \emph{normal coordinates} in which $g_{ij}(p) = \delta_{ij}$ and $\Gamma^k_{ij}(p) = 0$.
Geodesics through $p$ are the images of lines through $0$.
\end{proposition}

\begin{proof}
Since $\exp_p(tv) = \gamma_v(t)$, the differential of $\exp_p$ at $0$ sends $v$ to $\dot\gamma_v(0) = v$, and the inverse function theorem applies. In normal coordinates for an orthonormal basis of $T_pM$, $g_{ij}(p) = \delta_{ij}$, and the lines
$t \mapsto tv$ are geodesics, so the geodesic equation at $t = 0$ gives $\Gamma^k_{ij}(p)v^iv^j = 0$ for all $v$; as $\Gamma^k_{ij}$ is symmetric in $i$, $j$, it vanishes at $p$.
\end{proof}

\begin{theorem}[Gauss lemma]\label{riemannian-geometry:theorem-gauss-lemma}
Let $v \in \mathcal{E}_p$ and $w \in T_pM$, and identify $T_v(T_pM)$ with $T_pM$. Then $\langle d_v\exp_p(v), d_v\exp_p(w)\rangle = \langle v, w\rangle$.
\end{theorem}

\begin{proof}
Since $\mathcal{E}_p$ is open and star-shaped, $f(s, t) = \exp_p(t(v + sw))$ is defined and smooth for $|s|$ small and $t \in [0, 1]$. Let $T = \partial_tf$ and $S = \partial_sf$. For fixed $s$, $t \mapsto f(s, t)$ is the geodesic with initial
velocity $v + sw$, so $D_tT = 0$ and $|T|^2 = |v + sw|^2$ for all $t$. Using Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves} and Lemma~\ref{riemannian-geometry:lemma-symmetry},
\[
\partial_t\langle S, T\rangle = \langle D_tS, T\rangle + \langle S, D_tT\rangle = \langle D_sT, T\rangle = \tfrac12\partial_s|T|^2 = \tfrac12\partial_s|v + sw|^2 = \langle v + sw, w\rangle.
\]
At $s = 0$ this is $\langle v, w\rangle$. Also $f(s, 0) = p$, so $S(s, 0) = 0$. Integrating, $\langle S, T\rangle(0, 1) = \langle v, w\rangle$. Finally $T(0, 1) = \frac{d}{dt}\exp_p(tv)|_{t=1} = d_v\exp_p(v)$ and
$S(0, 1) = \frac{d}{ds}\exp_p(v + sw)|_{s=0} = d_v\exp_p(w)$.
\end{proof}

\begin{theorem}[Geodesics are locally minimising]\label{riemannian-geometry:theorem-local-minimisation}
Let $\varepsilon > 0$ be such that $\exp_p$ is a diffeomorphism from $B(0, \varepsilon) \subseteq T_pM$ onto an open set $U$ (a \emph{normal ball}).
\begin{enumerate}
\item For $|v| < \varepsilon$, every curve from $p$ to $\exp_p(v)$ has length $\geq |v|$, with equality if and only if it is a monotone reparametrisation of the radial geodesic $t \mapsto \exp_p(tv)$, $t \in [0, 1]$.
\item $d(p, \exp_p(v)) = |v|$ for $|v| < \varepsilon$, and $\exp_p(B(0, r))$ is the metric ball $\{x : d(p, x) < r\}$ for $0 < r \leq \varepsilon$.
\end{enumerate}
\end{theorem}

\begin{proof}
For $x \in U$ let $\rho(x) = |\exp_p^{-1}(x)|$, a continuous function, smooth on $U \setminus \{p\}$. Let $c \colon [a, b] \to U \setminus \{p\}$ be a piecewise smooth curve and write $c(t) = \exp_p(\rho(t)u(t))$ with $\rho(t) = \rho(c(t)) > 0$ and $|u(t)| = 1$.
Then $\dot c = d\exp_p(\dot\rho u + \rho\dot u)$ at the point $\rho u$, with $\langle u, \dot u\rangle = 0$. By the Gauss lemma, $d\exp_p(u)$ has length $1$ and is orthogonal to $d\exp_p(\rho\dot u)$ (both at $\rho u$), so
\[
|\dot c|^2 = \dot\rho^2 + |d\exp_p(\rho\dot u)|^2 \geq \dot\rho^2, \qquad\text{hence}\qquad L(c) \geq \int_a^b|\dot\rho|\,dt \geq \rho(b) - \rho(a). \tag{$**$}
\]
Now let $c \colon [a, b] \to M$ go from $p$ to $q = \exp_p(v)$, $0 < |v| = r < \varepsilon$. Suppose first that $c$ stays in $U$. Let $a'$ be the last time with $c(a') = p$; applying $(**)$ on $[a'', b]$ for $a'' \downarrow a'$ gives $L(c) \geq r$. If
$L(c) = r$, then $c$ is constant on $[a, a']$, and on $(a', b]$ equality holds in $(**)$: $\dot\rho \geq 0$ and $d\exp_p(\rho\dot u) = 0$, so $\dot u = 0$ ($d\exp_p$ being injective on $B(0, \varepsilon)$), $u$ is constant and $c$ is a monotone
reparametrisation of the radial geodesic. Conversely such reparametrisations have length $r$. If $c$ leaves $U$, choose $r < r' < \varepsilon$. The set $K = \exp_p(\bar B(0, r'))$ is compact, hence closed, and contains $p$; so $c$
leaves $K$, and by continuity $\rho(c(t))$ takes the value $r'$ at some first time $t_2$ with $c([a, t_2]) \subseteq K$. The first case applied to $c|_{[a, t_2]}$ gives $L(c) \geq r' > r$. This proves (1).

For (2), the radial geodesic has length $|v|$, so $d(p, \exp_p(v)) = |v|$ by (1). The ball $\exp_p(B(0, r))$ lies in the metric ball of radius $r$. Conversely, if $d(p, x) < r$, choose a curve $c$ from $p$ to $x$ with $L(c) < r' < r$; if
$x \notin \exp_p(B(0, r'))$, the argument above shows $L(c) \geq r'$, a contradiction.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-convex}
Every point also has a \emph{geodesically convex} neighbourhood, in which any two points are joined by a unique minimising geodesic, contained in the neighbourhood \cite[Chapter 3]{do-carmo-riemannian}.
\end{remark}

\begin{lemma}[Uniformly normal neighbourhoods]\label{riemannian-geometry:lemma-uniformly-normal}
Every $q \in M$ has a neighbourhood $W$ and a $\delta > 0$ such that for every $x \in W$, $\exp_x$ is defined on $B(0, \delta) \subseteq T_xM$, is a diffeomorphism from it onto an open set, and $\exp_x(B(0, \delta)) \supseteq W$.
\end{lemma}

\begin{proof}
Consider $E \colon \mathcal{E} \to M \times M$, $E(v) = (\pi(v), \exp(v))$. In coordinates $(x, v)$ near $0_q \in TM$, its differential at $0_q$ is $\begin{pmatrix}I & 0\\ I & I\end{pmatrix}$, since $\exp(0_x) = x$ and $d_0\exp_x = \mathrm{id}$; so by the
inverse function theorem $E$ is a diffeomorphism from a neighbourhood $O$ of $0_q$ onto a neighbourhood of $(q, q)$. Shrinking, we may take $O = \{v \in T_xM : x \in W', |v| < \delta\}$ for a neighbourhood $W'$ of $q$ and some $\delta > 0$
(the norm is continuous on $TM$), and choose a neighbourhood $W \subseteq W'$ of $q$ with $W \times W \subseteq E(O)$. For $x \in W$, $\exp_x$ is injective on $B(0, \delta)$ with invertible differential, hence a diffeomorphism onto an open
set; and for $y \in W$, $(x, y) = E(v)$ with $v \in O$, that is $y = \exp_x(v)$ with $|v| < \delta$.
\end{proof}

\begin{lemma}\label{riemannian-geometry:lemma-minimising-geodesic}
Let $c \colon [a, b] \to M$ be a piecewise smooth curve of constant speed with $L(c) = d(c(a), c(b))$. Then $c$ is a geodesic; in particular it is smooth.
\end{lemma}

\begin{proof}
Every subarc of $c$ is also minimising, since otherwise replacing it by a shorter curve would shorten $c$. Let $t_0 \in [a, b]$, and take $W$ and $\delta$ for $q = c(t_0)$ as in Lemma~\ref{riemannian-geometry:lemma-uniformly-normal}. For
$t_1 < t_0 < t_2$ close to $t_0$ (or $t_1 = t_0$ at the endpoints), $c([t_1, t_2]) \subseteq W$ and $L(c|_{[t_1, t_2]}) < \delta$. Then $c|_{[t_1, t_2]}$ is a minimising curve from $x = c(t_1)$ to a point of the normal ball $\exp_x(B(0, \delta))$, so by
Theorem~\ref{riemannian-geometry:theorem-local-minimisation}(1) it is a monotone reparametrisation of a radial geodesic from $x$, and having constant speed, it is that geodesic, affinely reparametrised. So $c$ is a geodesic near every $t_0$.
\end{proof}

\begin{theorem}[Hopf--Rinow]\label{riemannian-geometry:theorem-hopf-rinow}
For a connected Riemannian manifold $(M, g)$ the following are equivalent: $(M, d)$ is a complete metric space; the geodesics are defined for all time; closed bounded subsets are compact. In that case any
two points are joined by a minimising geodesic. In particular compact Riemannian manifolds are geodesically complete.
\end{theorem}

\begin{proof}
We show more: the three conditions are also equivalent to (d) \emph{$\exp_p$ is defined on all of $T_pM$ for some $p$}; and (d) implies that every $q$ is joined to $p$ by a minimising geodesic.

\emph{Step 1: (d) gives minimising geodesics from $p$.} Let $q \neq p$, $r = d(p, q)$, and choose $0 < \delta < r$ with $\exp_p$ a diffeomorphism on $B(0, \delta')$ for some $\delta' > \delta$. The \emph{geodesic sphere}
$S = \exp_p(\{|w| = \delta\})$ is compact; let $x_0 = \exp_p(\delta u) \in S$, $|u| = 1$, minimise $d(\cdot, q)$ on $S$. Every curve from $p$ to $q$ meets $S$ (by the argument in the proof of
Theorem~\ref{riemannian-geometry:theorem-local-minimisation}), and points of $S$ have distance $\delta$ from $p$, so $r = \delta + d(x_0, q)$. Let $\gamma(t) = \exp_p(tu)$, defined for all $t$ by (d), and let
$T = \{t \in [\delta, r] : d(\gamma(t), q) = r - t\}$, a closed set containing $\delta$. Let $t_0 = \max T$ and suppose $t_0 < r$. Apply the same construction at $\gamma(t_0)$: for small $\eta > 0$ there is $x_1$ on the geodesic sphere of
radius $\eta$ about $\gamma(t_0)$ with $d(\gamma(t_0), q) = \eta + d(x_1, q)$, so $d(x_1, q) = r - t_0 - \eta$ and $d(p, x_1) \geq d(p, q) - d(x_1, q) = t_0 + \eta$. The curve going along $\gamma$ from $p$ to $\gamma(t_0)$ and then along the
radial geodesic to $x_1$ has length $t_0 + \eta$, so it is minimising, and by Lemma~\ref{riemannian-geometry:lemma-minimising-geodesic} (after parametrising by arc length) it is a geodesic, hence equal to $\gamma$ on $[0, t_0 + \eta]$.
So $x_1 = \gamma(t_0 + \eta)$ and $t_0 + \eta \in T$, a contradiction. Hence $t_0 = r$, $\gamma(r) = q$ and $L(\gamma|_{[0, r]}) = r$.

\emph{Step 2: the equivalences.} If closed bounded sets are compact, a Cauchy sequence, being bounded, has a convergent subsequence, hence converges; so $M$ is complete. If $M$ is complete, let $\gamma \colon [0, b) \to M$ be a maximal unit-speed
geodesic with $b < \infty$. For $t_k \uparrow b$, $d(\gamma(t_k), \gamma(t_l)) \leq |t_k - t_l|$, so $(\gamma(t_k))$ is Cauchy and converges to some $q$. Take $W$, $\delta$ for $q$ as in Lemma~\ref{riemannian-geometry:lemma-uniformly-normal}; then
geodesics with initial point in $W$ and speed $1$ are defined on $(-\delta, \delta)$. Choose $t_k > b - \delta$ with $\gamma(t_k) \in W$; the geodesic with initial velocity $\dot\gamma(t_k)$ is defined up to time $t_k + \delta > b$ and extends $\gamma$ by
uniqueness, a contradiction. So geodesics are defined for all time, which implies (d). Finally, assume (d) and let $K$ be closed and bounded, say $K \subseteq \{x : d(p, x) \leq R\}$. By Step 1, this ball is contained in
$\exp_p(\bar B(0, R))$, which is compact; so $K$ is compact. The last statements follow from Step 1 applied at every point, and from the compactness of closed bounded subsets of a compact space.
\end{proof}

\begin{counterexample}\label{riemannian-geometry:counterexample-incomplete}
Completeness is needed. In $M = \RR^2 \setminus \{0\}$ with the Euclidean metric, the geodesic $t \mapsto (1 - t, 0)$ is not defined at $t = 1$, the Cauchy sequence $(1/k, 0)$ has no limit, and the set $\{x \in M : |x| \leq 1\}$ is closed in $M$
and bounded but not compact. The points $(1, 0)$ and $(-1, 0)$ have distance $2$, the infimum of the lengths of curves avoiding the origin, but no
curve of length $2$ joins them in $M$, since in $\RR^2$ the only such curve is the segment through the origin.
\end{counterexample}

\begin{remark}\label{riemannian-geometry:remark-non-unique-geodesics}
Minimising geodesics need not be unique: on the round sphere every half great circle joins two antipodal points, and has length $\pi$. And geodesics stop minimising: a great circle arc longer than $\pi$ is a geodesic
but not the shortest curve between its endpoints. Both phenomena are explained by conjugate points (Section~\ref{riemannian-geometry:section-curvature}).
\end{remark}

\section{Parallel transport and holonomy}\label{riemannian-geometry:section-holonomy}

The \emph{holonomy group} $\mathrm{Hol}_p(g) \subseteq \GL(T_pM)$ of $(M, g)$ at $p$ is the holonomy group of the Levi-Civita connection (Definition~\ref{connections:definition-holonomy}), for piecewise smooth loops at $p$. The
\emph{restricted holonomy group} $\mathrm{Hol}^0_p(g)$ is the subgroup of transports along null-homotopic loops.

\begin{proposition}\label{riemannian-geometry:proposition-holonomy-basic}
Let $(M, g)$ be connected and $p, q \in M$.
\begin{enumerate}
\item $\mathrm{Hol}_p(g) \subseteq \mathrm{O}(T_pM, g_p)$.
\item For a curve $\sigma$ from $p$ to $q$, $\mathrm{Hol}_q(g) = P_\sigma\mathrm{Hol}_p(g)P_\sigma^{-1}$; so the holonomy groups at different points are conjugate.
\item $\mathrm{Hol}^0_p(g)$ is a normal subgroup of $\mathrm{Hol}_p(g)$, and $[\gamma] \mapsto P_\gamma^{-1}\mathrm{Hol}^0_p(g)$ is a surjective homomorphism $\pi_1(M, p) \to \mathrm{Hol}_p(g)/\mathrm{Hol}^0_p(g)$. In particular
$\mathrm{Hol}_p = \mathrm{Hol}^0_p$ if $M$ is simply connected.
\item $M$ is orientable if and only if $\mathrm{Hol}_p(g) \subseteq \mathrm{SO}(T_pM)$.
\item $\mathrm{Hol}^0_p(g)$ is trivial if and only if $g$ is flat, $R = 0$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) is Proposition~\ref{riemannian-geometry:proposition-geodesic-speed}. (2) Loops $\gamma$ at $p$ correspond bijectively to loops $\bar\sigma * \gamma * \sigma$ at $q$, with transport $P_\sigma P_\gamma P_\sigma^{-1}$
(Proposition~\ref{connections:proposition-parallel-transport}). (3) If $\beta$ is null-homotopic and $\alpha$ is a loop at $p$, then $\bar\alpha * \beta * \alpha$ is null-homotopic, with transport $P_\alpha P_\beta P_\alpha^{-1}$; so
$\mathrm{Hol}^0_p$ is normal. If $\gamma \simeq \gamma'$, then $\bar\gamma * \gamma'$ is null-homotopic and $P_{\gamma'}P_\gamma^{-1} \in \mathrm{Hol}^0_p$, so the map is well defined on homotopy classes, and it is multiplicative since
$P_{\gamma * \delta}^{-1} = P_\gamma^{-1}P_\delta^{-1}$. Surjectivity is clear. (4) If $M$ is oriented, a parallel frame along a curve is continuous, so the set of times at which it is positively oriented is open and closed; hence parallel
transport preserves orientations. Conversely, if $\mathrm{Hol}_p \subseteq \mathrm{SO}(T_pM)$, fix an orientation of $T_pM$ and transport it along curves from $p$; by (2) the result does not depend on the curve, and near each point the
transported frames along radial geodesics form a smooth local frame, so the orientation is continuous. (5) By Theorem~\ref{connections:theorem-flat-local-systems}, $R = 0$ if and only if parallel transport depends only on the
homotopy class of the path with fixed endpoints, which is equivalent to the triviality of transport around null-homotopic loops.
\end{proof}

\begin{theorem}[Holonomy principle]\label{riemannian-geometry:theorem-holonomy-principle}
Let $(M, g)$ be connected and let $E$ be a tensor bundle $T^{\otimes a} \otimes (T^*)^{\otimes b}$ of $M$ with the connection induced by $\nabla$. A tensor $\tau_p \in E_p$ is the value at $p$ of a parallel section of $E$ if and only if it is
invariant under $\mathrm{Hol}_p(g)$, acting on $E_p$ through its action on $T_pM$; the parallel section is then unique. Similarly, a subspace $V \subseteq T_pM$ is the fibre at $p$ of a parallel subbundle of $TM$ if and only if it is
$\mathrm{Hol}_p(g)$-invariant.
\end{theorem}

\begin{proof}
Parallel transport on $E$ is induced by that on $TM$. If $\tau$ is parallel, then along every loop $\gamma$ at $p$ the section $\tau \circ \gamma$ is parallel, so $P_\gamma\tau_p = \tau_p$. Conversely, if $\tau_p$ is invariant, define $\tau_q = P_\sigma\tau_p$
for any curve $\sigma$ from $p$ to $q$. If $\sigma'$ is another such curve, $P_{\sigma'}^{-1}P_\sigma$ is the transport along the loop $\sigma * \bar\sigma'$ at $p$, which lies in $\mathrm{Hol}_p$ and fixes $\tau_p$; so
$\tau_q$ is well defined. Near a point $q_0$, $\tau_q$ is the transport of $\tau_{q_0}$ along the radial geodesic from $q_0$ to $q$ in a normal ball, which depends smoothly on $q$ by smooth dependence of solutions of
differential equations; so $\tau$ is smooth. For every curve $c$, $\tau \circ c$ is the transport of $\tau_{c(0)}$ along $c$, hence parallel; so $\nabla\tau = 0$. Uniqueness holds because a parallel section is determined by its value at one
point of a connected manifold. The statement for subbundles is proved in the same way.
\end{proof}

\begin{example}\label{riemannian-geometry:example-holonomy}
\begin{enumerate}
\item On $\RR^n$ the Christoffel symbols vanish in the standard coordinates, so parallel transport is the identity and $\mathrm{Hol} = \{1\}$; the same holds for the flat torus, since parallel transport in the quotient is induced from
$\RR^n$ and translations have trivial differential.
\item For the flat Klein bottle $\RR^2/G$ (Example~\ref{riemannian-geometry:example-quotient-metrics}), a loop at $\pi(\tilde x)$ lifts to a curve from $\tilde x$ to $h\tilde x$ for some $h \in G$; transport in $\RR^2$ is the identity, and
$d\pi_{h\tilde x} = d\pi_{\tilde x} \circ (dh)^{-1}$, so the holonomy is $(dh)^{-1}$, the inverse of the linear part of $h$. Hence $\mathrm{Hol} = \{1, \mathrm{diag}(1, -1)\}$: the Klein bottle is flat, $\mathrm{Hol}^0 = \{1\}$, but
$\mathrm{Hol} \neq \{1\}$, and $\mathrm{Hol} \not\subseteq \mathrm{SO}(2)$, in accordance with non-orientability.
\item For the round sphere $S^2$, $\mathrm{Hol}_p = \mathrm{Hol}^0_p$ is a nontrivial subgroup of $\mathrm{SO}(2)$ by Proposition~\ref{riemannian-geometry:proposition-holonomy-basic}(3),(4),(5), as $S^2$ is simply connected, orientable and
not flat. Since $\mathrm{Hol}^0_p$ is a connected Lie subgroup (Theorem~\ref{riemannian-geometry:theorem-ambrose-singer} below), it is $\mathrm{SO}(2)$. Concretely, transport around the boundary of the octant
$\{x, y, z \geq 0\}$ rotates tangent vectors by $\pi/2$, the area enclosed.
\end{enumerate}
\end{example}

\begin{theorem}[Ambrose--Singer]\label{riemannian-geometry:theorem-ambrose-singer}
$\mathrm{Hol}_p(g)$ is a Lie subgroup of $\mathrm{O}(T_pM)$ with identity component $\mathrm{Hol}^0_p(g)$, and the Lie algebra of $\mathrm{Hol}_p(g)$ is spanned by the endomorphisms $P_\sigma^{-1} \circ R(P_\sigma u, P_\sigma v) \circ P_\sigma$ of
$T_pM$, for all curves $\sigma$ starting at $p$ and all $u, v \in T_pM$.
\end{theorem}

This is \cite[Chapter II]{kobayashi-nomizu}; see also \cite{joyce-special-holonomy}. The theorem makes Proposition~\ref{riemannian-geometry:proposition-holonomy-basic}(5) quantitative: curvature measures infinitesimal holonomy.

\begin{theorem}[de Rham decomposition]\label{riemannian-geometry:theorem-de-rham-decomposition}
If the holonomy representation of a connected Riemannian manifold on $T_pM$ is reducible, $T_pM = V_1 \oplus V_2$ with both summands $\mathrm{Hol}_p$-invariant and nonzero, then $M$ is locally isometric to a Riemannian product.
If moreover $M$ is complete and simply connected, it is globally isometric to a product $M_1 \times M_2$ of Riemannian manifolds with $T_pM_i = V_i$.
\end{theorem}

\begin{proof}[Sketch of proof]
By Theorem~\ref{riemannian-geometry:theorem-holonomy-principle}, $V_1$ and $V_2$ extend to parallel, mutually orthogonal distributions $D_1$ and $D_2$. A parallel distribution is involutive, as $[X, Y] = \nabla_XY - \nabla_YX$ lies in
$D_i$ when $X$, $Y$ do, so by the Frobenius theorem (Theorem~\ref{smooth-manifolds:theorem-frobenius}) there are local coordinates $(x, y)$ with $D_1$ spanned by the $\partial_x$ and $D_2$ by the $\partial_y$; the omitted verification is that in
these coordinates the metric has the product form $g_1(x) + g_2(y)$, which follows from the parallelism of $D_1$ and $D_2$. The global statement is \cite[Chapter IV]{kobayashi-nomizu}.
\end{proof}

\begin{theorem}[Berger]\label{riemannian-geometry:theorem-berger}
Let $(M, g)$ be a simply connected Riemannian manifold of dimension $n$ that is not locally a product (its holonomy representation is irreducible) and not locally symmetric ($\nabla R \neq 0$). Then $\mathrm{Hol}(g)$ is one
of: $\mathrm{SO}(n)$; $\mathrm{U}(m)$ with $n = 2m \geq 4$ (K\"ahler metrics); $\mathrm{SU}(m)$ with $n = 2m \geq 4$ (Ricci-flat K\"ahler, Calabi--Yau); $\mathrm{Sp}(m)$ with $n = 4m \geq 8$ (hyper-K\"ahler); $\mathrm{Sp}(m)\mathrm{Sp}(1)$ with
$n = 4m \geq 8$ (quaternionic K\"ahler); $G_2$ with $n = 7$; $\mathrm{Spin}(7)$ with $n = 8$.
\end{theorem}

This is \cite{berger-1955}; Berger's list also contained $\mathrm{Spin}(9)$ in dimension $16$, which was later shown to occur only for symmetric metrics. See \cite{joyce-special-holonomy} for the modern treatment and for the
existence of metrics with holonomy $G_2$ and $\mathrm{Spin}(7)$.

\begin{remark}\label{riemannian-geometry:remark-special-holonomy}
By the holonomy principle, the reduction of the holonomy group to a subgroup of $\mathrm{O}(n)$ is the same as the existence of a parallel tensor with that stabiliser. For instance $\mathrm{Hol} \subseteq \mathrm{U}(m)$ if and only
if there is a parallel orthogonal almost complex structure $J$, that is, $(M, g, J)$ is K\"ahler (Chapter~\ref{kahler}); and $\mathrm{Hol} \subseteq \mathrm{SU}(m)$ if and only if there is moreover a parallel complex volume form,
which for a simply connected K\"ahler manifold is equivalent to Ricci-flatness (Chapter~\ref{calabi-yau}). This is the dictionary between Riemannian holonomy and complex geometry that underlies Part~\ref{part-complex-geometry}.
\end{remark}

\section{Curvature}\label{riemannian-geometry:section-curvature}

\begin{definition}\label{riemannian-geometry:definition-curvature-tensor}
The \emph{Riemann curvature tensor} is the curvature of the Levi-Civita connection (Proposition~\ref{connections:proposition-curvature}),
$R(X, Y)Z = \nabla_X\nabla_YZ - \nabla_Y\nabla_XZ - \nabla_{[X,Y]}Z$, viewed as the $4$-tensor $R(X, Y, Z, W) = g(R(X, Y)Z, W)$. For a plane $\sigma \subseteq T_pM$ with orthonormal basis $u, v$, the
\emph{sectional curvature} is $K(\sigma) = R(u, v, v, u)$; the \emph{Ricci tensor} is $\mathrm{Ric}(X, Y) = \sum_iR(e_i, X, Y, e_i)$ for an orthonormal frame, and the \emph{scalar curvature} is
$s = \sum_i\mathrm{Ric}(e_i, e_i)$.
\end{definition}

\begin{proposition}\label{riemannian-geometry:proposition-curvature-symmetries}
$R$ is tensorial and satisfies $R(X, Y, Z, W) = -R(Y, X, Z, W) = -R(X, Y, W, Z) = R(Z, W, X, Y)$ and the first Bianchi identity
$R(X, Y)Z + R(Y, Z)X + R(Z, X)Y = 0$. The sectional curvatures determine $R$.
\end{proposition}

\begin{proof}
Tensoriality and the first antisymmetry are general facts about curvature. The second follows by polarising $R(X, Y, Z, Z) = 0$, which holds because the connection is metric:
$0 = XY g(Z,Z)/2 - \cdots$ expands to $g(R(X,Y)Z, Z)$. The first Bianchi identity follows from torsion-freeness and the Jacobi identity, and the symmetry in the pairs is a formal consequence of the other
three, obtained by adding the Bianchi identities for the four cyclic permutations. The last statement is the standard polarisation argument \cite[Chapter 4]{do-carmo-riemannian}.
\end{proof}

\begin{example}\label{riemannian-geometry:example-curvature}
Euclidean space has $R = 0$; the round sphere of radius $1$ has constant sectional curvature $1$ and $\mathrm{Ric} = (n-1)g$; hyperbolic space has constant curvature $-1$. A Lie group with a bi-invariant
metric has $R(X, Y)Z = \frac14[[X, Y], Z]$ for left-invariant fields, hence non-negative sectional curvature. A metric has constant sectional curvature $\kappa$ if and only if
$R(X, Y)Z = \kappa(\langle Y, Z\rangle X - \langle X, Z\rangle Y)$, since the right-hand side has the symmetries of Proposition~\ref{riemannian-geometry:proposition-curvature-symmetries} and sectional curvature $\kappa$.
\end{example}

\begin{lemma}\label{riemannian-geometry:lemma-curvature-families}
For $f$ as in Lemma~\ref{riemannian-geometry:lemma-symmetry} and a vector field $V$ along $f$, $D_sD_tV - D_tD_sV = R(\partial_sf, \partial_tf)V$.
\end{lemma}

\begin{proof}
Write $V = V^k(\partial_k \circ f)$. In $D_sD_tV$ the terms containing derivatives of the $V^k$ are symmetric in $s$ and $t$, so the left side is $V^k(D_sD_t - D_tD_s)(\partial_k \circ f)$. Now
$D_t(\partial_k \circ f) = \partial_tf^i(\nabla_{\partial_i}\partial_k) \circ f$, and
$D_sD_t(\partial_k \circ f) = \partial_s\partial_tf^i(\nabla_{\partial_i}\partial_k) \circ f + \partial_tf^i\partial_sf^j(\nabla_{\partial_j}\nabla_{\partial_i}\partial_k) \circ f$. Antisymmetrising and using $[\partial_i, \partial_j] = 0$ gives
$\partial_sf^j\partial_tf^iR(\partial_j, \partial_i)\partial_k = R(\partial_sf, \partial_tf)\partial_k$.
\end{proof}

\begin{definition}\label{riemannian-geometry:definition-jacobi}
A \emph{Jacobi field} along a geodesic $\gamma$ is a vector field $J$ along $\gamma$ with $D_t^2J + R(J, \dot\gamma)\dot\gamma = 0$. A point $\gamma(b)$ is \emph{conjugate} to $\gamma(a)$ along $\gamma$ if there is a nonzero Jacobi field
vanishing at $a$ and $b$.
\end{definition}

The Jacobi equation is a linear second order equation, so a Jacobi field is determined by $J(a)$ and $D_tJ(a)$, and the Jacobi fields along $\gamma$ form a vector space of dimension $2n$.

\begin{proposition}\label{riemannian-geometry:proposition-jacobi}
\begin{enumerate}
\item If $f(s, t)$ is a family of geodesics $t \mapsto f(s, t)$, then $J = \partial_sf(0, -)$ is a Jacobi field along $\gamma = f(0, -)$.
\item For $v \in \mathcal{E}_p$ and $w \in T_pM$, $J(t) = d_{tv}\exp_p(tw)$ is the Jacobi field along $\gamma(t) = \exp_p(tv)$ with $J(0) = 0$ and $D_tJ(0) = w$.
\item $d_v\exp_p$ is singular if and only if $\exp_p(v)$ is conjugate to $p$ along $t \mapsto \exp_p(tv)$, $t \in [0, 1]$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) With $T = \partial_tf$ and $S = \partial_sf$, Lemmas~\ref{riemannian-geometry:lemma-symmetry} and~\ref{riemannian-geometry:lemma-curvature-families} give
$D_tD_tS = D_tD_sT = D_sD_tT + R(T, S)T = R(T, S)T = -R(S, T)T$, since $D_tT = 0$. (2) Apply (1) to $f(s, t) = \exp_p(t(v + sw))$: $\partial_sf(0, t) = d_{tv}\exp_p(tw)$. It vanishes at $t = 0$, and
$D_tS(0, 0) = D_sT(0, 0) = \frac{d}{ds}(v + sw) = w$, the covariant derivative along the constant curve $s \mapsto p$ being the ordinary derivative in $T_pM$. (3) By (2) and uniqueness, the Jacobi fields vanishing at $0$ are exactly the
$d_{tv}\exp_p(tw)$, and at $t = 1$ this is $d_v\exp_p(w)$.
\end{proof}

\begin{example}\label{riemannian-geometry:example-jacobi-constant}
If $g$ has constant sectional curvature $\kappa$ and $\gamma$ has unit speed, then for a parallel unit field $E \perp \dot\gamma$, $R(E, \dot\gamma)\dot\gamma = \kappa E$ by Example~\ref{riemannian-geometry:example-curvature}, and the Jacobi fields
normal to $\gamma$ with $J(0) = 0$ are $J(t) = s_\kappa(t)E(t)$ with $s_\kappa(t) = t$, $\sin(\sqrt\kappa t)/\sqrt\kappa$ or $\sinh(\sqrt{-\kappa}t)/\sqrt{-\kappa}$ for $\kappa = 0$, $> 0$, $< 0$. On the unit sphere they vanish again at $t = \pi$: antipodal
points are conjugate, which explains Remark~\ref{riemannian-geometry:remark-non-unique-geodesics}. For $\kappa \leq 0$ there are no conjugate points.
\end{example}

\begin{theorem}[Cartan--Hadamard]\label{riemannian-geometry:theorem-cartan-hadamard}
Let $(M, g)$ be complete and connected with sectional curvature $K \leq 0$. Then for every $p$, $\exp_p \colon T_pM \to M$ is a covering map; if $M$ is simply connected, it is a diffeomorphism, so $M$ is diffeomorphic to $\RR^n$.
\end{theorem}

\begin{proof}[Sketch of proof]
Let $J$ be a Jacobi field along a geodesic $\gamma$ with $J(0) = 0$ and $D_tJ(0) \neq 0$, and $\phi = |J|^2$. Then
$\phi'' = 2|D_tJ|^2 - 2\langle R(J, \dot\gamma)\dot\gamma, J\rangle = 2|D_tJ|^2 - 2K(J, \dot\gamma)(|J|^2|\dot\gamma|^2 - \langle J, \dot\gamma\rangle^2) \geq 0$, where $K(J, \dot\gamma)$ is the sectional curvature of the plane they span (if they are
independent). So $\phi$ is convex with $\phi(0) = \phi'(0) = 0$ and $\phi''(0) > 0$, hence positive for $t > 0$: there are no conjugate points, and $\exp_p$ is a local diffeomorphism on $T_pM$ (defined everywhere by
Theorem~\ref{riemannian-geometry:theorem-hopf-rinow}) by Proposition~\ref{riemannian-geometry:proposition-jacobi}(3). The omitted verification is that a local diffeomorphism from a complete Riemannian manifold (here $T_pM$ with the
pulled back metric, whose radial lines are geodesics) that is a local isometry is a covering map \cite[Chapter 7]{do-carmo-riemannian}.
\end{proof}

\begin{theorem}[Gauss--Bonnet]\label{riemannian-geometry:theorem-gauss-bonnet}
Let $M$ be a compact oriented Riemannian surface, with Gaussian curvature $K$ (the sectional curvature of its tangent planes) and area form $dA$
(Proposition~\ref{riemannian-geometry:proposition-volume-form}). Then
\[
\int_MK\,dA = 2\pi\chi(M) .
\]
\end{theorem}

\begin{proof}
Let $J$ be the rotation by $+\pi/2$ in each tangent plane, for the metric and the orientation: $Je_1 = e_2$ for every oriented orthonormal frame $(e_1, e_2)$. It is a smooth endomorphism with $J^2 = -1$, so $TM$ becomes a
complex line bundle, with $i$ acting as $J$, whose underlying oriented real bundle is $TM$. Near a point choose an oriented orthonormal frame $(e_1, e_2)$ and define the $1$-form $\omega$ by $\nabla_Xe_1 = \omega(X)e_2$; since
$\nabla$ is metric, $\langle\nabla_Xe_1, e_1\rangle = 0$ and $\nabla_Xe_2 = -\omega(X)e_1$. Hence $\nabla J = J\nabla$, so $\nabla$ is a connection on the complex line bundle, and in the complex frame $e_1$ it reads
$\nabla e_1 = \omega \otimes Je_1 = i\omega \otimes e_1$: its connection form is $\theta = i\omega$ and its curvature is $F = d\theta = i\,d\omega$ (Proposition~\ref{connections:proposition-curvature}).

We compare $d\omega$ with $K$. From the definition of the curvature tensor,
\[
R(X, Y)e_1 = \nabla_X(\omega(Y)e_2) - \nabla_Y(\omega(X)e_2) - \omega([X, Y])e_2 = \big(X\omega(Y) - Y\omega(X) - \omega([X, Y])\big)e_2 = d\omega(X, Y)\,e_2,
\]
the terms $\mp\omega(X)\omega(Y)e_1$ cancelling. By Definition~\ref{riemannian-geometry:definition-curvature-tensor} and Proposition~\ref{riemannian-geometry:proposition-curvature-symmetries},
$K = R(e_1, e_2, e_2, e_1) = -R(e_1, e_2, e_1, e_2) = -\langle R(e_1, e_2)e_1, e_2\rangle = -d\omega(e_1, e_2)$. So $d\omega = -K\,e^1 \wedge e^2 = -K\,dA$ and $F = -iK\,dA$.

By Chern--Weil theory (Theorem~\ref{characteristic-classes:theorem-chern-weil} and Definition~\ref{characteristic-classes:definition-chern-forms}), $c_1(TM)$ is represented by $\frac{i}{2\pi}F = \frac{1}{2\pi}K\,dA$ for this connection.
For a complex line bundle the Euler class of the underlying oriented real bundle is $c_1$ (Theorem~\ref{characteristic-classes:theorem-euler}(2)), and $\int_Me(TM) = \chi(M)$ by
Theorem~\ref{characteristic-classes:theorem-euler}(3). Hence $\frac{1}{2\pi}\int_MK\,dA = \chi(M)$.
\end{proof}

\begin{corollary}\label{riemannian-geometry:corollary-gauss-bonnet}
Let $\Sigma_g$ be the closed orientable surface of genus $g$, so that $\chi(\Sigma_g) = 2 - 2g$ (Example~\ref{singular-homology:example-cellular-computations}).
\begin{enumerate}
\item The integral of the Gaussian curvature does not depend on the metric: for the round unit sphere, $K = 1$ and the area is $4\pi = 2\pi \cdot 2$.
\item If $\Sigma_g$ carries a metric with $K > 0$ everywhere, then $g = 0$; if it carries a metric with $K \geq 0$ everywhere, then $g \leq 1$; if it carries one with $K < 0$ everywhere, then $g \geq 2$.
\item The torus admits flat metrics, as $\RR^2/\ZZ^2$, but no metric with $K > 0$ everywhere or $K < 0$ everywhere.
\end{enumerate}
\end{corollary}

\begin{proof}
(1) is Theorem~\ref{riemannian-geometry:theorem-gauss-bonnet}. (2) The sign of $\int_MK\,dA = 2\pi(2 - 2g)$ is positive, non-negative or negative accordingly, and $2 - 2g > 0$ forces $g = 0$, $2 - 2g \geq 0$ forces
$g \leq 1$, and $2 - 2g < 0$ forces $g \geq 2$. (3) The torus has $\chi = 0$, so $K$ cannot have constant nonzero sign, while the flat metric induced from $\RR^2$ has $K = 0$ (Example~\ref{riemannian-geometry:example-curvature}).
\end{proof}

\section{The first and second variation of energy}\label{riemannian-geometry:section-variation}

Geodesics are the critical points of the energy of curves, and the second derivative of the energy along a geodesic is a quadratic form, the index form, in which
the curvature appears. This section proves both variation formulas and uses the second one to show that geodesics stop minimising after a conjugate point
(Jacobi's theorem) and that positive Ricci curvature bounds the diameter (the Bonnet--Myers theorem).

Let $\gamma \colon [a, b] \to M$ be a piecewise smooth curve. A \emph{variation} of $\gamma$ is a continuous map $f \colon (-\varepsilon, \varepsilon) \times [a, b] \to M$
with $f(0, t) = \gamma(t)$, together with a partition $a = t_0 < t_1 < \cdots < t_k = b$ such that $f$ is smooth on each $(-\varepsilon, \varepsilon) \times [t_{i-1}, t_i]$;
it is \emph{proper} if $f(s, a) = \gamma(a)$ and $f(s, b) = \gamma(b)$ for all $s$. We write $f_s = f(s, -)$, and $V(t) = \partial_sf(0, t)$ for the \emph{variation field},
a continuous vector field along $\gamma$, smooth on each $[t_{i-1}, t_i]$ (\emph{piecewise smooth}); it vanishes at $a$ and $b$ if $f$ is proper. The \emph{energy} of a
piecewise smooth curve $c \colon [a, b] \to M$ is $E(c) = \frac12\int_a^b|\dot c|^2\,dt$. On each piece we use the covariant derivatives $D_s$ and $D_t$ of
Lemma~\ref{riemannian-geometry:lemma-symmetry}.

Every piecewise smooth field $V$ along $\gamma$ is the variation field of a variation, for instance of
\[
f(s, t) = \exp_{\gamma(t)}(sV(t)) .
\]
Indeed the set of $(s, t)$ with $sV(t)$ in the domain $\mathcal{E}$ of $\exp$ is open and contains $\{0\} \times [a, b]$, so by compactness it contains
$(-\varepsilon, \varepsilon) \times [a, b]$ for some $\varepsilon > 0$; and $\partial_sf(0, t) = d_0\exp_{\gamma(t)}(V(t)) = V(t)$. This variation is proper if $V(a) = V(b) = 0$, and
its curves $s \mapsto f(s, t)$ are geodesics, so $D_s\partial_sf = 0$.

\begin{lemma}\label{riemannian-geometry:lemma-energy-length}
For a piecewise smooth curve $c \colon [a, b] \to M$, $L(c)^2 \leq 2(b - a)E(c)$, with equality if and only if $c$ has constant speed. Consequently, if $\gamma$ has constant
speed and $E(c) < E(\gamma)$ for a curve $c$ with the same interval, then $L(c) < L(\gamma)$.
\end{lemma}

\begin{proof}
By the Cauchy--Schwarz inequality, $L(c)^2 = \bigl(\int_a^b1 \cdot |\dot c|\,dt\bigr)^2 \leq (b - a)\int_a^b|\dot c|^2\,dt = 2(b - a)E(c)$, with equality if and only if $|\dot c|$ is
constant. If $\gamma$ has constant speed and $E(c) < E(\gamma)$, then $L(c)^2 \leq 2(b - a)E(c) < 2(b - a)E(\gamma) = L(\gamma)^2$.
\end{proof}

\begin{proposition}[First variation formula]\label{riemannian-geometry:proposition-first-variation}
Let $f$ be a variation of a piecewise smooth curve $\gamma$, with variation field $V$, and suppose that $\gamma$ is smooth on each $[t_{i-1}, t_i]$. Write
$\Delta_i\dot\gamma = \dot\gamma(t_i^+) - \dot\gamma(t_i^-)$ for the jump of the velocity at $t_i$, $0 < i < k$. Then $s \mapsto E(f_s)$ is smooth, and
\[
\frac{d}{ds}\Big|_{s=0}E(f_s) = \langle V, \dot\gamma\rangle\Big|_a^b - \sum_{i=1}^{k-1}\langle V(t_i), \Delta_i\dot\gamma\rangle - \int_a^b\langle V, D_t\dot\gamma\rangle\,dt .
\]
Consequently $\gamma$ is a critical point of the energy for all proper variations, that is, $\frac{d}{ds}|_{s=0}E(f_s) = 0$ for every proper $f$, if and only if $\gamma$ is a
geodesic.
\end{proposition}

\begin{proof}
On each piece the integrand of $E(f_s) = \sum_i\frac12\int_{t_{i-1}}^{t_i}|\partial_tf|^2\,dt$ is smooth on a compact set, so we may differentiate under the integral sign.
By Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves} and Lemma~\ref{riemannian-geometry:lemma-symmetry},
\[
\frac{\partial}{\partial s}\frac12|\partial_tf|^2 = \langle D_s\partial_tf, \partial_tf\rangle = \langle D_t\partial_sf, \partial_tf\rangle
= \frac{\partial}{\partial t}\langle\partial_sf, \partial_tf\rangle - \langle\partial_sf, D_t\partial_tf\rangle .
\]
At $s = 0$ this is $\partial_t\langle V, \dot\gamma\rangle - \langle V, D_t\dot\gamma\rangle$. Integrating over $[t_{i-1}, t_i]$ gives
$\langle V(t_i), \dot\gamma(t_i^-)\rangle - \langle V(t_{i-1}), \dot\gamma(t_{i-1}^+)\rangle - \int_{t_{i-1}}^{t_i}\langle V, D_t\dot\gamma\rangle\,dt$, and summing over $i$ gives the
formula, since $V$ is continuous.

If $\gamma$ is a geodesic, it is smooth, so there are no jumps, $D_t\dot\gamma = 0$, and $V(a) = V(b) = 0$ for proper variations; so the derivative vanishes. Conversely,
suppose it vanishes for all proper variations. The function $\phi(t) = \prod_i(t - t_i)^2$ is smooth, positive except at the $t_i$, and $V = \phi\,D_t\dot\gamma$ is a
piecewise smooth field vanishing at all the $t_i$; for the variation $\exp_{\gamma(t)}(sV(t))$ the formula gives $0 = -\int_a^b\phi|D_t\dot\gamma|^2\,dt$, so $D_t\dot\gamma = 0$ on
each piece. Next choose a piecewise smooth $V$ with $V(a) = V(b) = 0$ and $V(t_i) = \Delta_i\dot\gamma$, for instance a sum of products of bump functions with parallel
fields along $\gamma$; then $0 = -\sum_i|\Delta_i\dot\gamma|^2$. So $\dot\gamma$ is continuous, and $\gamma$ is a geodesic on each piece whose velocities match at the $t_i$; by
uniqueness of geodesics with given initial velocity, $\gamma$ is a single geodesic.
\end{proof}

In particular a curve of constant speed that minimises length between its endpoints minimises energy (Lemma~\ref{riemannian-geometry:lemma-energy-length}), hence is a
geodesic, which is Lemma~\ref{riemannian-geometry:lemma-minimising-geodesic} again.

\begin{definition}\label{riemannian-geometry:definition-index-form}
Let $\gamma \colon [a, b] \to M$ be a geodesic. The \emph{index form} of $\gamma$ is the symmetric bilinear form on piecewise smooth vector fields along $\gamma$
\[
I(V, W) = \int_a^b\bigl(\langle D_tV, D_tW\rangle - R(V, \dot\gamma, \dot\gamma, W)\bigr)\,dt .
\]
\end{definition}

It is symmetric because $R(V, \dot\gamma, \dot\gamma, W) = R(\dot\gamma, W, V, \dot\gamma) = R(W, \dot\gamma, \dot\gamma, V)$ by
Proposition~\ref{riemannian-geometry:proposition-curvature-symmetries}. If $V$ is smooth on the pieces $[t_{i-1}, t_i]$, integrating
$\frac{d}{dt}\langle D_tV, W\rangle = \langle D_t^2V, W\rangle + \langle D_tV, D_tW\rangle$ over each piece gives
\[
I(V, W) = \langle D_tV, W\rangle\Big|_a^b - \sum_{i=1}^{k-1}\langle\Delta_iD_tV, W(t_i)\rangle - \int_a^b\langle D_t^2V + R(V, \dot\gamma)\dot\gamma, W\rangle\,dt, \tag{$\ddagger$}
\]
with $\Delta_iD_tV = D_tV(t_i^+) - D_tV(t_i^-)$. In particular $I(J, W) = 0$ for a Jacobi field $J$ and every $W$ vanishing at $a$ and $b$.

\begin{proposition}[Second variation formula]\label{riemannian-geometry:proposition-second-variation}
Let $\gamma \colon [a, b] \to M$ be a geodesic and $f$ a variation of $\gamma$ with variation field $V$, and let $A(t) = D_s\partial_sf(0, t)$, the acceleration at $s = 0$ of
the curve $s \mapsto f(s, t)$. Then
\[
\frac{d^2}{ds^2}\Big|_{s=0}E(f_s) = I(V, V) + \langle A, \dot\gamma\rangle\Big|_a^b = \int_a^b\bigl(|D_tV|^2 - R(V, \dot\gamma, \dot\gamma, V)\bigr)\,dt + \langle A, \dot\gamma\rangle\Big|_a^b .
\]
The boundary term vanishes if $f$ is proper, and also if the curves $s \mapsto f(s, a)$ and $s \mapsto f(s, b)$ are geodesics, for instance for
$f(s, t) = \exp_{\gamma(t)}(sV(t))$.
\end{proposition}

\begin{proof}
As in the proof of Proposition~\ref{riemannian-geometry:proposition-first-variation}, on each piece $\frac{d}{ds}E = \int\langle D_t\partial_sf, \partial_tf\rangle\,dt$, and
differentiating again,
\[
\frac{d^2}{ds^2}E(f_s) = \sum_i\int_{t_{i-1}}^{t_i}\bigl(\langle D_sD_t\partial_sf, \partial_tf\rangle + \langle D_t\partial_sf, D_s\partial_tf\rangle\bigr)\,dt .
\]
At $s = 0$, $D_s\partial_tf = D_t\partial_sf = D_tV$, so the second term is $|D_tV|^2$. By Lemma~\ref{riemannian-geometry:lemma-curvature-families},
$D_sD_t\partial_sf = D_tD_s\partial_sf + R(\partial_sf, \partial_tf)\partial_sf$, and at $s = 0$, using $D_t\dot\gamma = 0$,
\[
\langle D_sD_t\partial_sf, \dot\gamma\rangle = \frac{\partial}{\partial t}\langle D_s\partial_sf, \dot\gamma\rangle + R(V, \dot\gamma, V, \dot\gamma)
= \frac{\partial}{\partial t}\langle D_s\partial_sf, \dot\gamma\rangle - R(V, \dot\gamma, \dot\gamma, V).
\]
The field $A$ at $t_i$ is the acceleration of the curve $s \mapsto f(s, t_i)$, which is the same curve for both pieces meeting at $t_i$; so $A$ is continuous, and so is
$\dot\gamma$. Hence the integrals of $\partial_t\langle A, \dot\gamma\rangle$ add up to $\langle A, \dot\gamma\rangle|_a^b$. If $f$ is proper, the curves $s \mapsto f(s, a)$
and $s \mapsto f(s, b)$ are constant, so $A(a) = A(b) = 0$; if they are geodesics, $A(a) = A(b) = 0$ by Proposition~\ref{riemannian-geometry:proposition-geodesic-speed}.
\end{proof}

\begin{theorem}[Jacobi]\label{riemannian-geometry:theorem-jacobi-conjugate}
Let $\gamma \colon [0, L] \to M$ be a geodesic, and suppose that $\gamma(t_0)$ is conjugate to $\gamma(0)$ along $\gamma$ for some $0 < t_0 < L$. Then there are piecewise
smooth curves from $\gamma(0)$ to $\gamma(L)$, arbitrarily close to $\gamma$, that are shorter than $\gamma$. In particular $\gamma$ is not a minimising geodesic.
\end{theorem}

\begin{proof}
Let $J$ be a nonzero Jacobi field on $[0, t_0]$ with $J(0) = J(t_0) = 0$. Then $D_tJ(t_0) \neq 0$, as a Jacobi field is determined by its value and derivative at one
point. Let $V = J$ on $[0, t_0]$ and $V = 0$ on $[t_0, L]$, a piecewise smooth field vanishing at $0$ and $L$. In $(\ddagger)$ for $V$ and a piecewise smooth $U$ vanishing at $0$
and $L$, the integral vanishes, because $J$ satisfies the Jacobi equation on $[0, t_0]$ and $V = 0$ on $[t_0, L]$, and the only remaining term comes from the jump
$\Delta D_tV = -D_tJ(t_0)$ at $t_0$; so $I(V, U) = \langle D_tJ(t_0), U(t_0)\rangle$. In particular $I(V, V) = \langle D_tJ(t_0), J(t_0)\rangle = 0$. Let $P$ be the parallel field
along $\gamma$ with $P(t_0) = -D_tJ(t_0)$, and $W = \sin(\pi t/L)P/\sin(\pi t_0/L)$, which vanishes at $0$ and $L$ and equals $-D_tJ(t_0)$ at $t_0$. Then
$I(V, W) = -|D_tJ(t_0)|^2 < 0$. Hence for small $\delta > 0$,
\[
I(V + \delta W, V + \delta W) = 2\delta I(V, W) + \delta^2I(W, W) < 0 .
\]
The proper variation $f(s, t) = \exp_{\gamma(t)}(s(V + \delta W)(t))$ has $\frac{d}{ds}E(f_s) = 0$ at $s = 0$ (Proposition~\ref{riemannian-geometry:proposition-first-variation})
and $\frac{d^2}{ds^2}E(f_s) < 0$ at $s = 0$ (Proposition~\ref{riemannian-geometry:proposition-second-variation}), so $E(f_s) < E(\gamma)$ for all small $s \neq 0$, and
$L(f_s) < L(\gamma)$ by Lemma~\ref{riemannian-geometry:lemma-energy-length}, as the geodesic $\gamma$ has constant speed.
\end{proof}

For example, a great circle on the unit sphere $S^n$ starting at $p$ meets the conjugate point $-p$ at time $\pi$ (Example~\ref{riemannian-geometry:example-jacobi-constant}),
and so it stops minimising after time $\pi$, in accordance with Remark~\ref{riemannian-geometry:remark-non-unique-geodesics}.

\begin{theorem}[Bonnet--Myers]\label{riemannian-geometry:theorem-bonnet-myers}
Let $(M, g)$ be complete and connected with $\mathrm{Ric} \geq (n - 1)\kappa g$ for some $\kappa > 0$. Then $M$ has diameter at most $\pi/\sqrt\kappa$; hence $M$ is compact, and $\pi_1(M)$ is finite.
\end{theorem}

\begin{proof}
Let $p, q \in M$ and $L = d(p, q)$, and suppose $L > \pi/\sqrt\kappa$. By Theorem~\ref{riemannian-geometry:theorem-hopf-rinow} there is a minimising geodesic
$\gamma \colon [0, L] \to M$ of unit speed from $p$ to $q$. Let $E_1, \ldots, E_{n-1}$ be parallel fields along $\gamma$ that together with $\dot\gamma$ form an orthonormal basis
at one point, hence at every point (Proposition~\ref{riemannian-geometry:proposition-geodesic-speed}), and put $V_i(t) = \sin(\pi t/L)E_i(t)$. Then $D_tV_i = (\pi/L)\cos(\pi t/L)E_i$,
and since $\sum_iR(E_i, \dot\gamma, \dot\gamma, E_i) = \mathrm{Ric}(\dot\gamma, \dot\gamma) \geq (n - 1)\kappa$ (Definition~\ref{riemannian-geometry:definition-curvature-tensor}),
\[
\sum_{i=1}^{n-1}I(V_i, V_i) = \int_0^L\Bigl((n - 1)\frac{\pi^2}{L^2}\cos^2\frac{\pi t}{L} - \sin^2\frac{\pi t}{L}\,\mathrm{Ric}(\dot\gamma, \dot\gamma)\Bigr)dt
\leq (n - 1)\frac L2\Bigl(\frac{\pi^2}{L^2} - \kappa\Bigr) < 0,
\]
using $\int_0^L\cos^2(\pi t/L)\,dt = \int_0^L\sin^2(\pi t/L)\,dt = L/2$. So $I(V_i, V_i) < 0$ for some $i$, and as in the proof of
Theorem~\ref{riemannian-geometry:theorem-jacobi-conjugate} the proper variation $\exp_{\gamma(t)}(sV_i(t))$ gives curves from $p$ to $q$ shorter than $L = d(p, q)$, which is
absurd. Hence every two points have distance at most $\pi/\sqrt\kappa$, and $M$, being closed and bounded in itself, is compact by
Theorem~\ref{riemannian-geometry:theorem-hopf-rinow}.

For the fundamental group, let $\pi \colon \widetilde{M} \to M$ be a universal covering (Theorem~\ref{homotopy:theorem-universal-cover}). It is a manifold
(Proposition~\ref{topological-manifolds:proposition-constructions}(4)) with a unique smooth structure making $\pi$ a local diffeomorphism
(Example~\ref{smooth-manifolds:example-smooth-quotients}), and $\tilde g = \pi^*g$ makes $\pi$ a local isometry, so $\widetilde{M}$ satisfies the same bound on the Ricci
curvature. In charts $y$ of $M$ and $y \circ \pi$ of $\widetilde{M}$ the metrics have the same coefficients, hence the same geodesic equations; so $\pi$ and its local inverses
map geodesics to geodesics. Consequently $\widetilde{M}$ is complete: if $\tilde\gamma$ is a geodesic of $\widetilde{M}$, the geodesic $\pi \circ \tilde\gamma$ extends to $\RR$
by completeness of $M$, its lift through $\tilde\gamma(0)$ (Proposition~\ref{homotopy:proposition-homotopy-lifting}) is locally a geodesic followed by a local inverse of $\pi$,
hence a geodesic, and it extends $\tilde\gamma$ to $\RR$. By the first part $\widetilde{M}$ is compact, so the fibre $\pi^{-1}(p)$, which is closed and discrete, is finite. By
Corollary~\ref{homotopy:corollary-lifting-consequences}(3), $\pi_1(M, p)$ acts on this fibre transitively with trivial stabiliser, since $\widetilde{M}$ is simply connected;
so $\pi_1(M, p)$ is finite.
\end{proof}

For example the round sphere of radius $1/\sqrt\kappa$ has $\mathrm{Ric} = (n - 1)\kappa g$ and diameter exactly $\pi/\sqrt\kappa$, so the bound is sharp; Cheng's theorem
(Theorem~\ref{riemannian-geometry:theorem-maximal-diameter}) says that it is the only case of equality. The hypothesis cannot be weakened to $\mathrm{Ric} > 0$: the paraboloid
$z = x^2 + y^2$ is complete, being closed in $\RR^3$, and has Gaussian curvature $4/(1 + 4x^2 + 4y^2)^2 > 0$
(Section~\ref{riemannian-geometry:section-submanifolds}), but it is not compact; its curvature tends to $0$ at infinity.

\begin{remark}\label{riemannian-geometry:remark-noncompact-positive}
Compactness in Theorem~\ref{riemannian-geometry:theorem-bonnet-myers} comes from the \emph{uniform} bound $\mathrm{Ric} \geq (n - 1)\kappa g$ with $\kappa > 0$, not from
positivity alone: a non-compact manifold can carry complete metrics with $K > 0$, such as the paraboloid, but no complete metric with $\mathrm{Ric} \geq \delta g$ for a
constant $\delta > 0$. On a compact manifold, positive curvature is automatically bounded below by a positive constant. Complete non-compact manifolds with $K > 0$ are
nevertheless very restricted: by the theorem of Gromoll and Meyer \cite{gromoll-meyer-1969} they are diffeomorphic to $\RR^n$, in particular contractible and simply
connected. More generally, by the soul theorem of Cheeger and Gromoll \cite{cheeger-gromoll-1972}, a complete non-compact manifold with $K \geq 0$ is diffeomorphic to the
normal bundle of a compact totally geodesic submanifold, its \emph{soul}, and Perelman \cite{perelman-1994} proved that the soul is a point as soon as all sectional
curvatures are positive at one point. Neither completeness nor strict positivity can be dropped: the sphere minus two antipodal points has $K = 1$ and $\pi_1 = \ZZ$
(Counterexample~\ref{riemannian-geometry:counterexample-synge} below), and the flat cylinder $S^1 \times \RR$ is complete with $K = 0$ and $\pi_1 = \ZZ$; its soul is a circle.
\end{remark}


The second variation formula also controls the topology of compact manifolds of positive curvature through their isometries. The key step is a fixed point theorem.

\begin{theorem}[Weinstein]\label{riemannian-geometry:theorem-weinstein}
Let $M$ be a compact connected oriented Riemannian manifold of dimension $n$ with positive sectional curvature, and $F$ an isometry of $M$ that preserves the
orientation if $n$ is even and reverses it if $n$ is odd. Then $F$ has a fixed point.
\end{theorem}

\begin{proof}
Suppose not. The function $x \mapsto d(x, F(x))$ is continuous and positive on the compact $M$; let $x_0$ be a minimum point, $L = d(x_0, F(x_0)) > 0$, and
$\gamma \colon [0, L] \to M$ a minimising geodesic of unit speed from $x_0$ to $F(x_0)$ (Theorem~\ref{riemannian-geometry:theorem-hopf-rinow}). An isometry maps geodesics to
geodesics, since in charts $y$ and $y \circ F$ the metric has the same coefficients (as in the proof of Theorem~\ref{riemannian-geometry:theorem-bonnet-myers}), and it
preserves lengths of curves, hence distances.

\emph{Step 1: $d_{x_0}F(\dot\gamma(0)) = \dot\gamma(L)$.} Let $m = \gamma(L/2)$. Then
$d(m, F(m)) \leq d(m, F(x_0)) + d(F(x_0), F(m)) = L/2 + d(x_0, m) = L$, and $d(m, F(m)) \geq L$ by the choice of $x_0$. So the curve that follows $\gamma$ from $m$ to $F(x_0)$
and then $F \circ \gamma$ from $F(x_0)$ to $F(m)$ has unit speed and length $d(m, F(m))$; by Lemma~\ref{riemannian-geometry:lemma-minimising-geodesic} it is a geodesic,
hence smooth at $F(x_0)$, where its two velocities are $\dot\gamma(L)$ and $d_{x_0}F(\dot\gamma(0))$.

\emph{Step 2: a closed-up parallel field.} Let $P \colon T_{x_0}M \to T_{F(x_0)}M$ be parallel transport along $\gamma$. It is a linear isometry
(Proposition~\ref{riemannian-geometry:proposition-geodesic-speed}) and preserves orientations, as in the proof of
Proposition~\ref{riemannian-geometry:proposition-holonomy-basic}(4). So $B = P^{-1} \circ d_{x_0}F$ is an orthogonal map of $T_{x_0}M$ with $\det B = 1$ if $F$ preserves the
orientation and $\det B = -1$ if it reverses it; in both cases of the theorem $\det B = (-1)^n$. As $\dot\gamma$ is parallel, $P(\dot\gamma(0)) = \dot\gamma(L)$, so $B$ fixes
$\dot\gamma(0)$ by Step 1, and preserves the orthogonal complement $E$ of $\dot\gamma(0)$, of dimension $m = n - 1$, with $\det(B|_E) = (-1)^{m+1}$. An orthogonal map $C$ of an
$m$-dimensional space with $\det C = (-1)^{m+1}$ has the eigenvalue $1$:
$\det(C - I) = \det C\,\det(I - C^T) = \det C\,\det(I - C) = (-1)^{m+1}(-1)^m\det(C - I) = -\det(C - I)$. So there is a unit vector $v \in E$ with $Bv = v$. Let $V$ be the parallel
field along $\gamma$ with $V(0) = v$; then $|V| = 1$, $V \perp \dot\gamma$, and $V(L) = Pv = d_{x_0}F(v)$.

\emph{Step 3: a shorter curve.} Let $f(s, t) = \exp_{\gamma(t)}(sV(t))$ and $x_s = \exp_{x_0}(sv)$. Then $f(s, 0) = x_s$ and, since $F$ maps geodesics to geodesics,
$f(s, L) = \exp_{F(x_0)}(s\,d_{x_0}F(v)) = F(x_s)$. By Proposition~\ref{riemannian-geometry:proposition-first-variation} the first derivative of $E(f_s)$ at $s = 0$ is
$\langle V, \dot\gamma\rangle|_0^L = 0$. By Proposition~\ref{riemannian-geometry:proposition-second-variation}, whose boundary term vanishes because the curves
$s \mapsto f(s, t)$ are geodesics, and since $D_tV = 0$,
\[
\frac{d^2}{ds^2}\Big|_{s=0}E(f_s) = -\int_0^LR(V, \dot\gamma, \dot\gamma, V)\,dt < 0,
\]
the integrand being the sectional curvature of the plane spanned by the orthonormal vectors $V$ and $\dot\gamma$. So $E(f_s) < E(\gamma)$ for small $s \neq 0$, and
$L(f_s) < L$ by Lemma~\ref{riemannian-geometry:lemma-energy-length}. But $f_s$ joins $x_s$ to $F(x_s)$, so $L(f_s) \geq d(x_s, F(x_s)) \geq L$, a contradiction.
\end{proof}

\begin{corollary}[Synge]\label{riemannian-geometry:corollary-synge}
Let $M$ be a compact connected Riemannian manifold with positive sectional curvature.
\begin{enumerate}
\item If $n$ is even and $M$ is orientable, then $M$ is simply connected.
\item If $n$ is odd, then $M$ is orientable.
\end{enumerate}
\end{corollary}

\begin{proof}
(1) The sectional curvature $K(u, v) = R(u, v, v, u)$ is continuous and positive on the compact set of orthonormal pairs of tangent vectors, so $K \geq k$ for some $k > 0$,
and $\mathrm{Ric} \geq (n - 1)kg$. Let $\pi \colon \widetilde{M} \to M$ be a universal covering with the metric $\pi^*g$. As in the proof of
Theorem~\ref{riemannian-geometry:theorem-bonnet-myers}, $\widetilde{M}$ is complete, hence compact by that theorem, and it has positive sectional curvature. Orient
$\widetilde{M}$ by pulling back a volume form $\mu$ of $M$ by the local diffeomorphism $\pi$. A deck transformation $\tau$ satisfies $\pi \circ \tau = \pi$, so it is an isometry
and $\tau^*\pi^*\mu = \pi^*\mu$: it preserves the orientation. By Theorem~\ref{riemannian-geometry:theorem-weinstein}, $\tau$ has a fixed point, so it is the identity, by the
uniqueness of lifts (Proposition~\ref{homotopy:proposition-lifting-criterion}). Hence $\mathrm{Deck}(\pi)$ is trivial, and so is $\pi_1(M) \cong \mathrm{Deck}(\pi)$
(Proposition~\ref{homotopy:proposition-deck}).

(2) Suppose $M$ is not orientable, and let $p \colon \widehat{M} \to M$ be the orientation double covering (Definition~\ref{poincare-duality:definition-orientation-covering}),
which is connected by Proposition~\ref{poincare-duality:proposition-orientation-covering}(1). It is compact, a smooth manifold for which $p$ is a local diffeomorphism
(Example~\ref{smooth-manifolds:example-smooth-quotients}), and it has positive sectional curvature for $p^*g$. It is orientable: at a point $(x, \alpha)$, $p$ identifies the
local homology of $\widehat{M}$ with $H_n(M \mid x; \ZZ)$ (a sheet over a coordinate ball is mapped homeomorphically), and $(x, \alpha) \mapsto \alpha$ is a continuous choice of
generators, that is an orientation (Proposition~\ref{differential-forms:proposition-orientation}). The deck transformation $\tau(x, \alpha) = (x, -\alpha)$ is an isometry of
$p^*g$ without fixed points, and it reverses this orientation, because $p \circ \tau = p$ carries the generator $\alpha$ at $(x, \alpha)$ to $\alpha$ at $(x, -\alpha)$, where the
chosen generator is $-\alpha$. By Theorem~\ref{riemannian-geometry:theorem-weinstein} with $n$ odd, $\tau$ has a fixed point, a contradiction.
\end{proof}

\begin{counterexample}\label{riemannian-geometry:counterexample-synge}
Every hypothesis of Theorem~\ref{riemannian-geometry:theorem-weinstein} and Corollary~\ref{riemannian-geometry:corollary-synge} is needed.
\begin{enumerate}
\item \emph{The parity conditions in Weinstein's theorem.} The antipodal map $x \mapsto -x$ of the unit sphere $S^n$ is an isometry without fixed points. It is the restriction
of $-I$, of determinant $(-1)^{n+1}$, so it preserves the orientation of $S^n$ when $n$ is odd and reverses it when $n$ is even: exactly the cases excluded by the theorem.
\item \emph{Orientability in (1).} $\RR\PP^n = S^n/\{\pm1\}$ with the round metric has $K = 1$ and $\pi_1 = \ZZ/2$. For $n$ even it is not orientable
(Exercise~\ref{differential-forms:exercise-orientability-projective}).
\item \emph{Even dimension in (1).} For $n$ odd, $\RR\PP^n$ is orientable with $K = 1$ and $\pi_1 = \ZZ/2$; so are the lens spaces $S^3/(\ZZ/p)$
(Example~\ref{topological-manifolds:example-covering-actions}), with $\pi_1 = \ZZ/p$, as the group acts by unitary, hence orthogonal, maps.
\item \emph{Odd dimension in (2).} $\RR\PP^2$ has $K = 1$ and is not orientable.
\item \emph{Positive curvature, not just $K \geq 0$.} The flat torus $\RR^2/\ZZ^2$ is compact and orientable of even dimension, with $\pi_1 = \ZZ^2$. The product
$\RR\PP^2 \times S^1$ of the round $\RR\PP^2$ and a circle is compact of dimension $3$ with $K \geq 0$ (the curvature tensor of a Riemannian product is the sum of those of the factors, and the circle is flat), and it is not
orientable: by Exercise~\ref{riemannian-geometry:exercise-holonomy-product} its holonomy contains $h \times 1$ for $h \in \mathrm{Hol}(\RR\PP^2)$ with $\det h = -1$, which
exists by Proposition~\ref{riemannian-geometry:proposition-holonomy-basic}(4).
\item \emph{Compactness in (1).} The sphere minus two antipodal points, $S^2 \setminus \{\pm e_3\}$, has $K = 1$ and is orientable of even dimension, but by stereographic
projection it is homeomorphic to $\RR^2 \setminus \{0\}$, with $\pi_1 = \ZZ$.
\item \emph{Compactness in (2).} Let $A = \mathrm{diag}(-1, -1, -1, 1)$ acting on $U = S^3 \setminus \{\pm e_4\}$. Its fixed points on $S^3$ are $\pm e_4$, so the group
$\{1, A\}$ acts freely, and properly as it is finite, by isometries; $\det A = -1$, so $A$ reverses the orientation of $S^3$ (defined by the outward normal, which $A$
preserves). The quotient $U/\{1, A\}$ has $K = 1$ (Proposition~\ref{riemannian-geometry:proposition-quotient-metric}) and dimension $3$, and it is not orientable: an
orientation would pull back to an orientation of the connected $U$ preserved by $A$.
\end{enumerate}
\end{counterexample}

\begin{example}\label{riemannian-geometry:example-synge}
$\RR\PP^2 \times \RR\PP^2$ carries no metric of positive sectional curvature. Its fundamental group $\ZZ/2 \times \ZZ/2$ (Proposition~\ref{homotopy:proposition-product}) is
nontrivial, so by Corollary~\ref{riemannian-geometry:corollary-synge}(1) it could only be non-orientable. But then its orientation double covering would be a compact
orientable $4$-manifold of positive curvature whose fundamental group has order $2$, a subgroup of index $2$ of $\ZZ/2 \times \ZZ/2$
(Corollary~\ref{homotopy:corollary-lifting-consequences}), contradicting (1). Whether $S^2 \times S^2$ carries a metric of positive sectional curvature is a famous open question
of Hopf; Synge's theorem does not decide it, as $S^2 \times S^2$ is simply connected.
\end{example}

\begin{remark}\label{riemannian-geometry:remark-comparison}
Theorems~\ref{riemannian-geometry:theorem-cartan-hadamard} and~\ref{riemannian-geometry:theorem-bonnet-myers} are the simplest \emph{comparison theorems}, which compare a manifold with a space of constant curvature
through its Jacobi fields. The Rauch comparison theorem \cite[Chapter 10]{do-carmo-riemannian} and the Bishop--Gromov volume comparison \cite{lee-riemannian} are their refinements, and they lead to the sphere theorems and to
Gromov's compactness theorem.
\end{remark}

A different kind of rigidity concerns straight lines. A \emph{ray} in a Riemannian manifold is a unit speed geodesic $\gamma \colon [0, \infty) \to M$ with
$d(\gamma(s), \gamma(t)) = |s - t|$ for all $s, t$, and a \emph{line} is a unit speed geodesic $\gamma \colon \RR \to M$ with the same property for all $s, t \in \RR$: it
minimises length between any two of its points.

\begin{proposition}\label{riemannian-geometry:proposition-rays-lines}
Let $M$ be complete and connected.
\begin{enumerate}
\item If $M$ is not compact, every point is the starting point of a ray.
\item If there are a compact $C \subseteq M$ and points $p_k$, $q_k$ with $d(p_k, C) \to \infty$ and $d(q_k, C) \to \infty$ such that every curve from $p_k$ to $q_k$ meets
$C$, then $M$ contains a line.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) By Theorem~\ref{riemannian-geometry:theorem-hopf-rinow} closed bounded sets are compact, so $M$ is unbounded: choose $q_k$ with $L_k = d(p, q_k) \to \infty$ and minimising
unit speed geodesics $\gamma_k(t) = \exp_p(tv_k)$, $t \in [0, L_k]$, $|v_k| = 1$. By compactness of the unit sphere a subsequence has $v_k \to v$; let $\gamma(t) = \exp_p(tv)$, defined for
all $t$. For fixed $s, t \geq 0$ and $k$ large, $d(\gamma_k(s), \gamma_k(t)) = |s - t|$, since a subarc of a minimising curve is minimising; by continuity of $\exp$ and of $d$,
$d(\gamma(s), \gamma(t)) = |s - t|$.

(2) Let $\sigma_k$ be a minimising unit speed geodesic from $p_k$ to $q_k$. It meets $C$; shift its parameter so that $\sigma_k(0) \in C$, and $\sigma_k$ is defined on
$[-a_k, b_k]$ with $a_k = d(p_k, \sigma_k(0)) \geq d(p_k, C)$ and $b_k \geq d(q_k, C)$, both tending to $\infty$. The unit tangent vectors at points of the compact $C$ form a compact
set, so a subsequence has $\dot\sigma_k(0) \to v \in T_xM$, $x \in C$. As in (1), $\gamma(t) = \exp_x(tv)$ satisfies $d(\gamma(s), \gamma(t)) = |s - t|$ for all $s, t \in \RR$.
\end{proof}

For example, on the flat cylinder $S^{n-1} \times \RR$ the curves $\{x\} \times \RR$ are lines, and (2) applies with $C = S^{n-1} \times \{0\}$. On the hyperbolic plane every
geodesic is a line: it is complete (see the discussion after Definition~\ref{riemannian-geometry:definition-model-spaces}) with $K = -1$
(Example~\ref{riemannian-geometry:example-conformal-metrics}(1)), so by Theorem~\ref{riemannian-geometry:theorem-cartan-hadamard} each $\exp_p$ is a diffeomorphism, so the geodesic from $\gamma(s)$ through $\gamma(t)$ is the only one joining them, and the minimising geodesic given by
Theorem~\ref{riemannian-geometry:theorem-hopf-rinow} must be it.

\begin{theorem}[Cheeger--Gromoll splitting theorem]\label{riemannian-geometry:theorem-splitting}
Let $M$ be a complete connected Riemannian manifold with $\mathrm{Ric} \geq 0$ that contains a line. Then $M$ is isometric to a Riemannian product $N \times \RR$, where $N$ is a
complete manifold with $\mathrm{Ric} \geq 0$, and the line corresponds to a curve $\{x\} \times \RR$.
\end{theorem}

This is \cite{cheeger-gromoll-1971}. The proof uses the \emph{Busemann functions} $b^\pm(x) = \lim_{t \to \infty}\bigl(t - d(x, \gamma(\pm t))\bigr)$ of the line $\gamma$. The
triangle inequality gives $b^+ + b^- \leq 0$, with equality on $\gamma$. The condition $\mathrm{Ric} \geq 0$ gives, through the comparison of the Laplacian of distance functions
with the Euclidean one, that $b^+$ and $b^-$ are subharmonic in a weak sense; so $b^+ + b^-$ is subharmonic, and by the maximum principle it vanishes identically. Hence
$b^+ = -b^-$ is harmonic and smooth, with $|\nabla b^+| = 1$, and the Bochner formula $0 = \frac12\Delta|\nabla b^+|^2 = |\mathrm{Hess}\,b^+|^2 + \mathrm{Ric}(\nabla b^+, \nabla b^+)$
(with the analysts' sign of the Laplacian) forces $\mathrm{Hess}\,b^+ = 0$: the gradient of $b^+$ is a parallel unit vector field, and its flow splits off the factor $\RR$.
The same paper shows that the universal covering of a compact manifold with $\mathrm{Ric} \geq 0$ is isometric to $N \times \RR^k$ with $N$ compact.

\begin{corollary}\label{riemannian-geometry:corollary-no-lines}
A complete connected manifold with positive definite Ricci curvature, for instance with $K > 0$, contains no line. In particular it satisfies the hypothesis of
Proposition~\ref{riemannian-geometry:proposition-rays-lines}(2) for no compact $C$: outside a large compact set it cannot be separated into two unbounded pieces.
\end{corollary}

\begin{proof}
If it contained a line, it would be isometric to $N \times \RR$ by Theorem~\ref{riemannian-geometry:theorem-splitting}. In coordinates $(x, t)$ adapted to the product, the
metric is $g_N + dt^2$, and the Christoffel symbols with an index $t$ vanish (Theorem~\ref{riemannian-geometry:theorem-levi-civita}); so $\nabla\partial_t = 0$,
$R(X, \partial_t)\partial_t = 0$ for all $X$, and $\mathrm{Ric}(\partial_t, \partial_t) = 0$, a contradiction. The second statement follows from
Proposition~\ref{riemannian-geometry:proposition-rays-lines}(2).
\end{proof}

Both hypotheses of the splitting theorem are needed: the hyperbolic plane contains lines but is not a product, since a product $N \times \RR$ of dimension $2$ is flat, and
the flat cylinder shows that the factor $N$ need not be $\RR^{n-1}$. The paraboloid $z = x^2 + y^2$, which has $K > 0$, contains rays from every point but no line.

\section{Submanifolds and the second fundamental form}\label{riemannian-geometry:section-submanifolds}

Classical differential geometry studies curves and surfaces in $\RR^3$ through two quadratic forms: the \emph{first fundamental form}, the metric induced from
$\RR^3$, and the \emph{second fundamental form}, which measures how the surface bends in the ambient space. This section develops both for a submanifold of
any Riemannian manifold, proves the equations of Gauss and Codazzi that relate them to the curvature, and then specialises to surfaces in $\RR^3$, where Gauss's
equation becomes the Theorema Egregium.

Throughout, $(\overline{M}, \bar{g})$ is a Riemannian manifold with Levi-Civita connection $\overline{\nabla}$ and curvature $\overline{R}$, and
$M \subseteq \overline{M}$ is a submanifold with the induced metric $g = \bar{g}|_{TM}$, the \emph{first fundamental form}; we write $\langle -, - \rangle$ for
both. For $p \in M$, $T_p\overline{M} = T_pM \oplus N_pM$ with $N_pM = (T_pM)^\perp$, the \emph{normal space}, and $v = v^\top + v^\perp$ is the corresponding
decomposition. The normal spaces form the \emph{normal bundle} $NM$, a vector bundle over $M$: near each point it has a smooth frame, obtained by applying
Gram--Schmidt to a frame of $T\overline{M}$ extending one of $TM$. For $\overline{M} = \RR^N$ with the Euclidean metric, $\overline{\nabla}_XY$ is the directional
derivative of the components of $Y$, and $\overline{R} = 0$.

\begin{lemma}\label{riemannian-geometry:lemma-ambient-derivative}
Every vector field on an open subset of $M$ extends, near each point, to a vector field on an open subset of $\overline{M}$. For vector fields $X$, $Y$ on $M$ and
extensions $\overline{X}$, $\overline{Y}$, the value of $\overline{\nabla}_{\overline{X}}\overline{Y}$ at $p \in M$ depends only on $X$ and $Y$, and we denote it by
$\overline{\nabla}_XY$. Moreover $[\overline{X}, \overline{Y}]$ restricts on $M$ to $[X, Y]$.
\end{lemma}

\begin{proof}
In a slice chart $(x^1, \ldots, x^k, y^1, \ldots, y^m)$ of $\overline{M}$, in which $M$ is $\{y = 0\}$, a vector field $\sum_iY^i(x)\partial/\partial x^i$ on $M$ extends by
the same formula, with coefficients independent of $y$. The value of $\overline{\nabla}_{\overline{X}}\overline{Y}$ at $p$ depends only on $\overline{X}_p = X_p$, since a
connection is $C^\infty$-linear in its first argument, and it equals $D_t(\overline{Y} \circ \gamma)(0)$ for any curve $\gamma$ with $\dot\gamma(0) = X_p$
(Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves}); taking $\gamma$ in $M$, $\overline{Y} \circ \gamma = Y \circ \gamma$. Finally $X$ and
$\overline{X}$ are related by the inclusion $\iota$, and so are $Y$ and $\overline{Y}$, so $[X, Y]$ and $[\overline{X}, \overline{Y}]$ are $\iota$-related
(Lemma~\ref{lie-groups:lemma-related-fields}).
\end{proof}

\begin{theorem}[Gauss formula]\label{riemannian-geometry:theorem-gauss-formula}
For vector fields $X$, $Y$ tangent to $M$,
\[
\overline{\nabla}_XY = \nabla_XY + \mathrm{II}(X, Y),
\]
where $\nabla_XY = (\overline{\nabla}_XY)^\top$ is the Levi-Civita connection of $(M, g)$, and $\mathrm{II}(X, Y) = (\overline{\nabla}_XY)^\perp$, the \emph{second fundamental
form}, is a symmetric $C^\infty(M)$-bilinear map with values in the normal bundle.
\end{theorem}

\begin{proof}
The operator $\nabla^\top_XY = (\overline{\nabla}_XY)^\top$ is a connection on $TM$: it is $C^\infty$-linear in $X$, and
$\overline{\nabla}_X(fY) = (Xf)Y + f\overline{\nabla}_XY$ with $Y$ tangent gives the Leibniz rule. It is metric: for tangent $Y$ and $Z$,
$X\langle Y, Z\rangle = \langle\overline{\nabla}_XY, Z\rangle + \langle Y, \overline{\nabla}_XZ\rangle = \langle\nabla^\top_XY, Z\rangle + \langle Y, \nabla^\top_XZ\rangle$, as the normal parts are orthogonal to
$Y$ and $Z$. By Lemma~\ref{riemannian-geometry:lemma-ambient-derivative} and the torsion-freeness of $\overline{\nabla}$,
$\overline{\nabla}_XY - \overline{\nabla}_YX = [X, Y]$, which is tangent. Its tangent part says that $\nabla^\top$ is torsion-free, so $\nabla^\top = \nabla$ by the uniqueness in
Theorem~\ref{riemannian-geometry:theorem-levi-civita}; its normal part says $\mathrm{II}(X, Y) = \mathrm{II}(Y, X)$. Finally $\mathrm{II}$ is $C^\infty$-linear in $X$, and in $Y$
because $\mathrm{II}(X, fY) = ((Xf)Y + f\overline{\nabla}_XY)^\perp = f\,\mathrm{II}(X, Y)$.
\end{proof}

So the intrinsic connection of $M$ is the tangential part of the ambient one: for a surface in $\RR^3$, $\nabla_XY$ is the derivative of $Y$ in $\RR^3$ projected
back onto the tangent plane. A curve in $M$ is a geodesic of $M$ exactly when its acceleration in $\overline{M}$ is normal to $M$; for instance great circles
on the sphere (Exercise~\ref{riemannian-geometry:exercise-great-circles}).

The Gauss formula does not need the ambient extensions of Lemma~\ref{riemannian-geometry:lemma-ambient-derivative}, and it holds for immersions that are not injective, such
as a curve with self-intersections. Let $f \colon M \to \overline{M}$ be an immersion with the induced metric $g = f^*\bar g$, and let $\overline{\nabla}^f = f^*\overline{\nabla}$ be the
pullback connection on $f^*T\overline{M}$ (Proposition~\ref{connections:proposition-pullback-tensor}(1)): it differentiates sections of $f^*T\overline{M}$, such as $df(Y)$ for a
vector field $Y$ on $M$, in the direction of tangent vectors of $M$, and $\overline{\nabla}^f_X(s \circ f) = \overline{\nabla}_{df(X)}s$ for a vector field $s$ on $\overline{M}$.
The differential identifies $TM$ with the subbundle $df(TM)$ of $f^*T\overline{M}$, whose orthogonal complement $N_f$ is the normal bundle of $f$.

One often writes $\overline{\nabla}_{df(X)}(df(Y))$ for $\overline{\nabla}^f_X(df(Y))$. This is justified by
Remark~\ref{connections:remark-along-curves}: if near $p$ the section $df(Y)$ is of the form $\overline{Y} \circ f$ for a vector field $\overline{Y}$ on an open set of
$\overline{M}$, then $\overline{\nabla}^f_X(df(Y)) = \overline{\nabla}_{df(X)}\overline{Y}$, whatever the extension $\overline{Y}$; for an immersion such extensions exist near
every point, since $f$ is an embedding near each point. But $df(Y)$ itself is not a vector field on $\overline{M}$: it is defined only along $f$, and at a self-intersection
$f(p) = f(q)$ it has two values. For maps that are not immersions the abbreviation is even misleading: for a constant curve $c$ and a nonconstant field $V$ along it,
$\dot c = 0$ but $(c^*\overline{\nabla})_{d/dt}V = \dot V \neq 0$.

\begin{proposition}[Gauss formula for immersions]\label{riemannian-geometry:proposition-gauss-immersion}
For vector fields $X$, $Y$ on $M$,
\[
\overline{\nabla}^f_X\bigl(df(Y)\bigr) = df(\nabla_XY) + \mathrm{II}(X, Y),
\]
where $\nabla$ is the Levi-Civita connection of $g$ and $\mathrm{II}(X, Y) = (\overline{\nabla}^f_X(df(Y)))^\perp$ is a symmetric $C^\infty(M)$-bilinear map with values in $N_f$.
In particular $df(\nabla_XY) = (\overline{\nabla}^f_X(df(Y)))^\top$. For an embedding this is Theorem~\ref{riemannian-geometry:theorem-gauss-formula}, and for
$\overline{M} = \RR^N$, $\overline{\nabla}^f_X(df(Y))$ is the derivative $X(df(Y))$ of the $\RR^N$-valued function $df(Y)$.
\end{proposition}

\begin{proof}
\emph{$\overline{\nabla}^f$ is metric:} $X\langle\sigma, \tau\rangle = \langle\overline{\nabla}^f_X\sigma, \tau\rangle + \langle\sigma, \overline{\nabla}^f_X\tau\rangle$ for sections
$\sigma$, $\tau$ of $f^*T\overline{M}$. Both sides satisfy the same Leibniz rule when $\sigma$ or $\tau$ is multiplied by a function, and locally $f^*T\overline{M}$ has frames
$s_a \circ f$ pulled back from frames of $T\overline{M}$; for $\sigma = s \circ f$ and $\tau = t \circ f$ the identity is
$df(X)\langle s, t\rangle = \langle\overline{\nabla}_{df(X)}s, t\rangle + \langle s, \overline{\nabla}_{df(X)}t\rangle$ at $f(p)$.

\emph{Symmetry:} $\overline{\nabla}^f_X(df(Y)) - \overline{\nabla}^f_Y(df(X)) = df([X, Y])$. The difference of the two sides is $C^\infty(M)$-bilinear in $X$ and $Y$ (the terms with
derivatives of a function cancel, as for the torsion), so it suffices to take coordinate fields $X = \partial_i$, $Y = \partial_j$ of a chart $u$ of $M$. With a chart $y$ of
$\overline{M}$ and $f^a = y^a \circ f$, $df(\partial_j) = \sum_a\partial_jf^a\,(\partial_a \circ f)$ and
\[
\overline{\nabla}^f_{\partial_i}\bigl(df(\partial_j)\bigr) = \sum_a\Bigl(\partial_i\partial_jf^a + \sum_{b,c}\overline{\Gamma}^a_{bc}(f)\,\partial_if^b\,\partial_jf^c\Bigr)(\partial_a \circ f),
\]
which is symmetric in $i$ and $j$ since $\overline{\Gamma}^a_{bc} = \overline{\Gamma}^a_{cb}$; and $[\partial_i, \partial_j] = 0$.

\emph{Conclusion:} as in the proof of Theorem~\ref{riemannian-geometry:theorem-gauss-formula}, $\nabla^{\prime}_XY = df^{-1}\bigl((\overline{\nabla}^f_X(df(Y)))^\top\bigr)$ is a connection
on $TM$; it is metric for $g$ because $\overline{\nabla}^f$ is metric and $N_f \perp df(TM)$, and torsion-free by the tangential part of the symmetry. So $\nabla^{\prime} = \nabla$
(Theorem~\ref{riemannian-geometry:theorem-levi-civita}); the normal part of the symmetry gives $\mathrm{II}(X, Y) = \mathrm{II}(Y, X)$, and $\mathrm{II}$ is $C^\infty$-linear in $X$ and
in $Y$ as before. For an embedding and an extension $\overline{Y}$ of $Y$, $df(Y) = \overline{Y} \circ f$, so $\overline{\nabla}^f_X(df(Y)) = \overline{\nabla}_X\overline{Y}$. For
$\overline{M} = \RR^N$ the Christoffel symbols vanish.
\end{proof}

\begin{definition}\label{riemannian-geometry:definition-shape-operator}
For a normal vector field $\nu$ along $M$, the \emph{shape operator} (or \emph{Weingarten map}) is $S_\nu(X) = -(\overline{\nabla}_X\nu)^\top$, and the \emph{normal
connection} is $\nabla^\perp_X\nu = (\overline{\nabla}_X\nu)^\perp$.
\end{definition}

\begin{proposition}[Weingarten equation]\label{riemannian-geometry:proposition-weingarten}
For tangent $X$, $Y$ and normal $\nu$, $\langle S_\nu X, Y\rangle = \langle\mathrm{II}(X, Y), \nu\rangle$. In particular $S_\nu$ is self-adjoint on each tangent space, and
$\overline{\nabla}_X\nu = -S_\nu X + \nabla^\perp_X\nu$.
\end{proposition}

\begin{proof}
Differentiate $\langle Y, \nu\rangle = 0$: $0 = \langle\overline{\nabla}_XY, \nu\rangle + \langle Y, \overline{\nabla}_X\nu\rangle = \langle\mathrm{II}(X, Y), \nu\rangle - \langle Y, S_\nu X\rangle$. Self-adjointness
follows from the symmetry of $\mathrm{II}$, and the last formula is the decomposition of $\overline{\nabla}_X\nu$ into its two parts.
\end{proof}

\begin{theorem}[Gauss and Codazzi equations]\label{riemannian-geometry:theorem-gauss-codazzi}
For vector fields $X, Y, Z, W$ tangent to $M$:
\begin{enumerate}
\item (Gauss) $R(X, Y, Z, W) = \overline{R}(X, Y, Z, W) + \langle\mathrm{II}(Y, Z), \mathrm{II}(X, W)\rangle - \langle\mathrm{II}(X, Z), \mathrm{II}(Y, W)\rangle$;
\item (Codazzi) $(\overline{R}(X, Y)Z)^\perp = (\nabla_X\mathrm{II})(Y, Z) - (\nabla_Y\mathrm{II})(X, Z)$, where
$(\nabla_X\mathrm{II})(Y, Z) = \nabla^\perp_X(\mathrm{II}(Y, Z)) - \mathrm{II}(\nabla_XY, Z) - \mathrm{II}(Y, \nabla_XZ)$.
\end{enumerate}
In particular, for orthonormal $u, v \in T_pM$ the sectional curvatures of the plane they span satisfy
$K(u, v) = \overline{K}(u, v) + \langle\mathrm{II}(u, u), \mathrm{II}(v, v)\rangle - |\mathrm{II}(u, v)|^2$.
\end{theorem}

\begin{proof}
By the Gauss formula, $\overline{\nabla}_X\overline{\nabla}_YZ = \overline{\nabla}_X(\nabla_YZ) + \overline{\nabla}_X(\mathrm{II}(Y, Z))$, and applying the Gauss formula again to the tangent field
$\nabla_YZ$ and the Weingarten equation to the normal field $\mathrm{II}(Y, Z)$,
\[
\overline{\nabla}_X\overline{\nabla}_YZ = \nabla_X\nabla_YZ + \mathrm{II}(X, \nabla_YZ) - S_{\mathrm{II}(Y, Z)}X + \nabla^\perp_X(\mathrm{II}(Y, Z)).
\]
Also $\overline{\nabla}_{[X,Y]}Z = \nabla_{[X,Y]}Z + \mathrm{II}([X, Y], Z)$. Subtract the same expressions with $X$ and $Y$ exchanged, and use $[X, Y] = \nabla_XY - \nabla_YX$.
The tangential part gives $(\overline{R}(X, Y)Z)^\top = R(X, Y)Z - S_{\mathrm{II}(Y, Z)}X + S_{\mathrm{II}(X, Z)}Y$; pairing with $W$ and using the Weingarten equation gives (1). The
normal part gives
\[
(\overline{R}(X, Y)Z)^\perp = \nabla^\perp_X(\mathrm{II}(Y, Z)) + \mathrm{II}(X, \nabla_YZ) - \nabla^\perp_Y(\mathrm{II}(X, Z)) - \mathrm{II}(Y, \nabla_XZ) - \mathrm{II}(\nabla_XY - \nabla_YX, Z),
\]
which is (2) after regrouping. For the last statement take $X = W = u$ and $Y = Z = v$ in (1), recalling $K(u, v) = R(u, v, v, u)$.
\end{proof}

\begin{definition}\label{riemannian-geometry:definition-hypersurface-curvatures}
Let $M$ be a hypersurface ($\dim\overline{M} = \dim M + 1$) with a unit normal field $\nu$, which exists locally, and globally if $M$ and $\overline{M}$ are orientable.
The \emph{scalar second fundamental form} is $h(X, Y) = \langle\mathrm{II}(X, Y), \nu\rangle$, so $\mathrm{II} = h\,\nu$, and the shape operator $S = S_\nu$ satisfies
$\langle SX, Y\rangle = h(X, Y)$. The eigenvalues $\kappa_1, \ldots, \kappa_n$ of the self-adjoint map $S$ on $T_pM$ are the \emph{principal curvatures}, its eigenvectors the
\emph{principal directions}, $H = \frac1n\operatorname{tr}S$ is the \emph{mean curvature}, and $\det S = \kappa_1 \cdots \kappa_n$ the \emph{Gauss--Kronecker
curvature}. Changing $\nu$ to $-\nu$ changes the signs of $h$, $S$, the $\kappa_i$ and $H$.
\end{definition}

\begin{corollary}[Theorema Egregium]\label{riemannian-geometry:corollary-theorema-egregium}
Let $M^n \subseteq \RR^{n+1}$ be a hypersurface with unit normal $\nu$. Then
$R(X, Y, Z, W) = h(Y, Z)h(X, W) - h(X, Z)h(Y, W)$, and for orthonormal principal directions $e_i$, $e_j$ the sectional curvature is $K(e_i, e_j) = \kappa_i\kappa_j$. In
particular, for a surface in $\RR^3$ the Gaussian curvature is $K = \kappa_1\kappa_2 = \det S$: a quantity defined through the embedding is determined by the metric
alone, and so is preserved by isometries between surfaces.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{riemannian-geometry:theorem-gauss-codazzi}(1) with $\overline{R} = 0$ and $\mathrm{II} = h\nu$, $|\nu| = 1$. For principal directions,
$h(e_i, e_i) = \kappa_i$ and $h(e_i, e_j) = 0$ for $i \neq j$. For a surface the only plane is the tangent plane, and $\kappa_1\kappa_2$ is the determinant of $S$.
\end{proof}

Codazzi's equation for a hypersurface of $\RR^{n+1}$ says that $(\nabla_Xh)(Y, Z)$ is symmetric in all three arguments. Gauss's and Codazzi's equations are the
integrability conditions for the embedding: by Bonnet's theorem, a metric and a symmetric form on a simply connected surface satisfying them come from a surface
in $\RR^3$, unique up to rigid motions \cite[Chapter 4]{do-carmo-curves-surfaces}.

\begin{proposition}[Surfaces in coordinates]\label{riemannian-geometry:proposition-surface-coordinates}
Let $x \colon \Omega \to \RR^3$ be a parametrisation of a surface $S$ (Remark~\ref{smooth-manifolds:remark-do-carmo}), with coordinates $(u^1, u^2) = (u, v)$,
$x_i = \partial x/\partial u^i$, and unit normal $\nu = x_1 \times x_2/|x_1 \times x_2|$. Let
\[
g_{ij} = \langle x_i, x_j\rangle = \begin{pmatrix} E & F\\ F & G\end{pmatrix}, \qquad h_{ij} = \langle\partial_i\partial_jx, \nu\rangle = -\langle x_i, \partial_j\nu\rangle.
\]
Then the first and second fundamental forms are $E\,du^2 + 2F\,du\,dv + G\,dv^2$ and $h_{11}\,du^2 + 2h_{12}\,du\,dv + h_{22}\,dv^2$, the matrix of $S$ in the basis
$(x_1, x_2)$ is $(g_{ij})^{-1}(h_{ij})$, and
\[
K = \frac{h_{11}h_{22} - h_{12}^2}{EG - F^2}, \qquad H = \frac{Eh_{22} - 2Fh_{12} + Gh_{11}}{2(EG - F^2)}.
\]
More generally, for a submanifold of $\RR^N$ parametrised by $x(u^1, \ldots, u^n)$, the decomposition
$\partial_i\partial_jx = \Gamma^k_{ij}x_k + \mathrm{II}(x_i, x_j)$ of the second derivatives into tangent and normal parts gives the Christoffel symbols and the second
fundamental form, and $R_{ijkl} = R(x_i, x_j, x_k, x_l) = \langle\mathrm{II}_{jk}, \mathrm{II}_{il}\rangle - \langle\mathrm{II}_{ik}, \mathrm{II}_{jl}\rangle$ with
$\mathrm{II}_{ij} = \mathrm{II}(x_i, x_j)$.
\end{proposition}

\begin{proof}
In $\RR^N$, $\overline{\nabla}$ is the directional derivative, and the coordinate field $\partial/\partial u^j$ is $x_j$ along $S$, so
$\overline{\nabla}_{x_i}x_j = \partial_i\partial_jx$. The Gauss formula splits it as $\nabla_{x_i}x_j + \mathrm{II}(x_i, x_j)$, which gives the general statement, the tangent part
being $\Gamma^k_{ij}x_k$, and for surfaces $h_{ij} = \langle\partial_i\partial_jx, \nu\rangle$; differentiating $\langle x_i, \nu\rangle = 0$ gives the second expression. From
$\langle Sx_i, x_j\rangle = h_{ij}$, the matrix $A$ of $S$ satisfies $(g_{ij})A = (h_{ij})$. So $K = \det S = \det(h_{ij})/\det(g_{ij})$ and
$2H = \operatorname{tr}((g_{ij})^{-1}(h_{ij}))$, which are the formulas stated, using $(g_{ij})^{-1} = \frac{1}{EG - F^2}\bigl(\begin{smallmatrix} G & -F\\ -F & E\end{smallmatrix}\bigr)$. The
formula for $R_{ijkl}$ is Theorem~\ref{riemannian-geometry:theorem-gauss-codazzi}(1) with $\overline{R} = 0$.
\end{proof}

For a hypersurface of $\RR^{n+1}$ the unit normal is a map $\nu \colon M \to S^n$, the \emph{Gauss map}, and since $T_{\nu(p)}S^n = \nu(p)^\perp = T_pM$ its
differential is an endomorphism of $T_pM$. The shape operator is \emph{minus} this differential, and the next proposition writes the relation between the matrices
of $d\nu$, of the first fundamental form and of the second in coordinates.

\begin{proposition}[Weingarten equations in coordinates]\label{riemannian-geometry:proposition-weingarten-coordinates}
Let $M \subseteq \RR^{n+1}$ be a hypersurface with unit normal $\nu$ and shape operator $S = S_\nu$, parametrised by $x(u^1, \ldots, u^n)$, with $x_i = \partial x/\partial u^i$,
$g_{ij} = \langle x_i, x_j\rangle$, $h_{ij} = h(x_i, x_j) = \langle\partial_i\partial_jx, \nu\rangle$, and $(g^{ij})$ the inverse matrix of $(g_{ij})$.
\begin{enumerate}
\item $d\nu = -S$: the derivative $\partial_i\nu = \partial(\nu \circ x)/\partial u^i$ is tangent to $M$ and equals $-S(x_i)$.
\item Write $\partial_i\nu = \sum_ka_{ki}\,x_k$, so that $A = (a_{ki})$ is the matrix of $d\nu$ in the basis $(x_1, \ldots, x_n)$. Then
\[
h_{ij} = -\sum_{k=1}^na_{ki}\,g_{kj}, \qquad\text{that is,}\qquad (h_{ij}) = -A^T(g_{ij}), \qquad A = -(g_{ij})^{-1}(h_{ij}), \qquad a_{ki} = -\sum_lg^{kl}h_{li},
\]
and the matrix of $S$ in the same basis is $-A = (g_{ij})^{-1}(h_{ij})$.
\item The Gauss--Kronecker curvature is $\det S = \det(h_{ij})/\det(g_{ij})$, and $\det(d\nu) = (-1)^n\det(h_{ij})/\det(g_{ij})$.
\item For a surface in $\RR^3$, with $E, F, G$ as in Proposition~\ref{riemannian-geometry:proposition-surface-coordinates} and $\Delta = EG - F^2$,
\[
a_{11} = \frac{Fh_{12} - Gh_{11}}{\Delta}, \quad a_{12} = \frac{Fh_{22} - Gh_{12}}{\Delta}, \quad a_{21} = \frac{Fh_{11} - Eh_{12}}{\Delta}, \quad
a_{22} = \frac{Fh_{12} - Eh_{22}}{\Delta},
\]
and $K = \det(d\nu) = \det S = (h_{11}h_{22} - h_{12}^2)/\Delta$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Differentiating $\langle\nu, \nu\rangle = 1$ gives $\langle\partial_i\nu, \nu\rangle = 0$, so $\partial_i\nu$ is tangent. In $\RR^{n+1}$, $\overline{\nabla}_{x_i}\nu$ is the
derivative $\partial_i\nu$ of the components of $\nu$, and by Definition~\ref{riemannian-geometry:definition-shape-operator},
$S(x_i) = -(\overline{\nabla}_{x_i}\nu)^\top = -\partial_i\nu$.

(2) Differentiating $\langle x_j, \nu\rangle = 0$ in $u^i$ gives $h_{ij} = \langle\partial_i\partial_jx, \nu\rangle = -\langle x_j, \partial_i\nu\rangle$, and inserting
$\partial_i\nu = \sum_ka_{ki}x_k$ gives $h_{ij} = -\sum_ka_{ki}g_{kj} = -(A^T(g_{ij}))_{ij}$. Transposing, and using that $(g_{ij})$ and $(h_{ij})$ are symmetric,
$(h_{ij}) = -(g_{ij})A$, so $A = -(g_{ij})^{-1}(h_{ij})$. The matrix of $S = -d\nu$ is $-A$.

(3) $\det S = \det\bigl((g_{ij})^{-1}(h_{ij})\bigr) = \det(h_{ij})/\det(g_{ij})$, and $\det(d\nu) = \det(-S) = (-1)^n\det S$.

(4) Here $(g_{ij})^{-1} = \Delta^{-1}\bigl(\begin{smallmatrix} G & -F\\ -F & E\end{smallmatrix}\bigr)$, and multiplying out $-(g_{ij})^{-1}(h_{ij})$ gives the four entries.
For $n = 2$ the sign $(-1)^n$ is $+1$, and $K = \det S$ by Corollary~\ref{riemannian-geometry:corollary-theorema-egregium}.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-sign-conventions}
With $e = h_{11}$, $f = h_{12}$, $g = h_{22}$ (the letter $g$ here is not the metric), part (4) is the form in which the Weingarten equations appear in
\cite[Chapter 3]{do-carmo-curves-surfaces}, where one works with $d\nu$ itself and $\mathrm{II}_p(v) = -\langle d\nu_p(v), v\rangle$. The two conventions differ by the
sign $S = -d\nu$, which is the most common source of sign errors: if $a_{ki}$ denotes the matrix of $S$ instead of $d\nu$, the formula of (2) reads
$h_{ij} = +\sum_ka_{ki}g_{kj}$. For surfaces the sign does not affect $K$, because $\det(-S) = \det S$ for $2 \times 2$ matrices, but it changes the sign of the mean
curvature, and in odd dimension $n$ it changes the sign of the Gauss--Kronecker curvature. For example, for the unit sphere with the outward normal $\nu(p) = p$, the Gauss map
is the identity, $d\nu = \id$, $S = -\id$, and $(h_{ij}) = -(g_{ij})$, as in Example~\ref{riemannian-geometry:example-surfaces}(1) below.
\end{remark}

\begin{example}\label{riemannian-geometry:example-surfaces}
\begin{enumerate}
\item \emph{Spheres.} For the sphere of radius $r$ in $\RR^{n+1}$ with outward normal $\nu(p) = p/r$, $\overline{\nabla}_X\nu = X/r$, so $S = -\frac1r\id$, $h = -\frac1rg$, every
direction is principal with $\kappa = -1/r$, and all sectional curvatures are $\kappa_i\kappa_j = 1/r^2$, as in Example~\ref{riemannian-geometry:example-curvature}.
\item \emph{Cylinders.} For $\{x^2 + y^2 = 1\} \subseteq \RR^3$ with $\nu = (x, y, 0)$, $\overline{\nabla}_X\nu$ is the horizontal part of $X$, so the principal curvatures are
$-1$ (horizontal circles) and $0$ (vertical lines), $K = 0$ and $H = -\frac12$. The cylinder is locally isometric to the plane (unroll it), in accordance with the Theorema
Egregium, although it bends in $\RR^3$.
\item \emph{Graphs.} Let $S$ be the graph of $z = \phi(u, v)$ with $\phi(0) = 0$ and $d\phi(0) = 0$, parametrised by $x(u, v) = (u, v, \phi(u, v))$. At the origin
$x_1 = e_1$, $x_2 = e_2$, $\nu = e_3$, so $g_{ij} = \delta_{ij}$ and $h_{ij} = \partial_i\partial_j\phi(0)$: the second fundamental form is the Hessian of $\phi$, the principal
curvatures are its eigenvalues, and $K(0) = \det\operatorname{Hess}\phi(0)$. The paraboloid $z = u^2 + v^2$ has $K(0) = 4$, the saddle $z = u^2 - v^2$ has $K(0) = -4$: positive
curvature means that the surface lies locally on one side of its tangent plane, negative curvature that it crosses it.
\end{enumerate}
\end{example}

\section{Global theorems on surfaces}\label{riemannian-geometry:section-surfaces-global}

This section collects the classical global theorems on surfaces, in the form in which they appear in \cite[Chapters 4 and 5]{do-carmo-curves-surfaces}:
the Gauss--Bonnet theorem for regions with boundary, Minding's theorem on surfaces of constant curvature, and the theorems of Hadamard, Liebmann and Hilbert on
surfaces in $\RR^3$. Throughout, $M$ is an oriented Riemannian surface and $J$ is the rotation by $+\pi/2$ in its tangent planes, as in the proof of
Theorem~\ref{riemannian-geometry:theorem-gauss-bonnet}; for a surface in $\RR^3$ oriented by a unit normal $\nu$, $Jv = \nu \times v$
(Proposition~\ref{differential-forms:proposition-orientation-normal}).

\begin{definition}\label{riemannian-geometry:definition-geodesic-curvature}
Let $\alpha$ be a curve in $M$ parametrised by arc length, $|\alpha'| = 1$. Since $\langle D_s\alpha', \alpha'\rangle = \frac12\frac{d}{ds}|\alpha'|^2 = 0$, the covariant acceleration is
$D_s\alpha' = k_g\,J\alpha'$ for a function $k_g$, the \emph{geodesic curvature} of $\alpha$. So $\alpha$ is a geodesic if and only if $k_g = 0$. For a surface in $\RR^3$,
$D_s\alpha'$ is the tangential part of the acceleration $\alpha''$ (Theorem~\ref{riemannian-geometry:theorem-gauss-formula}), and $k_g = \langle\alpha'', \nu \times \alpha'\rangle$.
\end{definition}

For example, a circle of latitude at height $z_0 \in (-1, 1)$ on the unit sphere has geodesic curvature $\pm z_0/(1 - z_0^2)^{1/2}$, and the equator, $z_0 = 0$, is a geodesic.

A \emph{simple region} is a compact set $R \subseteq M$ contained in the domain of a positively oriented chart $x \colon \Omega \to M$, whose preimage is homeomorphic to a closed
disc and bounded by a simple closed curve that is piecewise smooth with nonvanishing velocity. Its boundary $\partial R$ is parametrised by arc length so that $R$ lies on the left
(that is, $J\alpha'$ points into $R$), and at a vertex where the smooth arcs meet, the \emph{exterior angle} $\theta_i \in (-\pi, \pi)$ is the oriented angle from the incoming to the
outgoing velocity; the \emph{interior angle} is $\pi - \theta_i$.

\begin{theorem}[Theorem of turning tangents]\label{riemannian-geometry:theorem-turning-tangents}
Let $R$ be a simple region, $(e_1, e_2)$ a positively oriented orthonormal frame on a neighbourhood of $R$, and write the unit tangent of $\partial R$ on each smooth arc as
$\alpha' = \cos\varphi\,e_1 + \sin\varphi\,e_2$ with $\varphi$ continuous on the arc. Then the sum over the arcs of the changes of $\varphi$, plus the sum of the exterior angles,
equals $2\pi$.
\end{theorem}

This is Hopf's Umlaufsatz \cite{hopf-1935}; see \cite[Chapters 4 and 5]{do-carmo-curves-surfaces}. For a smooth convex curve in the Euclidean plane it says that the tangent turns
exactly once.

\begin{theorem}[Local Gauss--Bonnet theorem]\label{riemannian-geometry:theorem-local-gauss-bonnet}
Let $R$ be a simple region with smooth boundary arcs $\alpha_i$ and exterior angles $\theta_1, \ldots, \theta_k$. Then
\[
\sum_i\int_{\alpha_i}k_g\,ds + \iint_RK\,dA + \sum_{j=1}^k\theta_j = 2\pi.
\]
\end{theorem}

\begin{proof}
Applying Gram--Schmidt to the coordinate fields of the chart gives a positively oriented orthonormal frame $(e_1, e_2)$ near $R$. Define the $1$-form $\omega$ by
$\nabla_Xe_1 = \omega(X)e_2$; then $\nabla_Xe_2 = -\omega(X)e_1$, and the computation in the proof of Theorem~\ref{riemannian-geometry:theorem-gauss-bonnet} gives
$d\omega = -K\,dA$. On a smooth arc, $\alpha' = \cos\varphi\,e_1 + \sin\varphi\,e_2$ with $\varphi$ smooth (lift the map to the circle through the covering $\RR \to S^1$,
Proposition~\ref{homotopy:proposition-homotopy-lifting}), and $J\alpha' = -\sin\varphi\,e_1 + \cos\varphi\,e_2$. Differentiating,
\[
D_s\alpha' = \varphi'J\alpha' + \cos\varphi\,\nabla_{\alpha'}e_1 + \sin\varphi\,\nabla_{\alpha'}e_2 = \varphi'J\alpha' + \omega(\alpha')J\alpha',
\]
so $k_g = \varphi' + \omega(\alpha')$. Summing over the arcs, $\sum_i\int_{\alpha_i}k_g\,ds = \sum_i(\text{change of }\varphi\text{ on }\alpha_i) + \int_{\partial R}\omega$. The first sum is
$2\pi - \sum_j\theta_j$ by Theorem~\ref{riemannian-geometry:theorem-turning-tangents}, and $\int_{\partial R}\omega = \iint_Rd\omega = -\iint_RK\,dA$ by Stokes' theorem in the chart, that
is, Green's theorem for a plane region bounded by a piecewise smooth simple closed curve \cite[Chapter 4]{do-carmo-curves-surfaces}
(Theorem~\ref{differential-forms:theorem-stokes} in the smooth case).
\end{proof}

\begin{corollary}\label{riemannian-geometry:corollary-geodesic-triangles}
Let $T$ be a \emph{geodesic triangle}: a simple region bounded by three geodesic arcs, with interior angles $\phi_1, \phi_2, \phi_3$. Then
\[
\phi_1 + \phi_2 + \phi_3 - \pi = \iint_TK\,dA.
\]
So the angles of a geodesic triangle add up to more than $\pi$ where $K > 0$, to $\pi$ where $K = 0$, and to less than $\pi$ where $K < 0$. On the unit sphere the area of a geodesic
triangle is its angle excess; for instance the octant bounded by the three coordinate planes has three right angles and area $\frac{3\pi}{2} - \pi = \frac{\pi}{2}$, one eighth of $4\pi$.
\end{corollary}

\begin{proof}
Here $k_g = 0$ on the sides and $\theta_j = \pi - \phi_j$, so Theorem~\ref{riemannian-geometry:theorem-local-gauss-bonnet} gives
$\iint_TK\,dA + 3\pi - (\phi_1 + \phi_2 + \phi_3) = 2\pi$.
\end{proof}

\begin{theorem}[Global Gauss--Bonnet theorem]\label{riemannian-geometry:theorem-global-gauss-bonnet}
Let $R \subseteq M$ be a compact region whose boundary consists of finitely many simple closed piecewise smooth curves $C_1, \ldots, C_m$, oriented with $R$ on the left, with
exterior angles $\theta_1, \ldots, \theta_k$ at their vertices. Then
\[
\iint_RK\,dA + \sum_i\int_{C_i}k_g\,ds + \sum_j\theta_j = 2\pi\chi(R).
\]
\end{theorem}

\begin{proof}[Sketch of proof]
Choose a triangulation of $R$ fine enough that every triangle is a simple region, with $F$ triangles, $E$ edges and $V$ vertices, so that $\chi(R) = V - E + F$. Apply
Theorem~\ref{riemannian-geometry:theorem-local-gauss-bonnet} to each triangle and add. Every interior edge is traversed twice in opposite directions, so the integrals of $k_g$ over
interior edges cancel, and those over boundary edges give the boundary term. The exterior angles of a triangle are $\pi$ minus its interior angles, and the interior
angles of all triangles add up to $2\pi$ at each of the $V_{\mathrm{int}}$ interior vertices, to $\pi - \theta_j$ at a corner of $\partial R$, and to $\pi$ at the other boundary
vertices. So the sum of all exterior angles of the triangles is $3\pi F - 2\pi V_{\mathrm{int}} - \pi V_\partial + \sum_j\theta_j$, where $V_\partial$ is the number of boundary
vertices, and adding the local formulas gives
$\iint_RK\,dA + \sum_i\int_{C_i}k_g\,ds + \sum_j\theta_j = -\pi F + 2\pi V_{\mathrm{int}} + \pi V_\partial$. Each triangle has three edges, each interior edge lies in two triangles
and each boundary edge in one, so $3F = 2E_{\mathrm{int}} + E_\partial$, and $E_\partial = V_\partial$ because the boundary is a union of closed curves; with these relations the right
side equals $2\pi(V - E + F)$. The omitted verification is the existence of such triangulations \cite[Chapter 4]{do-carmo-curves-surfaces}.
\end{proof}

For $R = M$ a closed surface there is no boundary, and the formula is Theorem~\ref{riemannian-geometry:theorem-gauss-bonnet}, $\iint_MK\,dA = 2\pi\chi(M)$. For a disc in the
plane bounded by a smooth convex curve, $K = 0$ and $\chi = 1$, and the formula says that the total geodesic curvature of the boundary is $2\pi$: the tangent turns once. For the
northern cap of the unit sphere above the latitude $z_0$, the area $2\pi(1 - z_0)$ plus the total geodesic curvature $2\pi(1 - z_0^2)^{1/2} \cdot z_0/(1 - z_0^2)^{1/2} = 2\pi z_0$ of
the boundary circle gives $2\pi = 2\pi\chi(\text{disc})$.

Many metrics on surfaces are given as a function times the Euclidean metric, and for them the Gaussian curvature has a simple formula.

\begin{proposition}[Conformal metrics on surfaces]\label{riemannian-geometry:proposition-conformal-curvature}
Let $\Omega \subseteq \RR^2$ be open, $u \colon \Omega \to \RR$ smooth, and $g = e^{2u}(dx^2 + dy^2)$. Then the Gaussian curvature of $g$ is
\[
K = -e^{-2u}\Delta u, \qquad \Delta = \partial_x^2 + \partial_y^2 .
\]
\end{proposition}

\begin{proof}
Here $g_{ij} = e^{2u}\delta_{ij}$, and the formula of Theorem~\ref{riemannian-geometry:theorem-levi-civita} gives
$\Gamma^k_{ij} = u_i\delta_{jk} + u_j\delta_{ik} - u_k\delta_{ij}$, where $u_1 = u_x$ and $u_2 = u_y$. So $\nabla_{\partial_x}\partial_x = u_x\partial_x - u_y\partial_y$ and
$\nabla_{\partial_y}\partial_x = u_y\partial_x + u_x\partial_y$. The fields $e_1 = e^{-u}\partial_x$, $e_2 = e^{-u}\partial_y$ form a positively oriented orthonormal frame, and
\[
\nabla_{\partial_x}e_1 = -u_xe^{-u}\partial_x + e^{-u}(u_x\partial_x - u_y\partial_y) = -u_y\,e_2, \qquad
\nabla_{\partial_y}e_1 = -u_ye^{-u}\partial_x + e^{-u}(u_y\partial_x + u_x\partial_y) = u_x\,e_2 .
\]
So the $1$-form $\omega$ with $\nabla_Xe_1 = \omega(X)e_2$ is $\omega = -u_y\,dx + u_x\,dy$, and $d\omega = \Delta u\,dx \wedge dy$. The computation in the proof of
Theorem~\ref{riemannian-geometry:theorem-gauss-bonnet} gives $d\omega = -K\,dA = -Ke^{2u}\,dx \wedge dy$, whence the formula.
\end{proof}

\begin{example}\label{riemannian-geometry:example-conformal-metrics}
\begin{enumerate}
\item \emph{The hyperbolic plane.} On $H = \{(x, y) : y > 0\}$ with $(dx^2 + dy^2)/y^2$, $u = -\log y$ and $\Delta u = 1/y^2$, so $K = -1$.
\item \emph{The Poincar\'e disc.} On the unit disc $\mathbb{D}$ with $4(dx^2 + dy^2)/(1 - x^2 - y^2)^2$, $u = \log2 - \log(1 - r^2)$ with $r^2 = x^2 + y^2$. For a radial function
$\Delta = \partial_r^2 + r^{-1}\partial_r$, and $\Delta\log(1 - r^2) = -4/(1 - r^2)^2$, so $K = -\frac{(1 - r^2)^2}4 \cdot \frac4{(1 - r^2)^2} = -1$. The \emph{Cayley map}
$w = i(1 + z)/(1 - z)$ is an isometry from the disc onto $H$: with $z = x + iy$ and $w$ written as a complex number, $|dw/dz| = 2/|1 - z|^2$ and
$\operatorname{Im}w = (1 - |z|^2)/|1 - z|^2$, so $|dw|/\operatorname{Im}w = 2|dz|/(1 - |z|^2)$.
\item \emph{The sphere.} In the chart of stereographic projection (Example~\ref{general-topology:example-stereographic}) the round metric of $S^2$ is
$4(dx^2 + dy^2)/(1 + x^2 + y^2)^2$, by a direct computation; here $\Delta\log(1 + r^2) = 4/(1 + r^2)^2$, and $K = 1$, as it should.
\end{enumerate}
\end{example}

\begin{theorem}[Minding; Cartan]\label{riemannian-geometry:theorem-minding}
Let $(M, g)$ and $(N, h)$ be Riemannian manifolds of the same dimension $n$ with the same constant sectional curvature $\kappa$, and let $p \in M$, $q \in N$ and a linear isometry
$I \colon T_pM \to T_qN$ be given. Then there are neighbourhoods $U \ni p$ and $V \ni q$ and an isometry $F \colon U \to V$ with $F(p) = q$ and $d_pF = I$. In particular two surfaces
with the same constant Gaussian curvature are locally isometric, and every surface of constant curvature $0$, $1$ or $-1$ is locally isometric to the plane, the unit sphere or the
hyperbolic plane.
\end{theorem}

\begin{proof}
Let $s_\kappa$ be the solution of $s'' + \kappa s = 0$ with $s(0) = 0$ and $s'(0) = 1$ (so $s_\kappa(t) = t$, $\sin(\sqrt\kappa t)/\sqrt\kappa$ or $\sinh(\sqrt{-\kappa}t)/\sqrt{-\kappa}$). We
first compute the differential of $\exp_p$. Let $v \in T_pM$ be in the domain of $\exp_p$, $v \neq 0$, $\ell = |v|$, $u = v/\ell$, and $\gamma(t) = \exp_p(tv)$; for $w \in T_pM$ write
$w = w_\parallel + w_\perp$ with $w_\parallel$ a multiple of $u$ and $w_\perp \perp u$, and let $P_t$ be parallel transport along $\gamma$. By
Proposition~\ref{riemannian-geometry:proposition-jacobi}(2), $J(t) = d_{tv}\exp_p(tw)$ is the Jacobi field along $\gamma$ with $J(0) = 0$ and $D_tJ(0) = w$. The fields
$J_1(t) = tP_t(w_\parallel)$ and $J_2(t) = \ell^{-1}s_\kappa(\ell t)P_t(w_\perp)$ are Jacobi fields: $P_t(u) = \dot\gamma(t)/\ell$, so $J_1$ is a multiple of $t\dot\gamma$, with $D_t^2J_1 = 0$ and
$R(J_1, \dot\gamma)\dot\gamma = 0$; and $P_t(w_\perp) \perp \dot\gamma$, so by the curvature formula of Example~\ref{riemannian-geometry:example-curvature},
$R(J_2, \dot\gamma)\dot\gamma = \kappa|\dot\gamma|^2J_2 = \kappa\ell^2J_2$, while $D_t^2J_2 = \ell s_\kappa''(\ell t)P_t(w_\perp) = -\kappa\ell^2J_2$. They have the initial values $J_1(0) = J_2(0) = 0$,
$D_tJ_1(0) = w_\parallel$ and $D_tJ_2(0) = w_\perp$, so $J = J_1 + J_2$ by uniqueness, and since $P_1$ is an isometry and $P_1(w_\parallel) \perp P_1(w_\perp)$,
\[
|d_v\exp_p(w)|^2 = |J(1)|^2 = |w_\parallel|^2 + \Bigl(\frac{s_\kappa(\ell)}{\ell}\Bigr)^2|w_\perp|^2.
\]
The same formula holds on $N$. Now choose $\varepsilon > 0$ such that $\exp_p$ and $\exp_q$ map the balls of radius $\varepsilon$ diffeomorphically onto open sets $U$ and $V$
(Proposition~\ref{riemannian-geometry:proposition-exponential}), and let $F = \exp_q \circ I \circ (\exp_p|_{B_\varepsilon})^{-1} \colon U \to V$, a diffeomorphism with $F(p) = q$ and $d_pF = I$.
For $x = \exp_p(v)$ and $w \in T_pM$, the chain rule gives $d_xF(d_v\exp_p(w)) = d_{Iv}\exp_q(Iw)$, and since $I$ preserves lengths and orthogonality, the formula above gives the same
length for $d_v\exp_p(w)$ and $d_{Iv}\exp_q(Iw)$ (for $v = 0$ both differentials are the identity). As the $d_v\exp_p(w)$ fill $T_xM$, $d_xF$ preserves lengths, hence inner products, so $F$
is an isometry.
\end{proof}

For instance the cylinder and the cone (without its vertex) are locally isometric to the plane, and the pseudosphere, the surface of revolution of the tractrix, to the hyperbolic
plane. Minding's theorem is local: the flat torus $\RR^2/\ZZ^2$ is locally but not globally isometric to the plane.

\begin{lemma}\label{riemannian-geometry:lemma-elliptic-point}
A compact surface $S \subseteq \RR^3$ has a point where $K > 0$. More precisely, if $p_0 \in S$ maximises the distance $R$ to the origin, then $K(p_0) \geq 1/R^2$.
\end{lemma}

\begin{proof}
The function $|x|^2$ on $S$ has a maximum $R^2 > 0$ at some $p_0$, since $S$ is compact (and not a point). Its differential $2\langle p_0, -\rangle$ vanishes on $T_{p_0}S$, so
$\nu = p_0/R$ is a unit normal at $p_0$. For a unit-speed curve $\alpha$ in $S$ with $\alpha(0) = p_0$, the function $|\alpha|^2$ has a maximum at $0$, so
$0 \geq (|\alpha|^2)''(0) = 2\langle\alpha''(0), p_0\rangle + 2|\alpha'(0)|^2$, that is, $\langle\alpha''(0), \nu\rangle \leq -1/R$. The normal part of $\alpha''(0)$ is
$\mathrm{II}(\alpha'(0), \alpha'(0))$ (Theorem~\ref{riemannian-geometry:theorem-gauss-formula}, applied to the covariant acceleration of $\alpha$ in $\RR^3$), so
$h(w, w) \leq -1/R$ for every unit tangent vector $w$ at $p_0$. Both principal curvatures are therefore $\leq -1/R$, and $K(p_0) = \kappa_1\kappa_2 \geq 1/R^2$
(Corollary~\ref{riemannian-geometry:corollary-theorema-egregium}).
\end{proof}

So there is no compact surface in $\RR^3$ with $K \leq 0$ everywhere; in particular the flat torus cannot be embedded isometrically in $\RR^3$ as a smooth surface.

\begin{lemma}\label{riemannian-geometry:lemma-proper-covering}
Let $f \colon X \to Y$ be a local homeomorphism from a compact Hausdorff space to a connected Hausdorff space. Then $f$ is surjective and a covering map with finite fibres.
\end{lemma}

\begin{proof}
A local homeomorphism is open, and a continuous map from a compact space to a Hausdorff space is closed (Proposition~\ref{general-topology:proposition-compact-basic}); so $f(X)$ is
open, closed and nonempty, hence equal to $Y$. A fibre $f^{-1}(y)$ is discrete, because $f$ is injective near each point, and closed in the compact $X$, hence finite, say
$\{x_1, \ldots, x_k\}$. Choose pairwise disjoint open sets $U_i \ni x_i$ mapped homeomorphically onto open sets $V_i \ni y$. The set $f(X \setminus \bigcup_iU_i)$ is compact, hence
closed, and does not contain $y$, so $V = \bigcap_iV_i \setminus f(X \setminus \bigcup_iU_i)$ is an open neighbourhood of $y$ with $f^{-1}(V) = \bigsqcup_i(U_i \cap f^{-1}(V))$, each piece
mapped homeomorphically onto $V$.
\end{proof}

\begin{theorem}[Hadamard]\label{riemannian-geometry:theorem-hadamard}
Let $S \subseteq \RR^3$ be a compact connected surface with Gaussian curvature $K > 0$ everywhere. Then $S$ is orientable, its Gauss map $\nu \colon S \to S^2$ (for the unit normal
making the second fundamental form negative definite) is a diffeomorphism, $S$ is diffeomorphic to $S^2$, and $\iint_SK\,dA = 4\pi$.
\end{theorem}

\begin{proof}
At each point $K = \det S_\nu > 0$, so the self-adjoint shape operator is definite, and exactly one of the two unit normals makes it negative definite; near each point this normal is
$\pm$ a smooth local normal with a locally constant sign, so it is a unit normal field, and $S$ is orientable (Proposition~\ref{differential-forms:proposition-orientation-normal}). For
tangent $X$, $\overline{\nabla}_X\nu$ is tangent, because $\langle\nu, \nu\rangle = 1$, so $d_p\nu(X) = \overline{\nabla}_X\nu = -S_\nu X$
(Proposition~\ref{riemannian-geometry:proposition-weingarten}); as $T_{\nu(p)}S^2 = \nu(p)^\perp = T_pS$ and $\det(-S_\nu) = K > 0$, $d_p\nu$ is an isomorphism, and $\nu$ is a local
diffeomorphism (Theorem~\ref{smooth-manifolds:theorem-inverse-function-manifolds}). By Lemma~\ref{riemannian-geometry:lemma-proper-covering} it is a covering map, and since $S^2$ is simply
connected (Corollary~\ref{homotopy:corollary-spheres}) and $S$ connected, it has one sheet (Corollary~\ref{homotopy:corollary-lifting-consequences}(3)): $\nu$ is a bijective local
diffeomorphism, hence a diffeomorphism. Orient $S$ by $\nu$ and $S^2$ by its outward normal; then $T_pS$ and $T_{\nu(p)}S^2$ are the same oriented plane, $\nu^*(dA_{S^2}) = \det(d\nu)\,dA_S = K\,dA_S$,
and $\iint_SK\,dA = \int_{S^2}dA = 4\pi$ (Proposition~\ref{differential-forms:proposition-integral-well-defined}).
\end{proof}

The value $4\pi$ agrees with the Gauss--Bonnet theorem, $2\pi\chi(S^2)$. Hadamard also proved that such a surface is the boundary of a convex body: it lies on one side of each
of its tangent planes \cite[Chapter 5]{do-carmo-curves-surfaces}.

\begin{theorem}[Liebmann]\label{riemannian-geometry:theorem-liebmann}
A compact connected surface in $\RR^3$ with constant Gaussian curvature $K$ is a round sphere of radius $1/\sqrt{K}$.
\end{theorem}

\begin{proof}[Sketch of proof]
By Lemma~\ref{riemannian-geometry:lemma-elliptic-point} the constant $K$ is positive. With the unit normal $\nu$ of Theorem~\ref{riemannian-geometry:theorem-hadamard} the
principal curvatures are negative; write them as $-\lambda_1 \leq -\lambda_2$, so that $\lambda_1 \geq \lambda_2 > 0$ and $\lambda_1\lambda_2 = K$. The continuous function $\lambda_1$
attains its maximum at some $p$, where $\lambda_2 = K/\lambda_1$ attains its minimum. \emph{Hilbert's lemma}, a computation with the Codazzi equation in principal coordinates, says
that at a non-umbilic point ($\lambda_1 > \lambda_2$) where $\lambda_1$ has a local maximum and $\lambda_2$ a local minimum, $K \leq 0$. So $p$ is umbilic and
$\lambda_1(p) = \sqrt K$. Hence $\lambda_1 \leq \sqrt K$ everywhere, while $\lambda_1^2 \geq \lambda_1\lambda_2 = K$; so $\lambda_1 = \lambda_2 = \sqrt K$ everywhere, every point is umbilic,
and $S_\nu = -\sqrt K\,\id$. Then $d\nu = -S_\nu = \sqrt K\,\id$, so $x - \nu/\sqrt K$ is constant on the connected surface $S$: every point lies at distance $1/\sqrt K$ from this
fixed point. The omitted verification is Hilbert's lemma \cite[Chapter 5]{do-carmo-curves-surfaces}.
\end{proof}

\begin{theorem}[Hilbert]\label{riemannian-geometry:theorem-hilbert}
There is no complete surface of constant negative Gaussian curvature isometrically immersed in $\RR^3$. In particular the hyperbolic plane cannot be realised as a complete
surface in $\RR^3$.
\end{theorem}

This is \cite{hilbert-1901}; see \cite[Chapter 5]{do-carmo-curves-surfaces} for a complete proof. In outline: on such a surface the asymptotic directions (where $h(w, w) = 0$) give
coordinates in which the metric has the form $du^2 + 2\cos\theta\,du\,dv + dv^2$, and the Gauss equation becomes the sine-Gordon equation $\theta_{uv} = \sin\theta$; these
coordinates cover the whole hyperbolic plane, whose area is infinite, while the Gauss--Bonnet theorem bounds the area of every coordinate quadrilateral by $2\pi$, a contradiction.
Incomplete surfaces of constant negative curvature do exist, such as the pseudosphere; and by Theorem~\ref{riemannian-geometry:theorem-minding} each is locally isometric to
the hyperbolic plane.

\section{Minimal surfaces}\label{riemannian-geometry:section-minimal}

A soap film spanning a wire takes the shape that minimises its area. Its surface is therefore a critical point of the area, and the first variation formula below
identifies these critical points: they are the hypersurfaces with mean curvature $H = 0$, the \emph{minimal} ones. We give the classical examples, show that a minimal
graph has the least area among all surfaces with the same boundary, compute the second variation, and prove Bernstein's theorem: a minimal graph over the whole plane is
a plane. Throughout, $M \subseteq \RR^{n+1}$ is a hypersurface with unit normal $\nu$, shape operator $S = -d\nu$ and mean curvature $H = \frac1n\operatorname{tr}S$
(Definition~\ref{riemannian-geometry:definition-hypersurface-curvatures} and Proposition~\ref{riemannian-geometry:proposition-weingarten-coordinates}), and we write
$|S|^2 = \operatorname{tr}(S^2) = \sum_i\kappa_i^2$.

\begin{proposition}[First and second variation of area]\label{riemannian-geometry:proposition-variation-area}
Let $x \colon \Omega \to \RR^{n+1}$ be a parametrisation of $M$, $\phi$ a smooth function with compact support $K \subseteq \Omega$, and $x_s = x + s\phi\nu$ (with
$\nu = \nu \circ x$). For small $|s|$ the matrix $g(s)_{ij} = \langle\partial_ix_s, \partial_jx_s\rangle$ is positive definite on $K$; let
$A(s) = \int_K\sqrt{\det g(s)}\,du$, the area of the piece $x_s(K)$. Then, with $dA = \sqrt{\det g}\,du$ and $|\nabla\phi|^2 = \sum_{i,j}g^{ij}\partial_i\phi\,\partial_j\phi$,
\[
A^{\prime}(0) = -n\int_KH\phi\,dA, \qquad\text{and, if } H = 0, \qquad A^{\prime\prime}(0) = \int_K\bigl(|\nabla\phi|^2 - |S|^2\phi^2\bigr)\,dA .
\]
\end{proposition}

\begin{proof}
Write $x_i = \partial_ix$ and $\nu_i = \partial_i\nu$. Then $\partial_ix_s = x_i + s(\partial_i\phi\,\nu + \phi\nu_i)$. Since $\langle x_i, \nu\rangle = 0$, $\langle\nu_i, \nu\rangle = 0$
and $|\nu| = 1$,
\[
g(s)_{ij} = g_{ij} + s\phi\bigl(\langle x_i, \nu_j\rangle + \langle\nu_i, x_j\rangle\bigr) + s^2\bigl(\partial_i\phi\,\partial_j\phi + \phi^2\langle\nu_i, \nu_j\rangle\bigr).
\]
By Proposition~\ref{riemannian-geometry:proposition-weingarten-coordinates}, $\langle x_i, \nu_j\rangle = -h_{ij}$, and $\nu_i = -\sum_kS^k_ix_k$ with $(S^k_i) = g^{-1}h$ the
matrix of $S$, so $\langle\nu_i, \nu_j\rangle = (S^Tg\,S)_{ij} = (hg^{-1}h)_{ij}$. Hence $g(s) = g(I + sY + s^2Z)$ with $Y = -2\phi\,g^{-1}h$ and
$Z = g^{-1}(\nabla\phi\,\nabla\phi^T) + \phi^2(g^{-1}h)^2$, where $\nabla\phi$ is the column $(\partial_i\phi)$; in particular $g(s)$ is positive definite on $K$ for small $|s|$. For a
matrix $X$, $\det(I + X) = 1 + \operatorname{tr}X + \frac12\bigl((\operatorname{tr}X)^2 - \operatorname{tr}(X^2)\bigr) + O(|X|^3)$ (expand the characteristic polynomial, or use
the eigenvalues), and $\sqrt{1 + \epsilon} = 1 + \frac\epsilon2 - \frac{\epsilon^2}8 + O(\epsilon^3)$. With $X = sY + s^2Z$ this gives
\[
\sqrt{\det g(s)} = \sqrt{\det g}\,\Bigl(1 + \frac s2\operatorname{tr}Y + s^2\Bigl(\frac12\operatorname{tr}Z + \frac18(\operatorname{tr}Y)^2 - \frac14\operatorname{tr}(Y^2)\Bigr) + O(s^3)\Bigr),
\]
uniformly on $K$. Now $\operatorname{tr}Y = -2\phi\operatorname{tr}S = -2nH\phi$, $\operatorname{tr}(Y^2) = 4\phi^2|S|^2$ and $\operatorname{tr}Z = |\nabla\phi|^2 + \phi^2|S|^2$. So the
coefficient of $s$ is $-nH\phi$, and if $H = 0$ the coefficient of $s^2$ is $\frac12(|\nabla\phi|^2 + \phi^2|S|^2) - \phi^2|S|^2 = \frac12(|\nabla\phi|^2 - |S|^2\phi^2)$. Integrating
over $K$ gives $A^{\prime}(0)$ and $A^{\prime\prime}(0)$.
\end{proof}

\begin{definition}\label{riemannian-geometry:definition-minimal}
A hypersurface $M \subseteq \RR^{n+1}$ is \emph{minimal} if $H = 0$ everywhere. A minimal hypersurface is \emph{stable} if $A^{\prime\prime}(0) \geq 0$ for all variations as in
Proposition~\ref{riemannian-geometry:proposition-variation-area}.
\end{definition}

By the proposition, $M$ is minimal if and only if $A^{\prime}(0) = 0$ for every such variation: if $H(u_0) \neq 0$, a function $\phi \geq 0$ supported where $H$ has the sign of
$H(u_0)$, with $\phi(u_0) > 0$, gives $A^{\prime}(0) \neq 0$. The word ``minimal'' refers to this critical point property; a minimal surface need not minimise area. For a
minimal surface in $\RR^3$ the principal curvatures are $\pm\kappa$, so $|S|^2 = 2\kappa^2 = -2K$, and the second variation is $\int_K(|\nabla\phi|^2 + 2K\phi^2)\,dA$.

\begin{example}\label{riemannian-geometry:example-minimal-surfaces}
We use Proposition~\ref{riemannian-geometry:proposition-surface-coordinates}.
\begin{enumerate}
\item \emph{Planes} are minimal, with $S = 0$.
\item \emph{The catenoid}, the surface of revolution of the catenary, $x(u, v) = (\cosh v\cos u, \cosh v\sin u, v)$. Here $E = G = \cosh^2v$, $F = 0$, the unit normal is
$\nu = (\cos u, \sin u, -\sinh v)/\cosh v$, and $h_{11} = -1$, $h_{12} = 0$, $h_{22} = 1$. So $H = (Eh_{22} + Gh_{11})/(2EG) = 0$, and $K = -1/\cosh^4v$.
\item \emph{The helicoid}, swept out by a line rotating about an axis while moving along it, $x(u, v) = (v\cos u, v\sin u, u)$. Here $E = 1 + v^2$, $F = 0$, $G = 1$,
$\nu = (-\sin u, \cos u, -v)/\sqrt{1 + v^2}$, and $h_{11} = h_{22} = 0$, $h_{12} = 1/\sqrt{1 + v^2}$. So $H = 0$ and $K = -1/(1 + v^2)^2$.
\end{enumerate}
\end{example}

\begin{proposition}[Minimal surface equation]\label{riemannian-geometry:proposition-minimal-surface-equation}
Let $u \colon \Omega \to \RR$ be smooth on an open $\Omega \subseteq \RR^2$, write $p = u_x$, $q = u_y$ and $W = \sqrt{1 + p^2 + q^2}$, and give the graph of $u$ the upward unit
normal $\nu = (-p, -q, 1)/W$. Then its mean curvature is
\[
2H = \frac{(1 + q^2)u_{xx} - 2pq\,u_{xy} + (1 + p^2)u_{yy}}{W^3} = \frac{\partial}{\partial x}\Bigl(\frac pW\Bigr) + \frac{\partial}{\partial y}\Bigl(\frac qW\Bigr).
\]
So the graph is minimal if and only if $u$ solves the \emph{minimal surface equation} $(1 + q^2)u_{xx} - 2pq\,u_{xy} + (1 + p^2)u_{yy} = 0$.
\end{proposition}

\begin{proof}
For $x(x, y) = (x, y, u(x, y))$: $E = 1 + p^2$, $F = pq$, $G = 1 + q^2$, $EG - F^2 = W^2$, and $h_{11} = u_{xx}/W$, $h_{12} = u_{xy}/W$, $h_{22} = u_{yy}/W$, since the second
derivatives of $x$ are $(0, 0, u_{xx})$, $(0, 0, u_{xy})$, $(0, 0, u_{yy})$. The formula for $H$ in
Proposition~\ref{riemannian-geometry:proposition-surface-coordinates} gives the first expression. For the second,
$\partial_x(p/W) = (u_{xx}W^2 - p(pu_{xx} + qu_{xy}))/W^3$ and $\partial_y(q/W) = (u_{yy}W^2 - q(pu_{xy} + qu_{yy}))/W^3$, whose sum is the first expression, as
$W^2 - p^2 = 1 + q^2$ and $W^2 - q^2 = 1 + p^2$.
\end{proof}

For example \emph{Scherk's surface}, the graph of $u = \log\cos y - \log\cos x$ over the square $|x|, |y| < \pi/2$, is minimal: $p = \tan x$, $q = -\tan y$, $u_{xy} = 0$, and
$(1 + q^2)u_{xx} + (1 + p^2)u_{yy} = \sec^2y\sec^2x - \sec^2x\sec^2y = 0$. It tends to $\pm\infty$ on the sides of the square, and cannot be extended to a larger domain.

\begin{proposition}[Minimal graphs minimise area]\label{riemannian-geometry:proposition-graphs-minimise}
Let $\Omega \subseteq \RR^2$ be open and convex, $u \colon \Omega \to \RR$ a solution of the minimal surface equation, and $G \subseteq \Omega \times \RR$ a compact piece of its
graph that is a surface with boundary, oriented by the upward normal. Let $\Sigma \subseteq \Omega \times \RR$ be a compact oriented surface with boundary
(Definition~\ref{differential-forms:definition-manifold-with-boundary}) with $\partial\Sigma = \partial G$ as oriented curves. Then $\mathrm{Area}(\Sigma) \geq \mathrm{Area}(G)$.
\end{proposition}

\begin{proof}
Extend the unit normal of the graph to the unit vector field $N(x, y, z) = (-p, -q, 1)/W$ on $\Omega \times \RR$, independent of $z$, and let
$\omega = \iota_N(dx \wedge dy \wedge dz)$. As in the proof of Corollary~\ref{riemannian-geometry:corollary-divergence},
$d\omega = (\operatorname{div}N)\,dx \wedge dy \wedge dz$, and $\operatorname{div}N = -\partial_x(p/W) - \partial_y(q/W) = -2H = 0$ by
Proposition~\ref{riemannian-geometry:proposition-minimal-surface-equation}. The open set $\Omega \times \RR$ is convex, so $\omega = d\eta$ for a $1$-form $\eta$ by the Poincar\'e lemma
(Theorem~\ref{de-rham:theorem-homotopy-invariance}). By Stokes' theorem (Theorem~\ref{differential-forms:theorem-stokes}),
$\int_\Sigma\omega = \int_{\partial\Sigma}\eta = \int_{\partial G}\eta = \int_G\omega$. By Proposition~\ref{riemannian-geometry:proposition-flux}, on a surface with unit normal $n$
defining its orientation, $\omega$ restricts to $\langle N, n\rangle\,dA$. On $G$, $n = N$, so $\int_G\omega = \mathrm{Area}(G)$; on $\Sigma$, $\langle N, n\rangle \leq 1$, so
$\int_\Sigma\omega \leq \mathrm{Area}(\Sigma)$.
\end{proof}

The form $\omega$ is a \emph{calibration}: a closed form whose restriction to every surface is at most the area form, with equality on the graph. In particular minimal graphs
are stable, since the surfaces $x_s(K)$ of Proposition~\ref{riemannian-geometry:proposition-variation-area} stay in $\Omega \times \RR$ and have the same boundary as the graph for
small $|s|$. The comparison only involves surfaces inside the cylinder $\Omega \times \RR$ over a convex domain, where the calibration is defined; a minimal surface
that is not a graph, such as the catenoid, need not have this property.

The minimal surface equation is a nonlinear elliptic equation, and its entire solutions are much more rigid than harmonic functions: $x^2 - y^2$ is harmonic on all of
$\RR^2$ without being affine, but the only minimal graphs over the whole plane are planes. The proof below is Nitsche's \cite{nitsche-1957}; it reduces the statement to
Liouville's theorem through a theorem of J\"orgens on the Monge--Amp\`ere equation.

\begin{lemma}[J\"orgens]\label{riemannian-geometry:lemma-jorgens}
Let $\varphi \colon \RR^2 \to \RR$ be smooth with $\varphi_{xx} > 0$ and $\varphi_{xx}\varphi_{yy} - \varphi_{xy}^2 = 1$ everywhere. Then the Hessian of $\varphi$ is constant.
\end{lemma}

\begin{proof}
Write $a = \varphi_{xx}$, $b = \varphi_{xy}$, $c = \varphi_{yy}$, so $a > 0$ and $ac - b^2 = 1$; then $c > 0$, the Hessian $\mathcal{H}$ is positive definite, and
$a + c \geq 2\sqrt{ac} \geq 2$.

\emph{Step 1: the Lewy map.} Let $L \colon \RR^2 \to \RR^2$, $L(z) = z + \nabla\varphi(z)$. For $z_1, z_2 \in \RR^2$,
$\langle\nabla\varphi(z_2) - \nabla\varphi(z_1), z_2 - z_1\rangle = \int_0^1\langle\mathcal{H}(z_1 + t(z_2 - z_1))(z_2 - z_1), z_2 - z_1\rangle\,dt \geq 0$, so
\[
|L(z_2) - L(z_1)|^2 = |z_2 - z_1|^2 + 2\langle\nabla\varphi(z_2) - \nabla\varphi(z_1), z_2 - z_1\rangle + |\nabla\varphi(z_2) - \nabla\varphi(z_1)|^2 \geq |z_2 - z_1|^2 .
\]
So $L$ is injective, and its image is closed: if $L(z_k) \to w$, then $(z_k)$ is a Cauchy sequence, $z_k \to z$ and $L(z) = w$. Its differential $I + \mathcal{H}$ is positive
definite, hence invertible, so $L$ is a local diffeomorphism (Theorem~\ref{real-analysis:theorem-inverse-function}) and its image is open. As $\RR^2$ is connected, $L$ is a
diffeomorphism of $\RR^2$ onto itself.

\emph{Step 2: a holomorphic function.} Write $(\xi, \eta) = L(x, y)$ and $\zeta = \xi + i\eta$, and define $F(\zeta) = w + iv$ with $w = x - \varphi_x$ and $v = \varphi_y - y$,
where $(x, y) = L^{-1}(\xi, \eta)$; $F$ is smooth. From $d(\xi, \eta) = (I + \mathcal{H})\,d(x, y)$ and $d(w, v) = N\,d(x, y)$ with
$N = \bigl(\begin{smallmatrix} 1 - a & -b\\ b & c - 1\end{smallmatrix}\bigr)$, the Jacobian matrix of $(w, v)$ with respect to $(\xi, \eta)$ is
\[
N(I + \mathcal{H})^{-1} = \frac1{2 + a + c}\begin{pmatrix} 1 - a & -b\\ b & c - 1\end{pmatrix}\begin{pmatrix} 1 + c & -b\\ -b & 1 + a\end{pmatrix}
= \frac1{2 + a + c}\begin{pmatrix} c - a & -2b\\ 2b & c - a\end{pmatrix},
\]
using $\det(I + \mathcal{H}) = 1 + a + c + ac - b^2 = 2 + a + c$ and $ac - b^2 = 1$ to simplify the entries. This matrix satisfies the Cauchy--Riemann equations, so $F$ is an
entire function (Proposition~\ref{complex-analysis:proposition-cauchy-riemann}), with
\[
F^{\prime} = \frac{(c - a) + 2ib}{2 + a + c}, \qquad |F^{\prime}|^2 = \frac{(a + c)^2 - 4(ac - b^2)}{(a + c + 2)^2} = \frac{a + c - 2}{a + c + 2} < 1 .
\]

\emph{Step 3: Liouville.} The derivative $F^{\prime}$ is entire (Theorem~\ref{complex-analysis:theorem-analytic}) and bounded, hence constant
(Corollary~\ref{complex-analysis:corollary-liouville}). So $\rho = |F^{\prime}|^2 < 1$ is constant, hence so is $a + c = 2(1 + \rho)/(1 - \rho)$, and then the real and imaginary parts
of $F^{\prime}$ show that $c - a$ and $b$ are constant. So $a$, $b$ and $c$ are constant.
\end{proof}

\begin{theorem}[Bernstein]\label{riemannian-geometry:theorem-bernstein}
Let $u \colon \RR^2 \to \RR$ be a smooth solution of the minimal surface equation on the whole plane. Then $u$ is affine, $u(x, y) = \alpha x + \beta y + \gamma$: an entire
minimal graph is a plane.
\end{theorem}

\begin{proof}
With the notation of Proposition~\ref{riemannian-geometry:proposition-minimal-surface-equation}, write $\mathcal{M} = (1 + q^2)u_{xx} - 2pq\,u_{xy} + (1 + p^2)u_{yy}$, and
consider the smooth functions
\[
A = \frac{1 + p^2}{W}, \qquad B = \frac{pq}{W}, \qquad C = \frac{1 + q^2}{W}.
\]
A direct computation, using $p_x = u_{xx}$, $p_y = q_x = u_{xy}$ and $q_y = u_{yy}$, gives
\[
\frac{\partial A}{\partial y} - \frac{\partial B}{\partial x} = -\frac{q\mathcal{M}}{W^3}, \qquad \frac{\partial C}{\partial x} - \frac{\partial B}{\partial y} = -\frac{p\mathcal{M}}{W^3}
\]
(for the first, both $W^3\partial_yA$ and $W^3\partial_xB$ are polynomials in $p, q, u_{xx}, u_{xy}, u_{yy}$, and their difference is $-q\mathcal{M}$; the second follows by
exchanging $x$ and $y$). As $\mathcal{M} = 0$, the $1$-forms $A\,dx + B\,dy$ and $B\,dx + C\,dy$ are closed on $\RR^2$, hence exact by the Poincar\'e lemma
(Theorem~\ref{de-rham:theorem-homotopy-invariance}): there are smooth $P$, $Q$ with $dP = A\,dx + B\,dy$ and $dQ = B\,dx + C\,dy$. Then $P\,dx + Q\,dy$ is closed, since
$P_y = B = Q_x$, so it is $d\varphi$ for a smooth $\varphi$. Its Hessian is
\[
\varphi_{xx} = A = \frac{1 + p^2}{W} > 0, \qquad \varphi_{xy} = B, \qquad \varphi_{yy} = C, \qquad \varphi_{xx}\varphi_{yy} - \varphi_{xy}^2 = \frac{(1 + p^2)(1 + q^2) - p^2q^2}{W^2} = 1 .
\]
By Lemma~\ref{riemannian-geometry:lemma-jorgens}, $A$, $B$ and $C$ are constant. Then $A + C = (2 + p^2 + q^2)/W = W + 1/W$ is constant, and so is $W \geq 1$, since
$t \mapsto t + 1/t$ is injective on $[1, \infty)$. Hence $p^2 = AW - 1$ and $q^2 = CW - 1$ are constant, and so are $p$ and $q$, being continuous on the connected plane with
constant square. So $u$ has constant gradient and is affine.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-bernstein-problem}
The \emph{Bernstein problem} asks whether an entire minimal graph over $\RR^n$, a solution of $\operatorname{div}(\nabla u/\sqrt{1 + |\nabla u|^2}) = 0$ on $\RR^n$, must be
affine. The answer is yes for $n \leq 7$ (De Giorgi for $n = 3$, Almgren for $n = 4$, Simons \cite{simons-1968} for $n \leq 7$) and no for $n \geq 8$: Bombieri, De Giorgi and
Giusti \cite{bombieri-de-giorgi-giusti-1969} constructed a non-affine entire minimal graph over $\RR^8$, related to the area-minimising cone over
$S^3 \times S^3 \subseteq \RR^8$. The existence of minimal surfaces spanning a given curve, Plateau's problem, is a separate and deep theory.
\end{remark}

\section{The Hodge star and the Laplacian}\label{riemannian-geometry:section-hodge-star}

\begin{definition}\label{riemannian-geometry:definition-hodge-star}
Let $(M, g)$ be an oriented Riemannian $n$-manifold. The metric induces inner products on $\Lambda^kT^*_pM$ (Proposition~\ref{multilinear-algebra:proposition-induced-inner-product}), the volume form
$\mathrm{vol}_g$ (Proposition~\ref{riemannian-geometry:proposition-volume-form}), and the \emph{Hodge star} $* \colon \Omega^k(M) \to \Omega^{n-k}(M)$, characterised pointwise by $\alpha \wedge *\beta = \langle\alpha, \beta\rangle\,\mathrm{vol}_g$
(Theorem~\ref{multilinear-algebra:theorem-hodge-star}); it satisfies $** = (-1)^{k(n-k)}$ on $k$-forms. In coordinates $\mathrm{vol}_g = \sqrt{\det(g_{ij})}\,dx^1 \wedge \cdots \wedge dx^n$ in a positively
oriented chart.
\end{definition}

\begin{definition}\label{riemannian-geometry:definition-laplacian}
On a compact oriented Riemannian manifold define the $L^2$ inner product on $\Omega^k(M)$ by
\[
(\alpha, \beta) = \int_M\alpha \wedge *\beta = \int_M\langle\alpha, \beta\rangle\,\mathrm{vol}_g,
\]
the \emph{codifferential} $\delta = (-1)^{n(k+1)+1}*d* \colon \Omega^k(M) \to \Omega^{k-1}(M)$, and the \emph{Hodge Laplacian} $\Delta = d\delta + \delta d$. A form is \emph{harmonic} if $\Delta\alpha = 0$; the space of
harmonic $k$-forms is $\mathcal{H}^k(M)$.
\end{definition}

\begin{proposition}\label{riemannian-geometry:proposition-adjoint}
On a compact oriented Riemannian manifold:
\begin{enumerate}
\item $(d\alpha, \beta) = (\alpha, \delta\beta)$ for $\alpha \in \Omega^{k-1}$, $\beta \in \Omega^k$: the codifferential is the formal adjoint of $d$;
\item $\Delta$ is self-adjoint and non-negative: $(\Delta\alpha, \alpha) = \|d\alpha\|^2 + \|\delta\alpha\|^2$;
\item $\Delta\alpha = 0$ if and only if $d\alpha = 0$ and $\delta\alpha = 0$;
\item $*\Delta = \Delta*$, so $*$ maps harmonic $k$-forms isomorphically to harmonic $(n-k)$-forms;
\item $\Delta$ on functions is $-\sum_{i}\partial_i^2$ in normal coordinates at a point, and in general
$\Delta f = -\frac{1}{\sqrt{\det g}}\sum_{i,j}\partial_i(\sqrt{\det g}\,g^{ij}\partial_jf)$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) By Stokes' theorem (Theorem~\ref{differential-forms:theorem-stokes}), $0 = \int_Md(\alpha \wedge *\beta) = \int_Md\alpha \wedge *\beta + (-1)^{k-1}\int_M\alpha \wedge d{*}\beta$. Using $** = (-1)^{k(n-k)}$ one
rewrites $d*\beta = \pm *{*}d{*}\beta$, and the sign in the definition of $\delta$ is chosen exactly so that the identity becomes $(d\alpha, \beta) = (\alpha, \delta\beta)$.

(2) $(\Delta\alpha, \beta) = (d\delta\alpha, \beta) + (\delta d\alpha, \beta) = (\delta\alpha, \delta\beta) + (d\alpha, d\beta)$, which is symmetric; taking $\beta = \alpha$ gives the formula. (3) If $\Delta\alpha = 0$ then
$\|d\alpha\|^2 + \|\delta\alpha\|^2 = 0$, so both vanish; the converse is clear. (4) follows from the definition of $\delta$ and $** = \pm 1$: $*d\delta = \delta d*$ and $*\delta d = d\delta*$ up to the signs, which
cancel. (5) is a computation with the coordinate formula for $*$ and $d$, using $\mathrm{vol}_g = \sqrt{\det g}\,dx^1\wedge\cdots\wedge dx^n$; in normal coordinates at $p$ the first derivatives of $g$ vanish, so
$\Delta f(p) = -\sum_i\partial_i^2f(p)$.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-weitzenbock}
The Levi-Civita connection gives a second Laplacian, the \emph{connection Laplacian} $\nabla^*\nabla$, and the two differ by curvature: the \emph{Weitzenb\"ock formula} states
$\Delta = \nabla^*\nabla + \mathcal{R}$, where $\mathcal{R}$ is an endomorphism built from the curvature tensor; on $1$-forms $\mathcal{R} = \mathrm{Ric}$. Since $(\nabla^*\nabla\alpha, \alpha) = \|\nabla\alpha\|^2 \geq 0$,
this gives vanishing theorems: if $\mathrm{Ric} > 0$ on a compact oriented manifold, every harmonic $1$-form vanishes, hence $b_1(M) = 0$ by the Hodge theorem (Bochner's theorem,
Chapter~\ref{hodge-theorem}). For proofs see \cite[Chapter 6]{warner-foundations} and \cite[Chapter 4]{lee-riemannian}.
\end{remark}

\begin{example}\label{riemannian-geometry:example-harmonic-forms}
On a compact connected oriented manifold the harmonic $0$-forms are the constants, since $\Delta f = 0$ implies $df = 0$; by (4) the harmonic $n$-forms are the multiples of $\mathrm{vol}_g$. On the flat torus
$\RR^n/\ZZ^n$, the harmonic forms are exactly the forms with constant coefficients in the coordinates $dx^i$, a space of dimension $\binom{n}{k}$, which matches $b_k(T^n)$
(Exercise~\ref{de-rham:exercise-torus}).
\end{example}

\section{Isometries, Killing fields and space forms}\label{riemannian-geometry:section-killing}

The symmetries of a Riemannian manifold are its isometries, and the infinitesimal symmetries are the vector fields whose flows consist of isometries, the Killing
fields. This section shows that both are rigid: an isometry is determined by its value and its differential at one point, a Killing field by its value and its
covariant derivative at one point, and so the Killing fields form a Lie algebra of dimension at most $n(n + 1)/2$. The spaces with the most symmetry are those of
constant curvature, and the theorem of Killing and Hopf classifies them: a complete simply connected manifold of constant curvature is Euclidean space, a round
sphere or hyperbolic space, and every complete manifold of constant curvature is a quotient of one of these. At the end we state two rigidity theorems of a more
difficult kind, Cheng's maximal diameter theorem and the sphere theorem. Throughout, $M$ has dimension $n$.

A smooth map $F \colon (N, h) \to (M, g)$ between Riemannian manifolds of the same dimension is a \emph{local isometry} if $F^*g = h$, that is, if every
$d_xF$ is a linear isometry. By the inverse function theorem (Theorem~\ref{smooth-manifolds:theorem-inverse-function-manifolds}) it is then a local diffeomorphism,
and it restricts to isometries between small open sets.

\begin{lemma}\label{riemannian-geometry:lemma-local-isometries}
Let $F \colon N \to M$ be a local isometry.
\begin{enumerate}
\item $F$ maps geodesics to geodesics: if $v \in T_xN$ is in the domain of $\exp_x$, then $d_xF(v)$ is in the domain of $\exp_{F(x)}$ and
$F(\exp_x(v)) = \exp_{F(x)}(d_xF(v))$.
\item If $N$ is connected and $F^{\prime} \colon N \to M$ is another local isometry with $F^{\prime}(x_0) = F(x_0)$ and $d_{x_0}F^{\prime} = d_{x_0}F$ at one point $x_0$, then
$F^{\prime} = F$.
\end{enumerate}
\end{lemma}

\begin{proof}
(1) Let $x \in N$, and choose an open $U \ni x$ that $F$ maps diffeomorphically onto an open $V$, and a chart $y$ on $V$. Then $y \circ F$ is a chart on $U$, and
since $F^*g = h$, the coefficients of $h$ in the chart $y \circ F$ are the same functions as those of $g$ in the chart $y$. So the Christoffel symbols
(Theorem~\ref{riemannian-geometry:theorem-levi-civita}) and the geodesic equations (Definition~\ref{riemannian-geometry:definition-geodesic}) are the same, and a curve
$c$ in $U$ is a geodesic if and only if $F \circ c$ is one. Being a geodesic is a local condition, so $F \circ \gamma$ is a geodesic whenever $\gamma$ is. For $v$ in the
domain of $\exp_x$, the curve $F \circ \gamma_v$ is a geodesic defined on $[0, 1]$ with initial velocity $d_xF(v)$; by uniqueness it is the geodesic
$\gamma_{d_xF(v)}$, which is therefore defined on $[0, 1]$, and evaluating at $t = 1$ gives the formula.

(2) Let $A = \{x \in N : F^{\prime}(x) = F(x),\ d_xF^{\prime} = d_xF\}$, which contains $x_0$. It is closed, as $F$, $F^{\prime}$ and their differentials are continuous (in charts,
the Jacobian matrices are continuous). It is open: if $x \in A$ and $\exp_x$ maps a ball $B(0, \varepsilon)$ diffeomorphically onto a neighbourhood $W$ of $x$
(Proposition~\ref{riemannian-geometry:proposition-exponential}), then by (1), $F(\exp_x v) = \exp_{F(x)}(d_xFv) = F^{\prime}(\exp_x v)$ for $|v| < \varepsilon$, so
$F = F^{\prime}$ on $W$, and hence also $dF = dF^{\prime}$ on $W$. As $N$ is connected, $A = N$.
\end{proof}

For example, for a geodesic $\gamma$ and an isometry $F$, $F \circ \gamma$ is the geodesic with initial velocity $dF(\dot\gamma(0))$; this is the argument of
Exercise~\ref{riemannian-geometry:exercise-great-circles}, which finds the geodesics of the sphere as fixed sets of reflections.

\begin{definition}\label{riemannian-geometry:definition-killing}
A vector field $X$ on $M$ with local flow $\Phi$ (Theorem~\ref{smooth-manifolds:theorem-flow}) is a \emph{Killing field} if every $\Phi_t$ is an isometry from its
domain onto its image: $\langle d\Phi_t(v), d\Phi_t(w)\rangle = \langle v, w\rangle$ whenever $\Phi_t$ is defined at the base point of $v$ and $w$. For vector fields
$X$, $Y$, $Z$ we put
\[
(L_Xg)(Y, Z) = X\langle Y, Z\rangle - \langle[X, Y], Z\rangle - \langle Y, [X, Z]\rangle .
\]
\end{definition}

\begin{proposition}\label{riemannian-geometry:proposition-killing}
The expression $L_Xg$ is a symmetric tensor, and $(L_Xg)_p(v, w) = \frac{d}{dt}\big|_{t=0}\langle d_p\Phi_t(v), d_p\Phi_t(w)\rangle$ for $v, w \in T_pM$ (the \emph{Lie derivative}
of $g$). Moreover $(L_Xg)(Y, Z) = \langle\nabla_YX, Z\rangle + \langle Y, \nabla_ZX\rangle$. Consequently the following are equivalent:
\begin{enumerate}
\item $X$ is a Killing field;
\item $X\langle Y, Z\rangle = \langle[X, Y], Z\rangle + \langle Y, [X, Z]\rangle$ for all vector fields $Y$, $Z$;
\item for every $p$, the endomorphism $v \mapsto \nabla_vX$ of $T_pM$ is skew-adjoint: $\langle\nabla_vX, w\rangle + \langle v, \nabla_wX\rangle = 0$.
\end{enumerate}
In coordinates, (3) reads $\nabla_iX_j + \nabla_jX_i = 0$, where $\nabla_iX_j = \langle\nabla_{\partial_i}X, \partial_j\rangle$ (\emph{Killing's equation}).
\end{proposition}

\begin{proof}
Since $\nabla$ is metric and torsion-free, $X\langle Y, Z\rangle = \langle\nabla_XY, Z\rangle + \langle Y, \nabla_XZ\rangle$ and $[X, Y] = \nabla_XY - \nabla_YX$;
substituting gives $(L_Xg)(Y, Z) = \langle\nabla_YX, Z\rangle + \langle Y, \nabla_ZX\rangle$. The right side depends only on the values of $Y$ and $Z$ at each point and
is symmetric, so $L_Xg$ is a symmetric tensor.

Fix $p$ and vector fields $Y$, $Z$ near $p$. Let $V(t) = d\Phi_{-t}(Y_{\Phi_t(p)})$ and $W(t) = d\Phi_{-t}(Z_{\Phi_t(p)})$, curves in $T_pM$ with $V(0) = Y_p$,
$W(0) = Z_p$, and by Proposition~\ref{smooth-manifolds:proposition-bracket}, $V^{\prime}(0) = [X, Y]_p$ and $W^{\prime}(0) = [X, Z]_p$. For $a, b \in T_pM$ put
$G(t, a, b) = \langle d_p\Phi_t(a), d_p\Phi_t(b)\rangle$, smooth in $t$ and bilinear in $(a, b)$. Since $d\Phi_t(V(t)) = Y_{\Phi_t(p)}$,
\[
\langle Y, Z\rangle(\Phi_t(p)) = G(t, V(t), W(t)),
\]
and differentiating at $t = 0$ (the derivative of $t \mapsto \Phi_t(p)$ is $X_p$),
$X_p\langle Y, Z\rangle = \partial_tG(0, Y_p, Z_p) + \langle[X, Y]_p, Z_p\rangle + \langle Y_p, [X, Z]_p\rangle$. So $(L_Xg)_p(Y_p, Z_p) = \partial_tG(0, Y_p, Z_p)$, and
every pair of tangent vectors at $p$ is of the form $(Y_p, Z_p)$.

(1) $\Rightarrow$ (2): if $X$ is Killing, $G(t, a, b) = \langle a, b\rangle$ does not depend on $t$, so $L_Xg = 0$. (2) $\Rightarrow$ (1): let $v, w \in T_pM$ and
$f(t) = G(t, v, w)$, on the interval of $t$ for which $\Phi_t(p)$ is defined. For $t_0$ in it, $\Phi_{t_0 + s} = \Phi_s \circ \Phi_{t_0}$ near $p$, so
$f(t_0 + s) = \langle d\Phi_s(v^{\prime}), d\Phi_s(w^{\prime})\rangle$ with $v^{\prime} = d_p\Phi_{t_0}(v)$ and $w^{\prime} = d_p\Phi_{t_0}(w)$, and
$f^{\prime}(t_0) = (L_Xg)(v^{\prime}, w^{\prime}) = 0$. So $f$ is constant, equal to $f(0) = \langle v, w\rangle$. (2) $\Leftrightarrow$ (3) is the formula
$(L_Xg)(Y, Z) = \langle\nabla_YX, Z\rangle + \langle Y, \nabla_ZX\rangle$.
\end{proof}

\begin{example}\label{riemannian-geometry:example-killing-fields}
\begin{enumerate}
\item \emph{Euclidean space.} On $\RR^n$, $\nabla_vX = DX(x)v$, so $X$ is a Killing field if and only if the matrix $DX(x)$ is skew-symmetric for every $x$. This holds for
$X(x) = Ax + b$ with $A^T = -A$, whose flow $\Phi_t(x) = e^{tA}x + \int_0^te^{sA}b\,ds$ consists of rigid motions ($e^{tA}$ is orthogonal, since
$(e^{tA})^Te^{tA} = e^{-tA}e^{tA} = I$). Conversely, if $\partial_iX^j + \partial_jX^i = 0$ everywhere, then $T_{kij} = \partial_k\partial_iX^j$ is symmetric in
$k, i$ (Theorem~\ref{real-analysis:theorem-schwarz}) and antisymmetric in $i, j$, and a tensor with these two properties vanishes:
$T_{kij} = -T_{kji} = -T_{jki} = T_{jik} = T_{ijk} = -T_{ikj} = -T_{kij}$. So $DX$ is constant and $X(x) = Ax + b$.
\item \emph{Spheres.} For $A \in \mathfrak{so}(n + 1)$ the field $X(x) = Ax$ is tangent to $S^n$, as $\langle Ax, x\rangle = 0$, and its flow $e^{tA} \in \mathrm{SO}(n + 1)$
restricts to isometries of $S^n$; so $X$ is a Killing field on $S^n$. For $n = 2$ and $A = L_a$ (Example~\ref{lie-groups:example-so3}) it is the field
$x \mapsto a \times x$ of the rotations about the axis $a$.
\item \emph{Lie groups.} On a Lie group with a bi-invariant metric (Example~\ref{riemannian-geometry:example-metrics}) the left-invariant fields are Killing fields: the
flow of $X$ is $\Phi_t(g) = g\exp(tX)$ (Theorem~\ref{lie-groups:theorem-exponential}, translated by $L_g$), which is right multiplication by $\exp(tX)$, an isometry.
\end{enumerate}
\end{example}

\begin{proposition}\label{riemannian-geometry:proposition-killing-jacobi}
Let $X$ be a Killing field and $\gamma \colon I \to M$ a geodesic. Then $X \circ \gamma$ is a Jacobi field along $\gamma$, with $D_t(X \circ \gamma) = \nabla_{\dot\gamma}X$.
\end{proposition}

\begin{proof}
Let $[a, b] \subseteq I$. The domain of the flow is open and contains $\{0\} \times M$ (Theorem~\ref{smooth-manifolds:theorem-flow}), so by compactness of
$\gamma([a, b])$ there is $\varepsilon > 0$ such that $\Phi_s$ is defined on a neighbourhood of $\gamma([a, b])$ for $|s| < \varepsilon$. Each $\Phi_s$ is an isometry onto
its image, so by Lemma~\ref{riemannian-geometry:lemma-local-isometries}(1), $f(s, t) = \Phi_s(\gamma(t))$ is a family of geodesics $t \mapsto f(s, t)$. By
Proposition~\ref{riemannian-geometry:proposition-jacobi}(1), $\partial_sf(0, t) = X(\gamma(t))$ is a Jacobi field on $[a, b]$; as $[a, b]$ is arbitrary, it is one on $I$.
The formula for $D_t$ is Proposition~\ref{riemannian-geometry:proposition-covariant-derivative-curves}.
\end{proof}

\begin{theorem}\label{riemannian-geometry:theorem-killing-algebra}
Let $M$ be connected, and $p \in M$.
\begin{enumerate}
\item A Killing field $X$ with $X_p = 0$ and $\nabla_vX = 0$ for all $v \in T_pM$ vanishes identically.
\item The Killing fields form a Lie subalgebra $\mathfrak{iso}(M, g)$ of the Lie algebra $\mathfrak{X}(M)$ of vector fields
(Proposition~\ref{smooth-manifolds:proposition-bracket}), and the linear map $X \mapsto (X_p, \nabla X_p)$, where $\nabla X_p$ is the endomorphism $v \mapsto \nabla_vX$
of $T_pM$, is injective with values in $T_pM \oplus \mathfrak{so}(T_pM)$. Hence $\dim\mathfrak{iso}(M, g) \leq n + \frac{n(n-1)}{2} = \frac{n(n + 1)}{2}$.
\item The bound is attained by $\RR^n$ and by $S^n$, whose Killing fields are exactly those of Example~\ref{riemannian-geometry:example-killing-fields}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Let $Z$ be the set of points $q$ with $X_q = 0$ and $\nabla X_q = 0$; it is closed and contains $p$. If $q \in Z$ and $\exp_q$ maps $B(0, \varepsilon)$
diffeomorphically onto $W$, then for $|v| < \varepsilon$, $J(t) = X(\exp_q(tv))$ is a Jacobi field (Proposition~\ref{riemannian-geometry:proposition-killing-jacobi}) with
$J(0) = X_q = 0$ and $D_tJ(0) = \nabla_vX = 0$, hence $J = 0$ and $X(\exp_qv) = J(1) = 0$. So $X = 0$ on $W$, hence also $\nabla X = 0$ on $W$, and $Z$ is open. As $M$
is connected, $Z = M$.

(2) The map is linear, and it takes values in $T_pM \oplus \mathfrak{so}(T_pM)$ by Proposition~\ref{riemannian-geometry:proposition-killing}(3); by (1) it is
injective, which gives the dimension bound. Killing fields form a vector space, since $L_Xg$ is linear in $X$. For the bracket, let $X$ and $W$ satisfy
Proposition~\ref{riemannian-geometry:proposition-killing}(2). Then for all $Y$, $Z$,
\[
\begin{aligned}
XW\langle Y, Z\rangle &= X\langle[W, Y], Z\rangle + X\langle Y, [W, Z]\rangle \\
&= \langle[X, [W, Y]], Z\rangle + \langle[W, Y], [X, Z]\rangle + \langle[X, Y], [W, Z]\rangle + \langle Y, [X, [W, Z]]\rangle .
\end{aligned}
\]
Exchanging $X$ and $W$ and subtracting, the two middle terms cancel, and by the Jacobi identity $[X, [W, Y]] - [W, [X, Y]] = [[X, W], Y]$, so
$[X, W]\langle Y, Z\rangle = \langle[[X, W], Y], Z\rangle + \langle Y, [[X, W], Z]\rangle$: $[X, W]$ is a Killing field.

(3) On $\RR^n$ the fields $Ax + b$ with $A$ skew form a space of dimension $n(n - 1)/2 + n$, and by
Example~\ref{riemannian-geometry:example-killing-fields}(1) they are all the Killing fields. On $S^n$ the fields $x \mapsto Ax$, $A \in \mathfrak{so}(n + 1)$, form a space of
dimension $\dim\mathfrak{so}(n + 1) = n(n + 1)/2$: the map $A \mapsto (x \mapsto Ax)$ is injective because $S^n$ spans $\RR^{n+1}$. By (2) there are no other Killing
fields on $S^n$.
\end{proof}

\begin{remark}\label{riemannian-geometry:remark-myers-steenrod}
The isometries of $M$ form a group $\mathrm{Isom}(M, g)$. By the theorem of Myers and Steenrod it is a Lie group, compact if $M$ is compact, and its Lie algebra is
identified with the Lie algebra of complete Killing fields (on a compact manifold every Killing field is complete); so
$\dim\mathrm{Isom}(M, g) \leq n(n + 1)/2$ \cite[Chapter VI]{kobayashi-nomizu}. For instance $\mathrm{Isom}(\RR^n)$ is the group of rigid motions and
$\mathrm{Isom}(S^n) = \mathrm{O}(n + 1)$ (Corollary~\ref{riemannian-geometry:corollary-model-isometries} below), both of the maximal dimension. A manifold whose isometry
group has maximal dimension has constant curvature \cite[Chapter VI]{kobayashi-nomizu}.
\end{remark}

Curvature restricts the Killing fields of a compact manifold, by an integration argument of Bochner.

\begin{theorem}[Bochner]\label{riemannian-geometry:theorem-bochner-killing}
Let $M$ be compact and oriented, and $X$ a Killing field. Then, with $|\nabla X|^2 = \sum_i|\nabla_{e_i}X|^2$ for an orthonormal basis $(e_i)$,
\[
\int_M\bigl(|\nabla X|^2 - \mathrm{Ric}(X, X)\bigr)\,\mathrm{vol}_g = 0 .
\]
Hence if $\mathrm{Ric} \leq 0$ everywhere, every Killing field is parallel ($\nabla X = 0$); and if $\mathrm{Ric}$ is negative definite everywhere, every Killing field is zero,
and $\mathrm{Isom}(M, g)$ is finite.
\end{theorem}

\begin{proof}
Let $f = \frac12|X|^2$ and $p \in M$, and take normal coordinates at $p$ (Proposition~\ref{riemannian-geometry:proposition-exponential}), with
$\partial_i = e_i$ orthonormal at $p$. The $i$-th coordinate line $\gamma_i(t) = \exp_p(te_i)$ is a unit speed geodesic, so $\partial_i^2f(p) = (f \circ \gamma_i)^{\prime\prime}(0)$,
and $J = X \circ \gamma_i$ is a Jacobi field with $D_tJ(0) = \nabla_{e_i}X$ (Proposition~\ref{riemannian-geometry:proposition-killing-jacobi}). Hence
\[
(f \circ \gamma_i)^{\prime\prime} = \frac{d}{dt}\langle D_tJ, J\rangle = |D_tJ|^2 + \langle D_t^2J, J\rangle = |D_tJ|^2 - \langle R(J, \dot\gamma_i)\dot\gamma_i, J\rangle,
\]
which at $t = 0$ is $|\nabla_{e_i}X|^2 - R(X, e_i, e_i, X)$. Summing over $i$, and using $R(X, e_i, e_i, X) = R(e_i, X, X, e_i)$
(Proposition~\ref{riemannian-geometry:proposition-curvature-symmetries}) and Definition~\ref{riemannian-geometry:definition-curvature-tensor}, Proposition~\ref{riemannian-geometry:proposition-adjoint}(5)
gives
\[
\Delta f(p) = -\sum_i\partial_i^2f(p) = \mathrm{Ric}(X, X) - |\nabla X|^2 .
\]
By Proposition~\ref{riemannian-geometry:proposition-adjoint}(2), $\int_M\Delta f\,\mathrm{vol}_g = (\Delta f, 1) = (f, \Delta1) = 0$, which is the integral formula. If
$\mathrm{Ric} \leq 0$, the integrand $|\nabla X|^2 - \mathrm{Ric}(X, X)$ is a non-negative continuous function with integral $0$, so it vanishes: $\nabla X = 0$ and
$\mathrm{Ric}(X, X) = 0$; if $\mathrm{Ric}$ is negative definite, the latter forces $X = 0$. Then the compact Lie group $\mathrm{Isom}(M, g)$ has Lie algebra $0$
(Remark~\ref{riemannian-geometry:remark-myers-steenrod}), so it is discrete and compact, hence finite.
\end{proof}

For the flat torus $\RR^n/\ZZ^n$ the Killing fields are therefore the parallel fields, the constant fields $\sum_ic^i\partial_i$, whose flows are the translations; and a
compact surface of constant curvature $-1$ has only finitely many isometries.

We now turn to spaces of constant curvature. Recall that $s_\kappa(t)$ is $t$, $\sin(\sqrt\kappa t)/\sqrt\kappa$ or $\sinh(\sqrt{-\kappa}t)/\sqrt{-\kappa}$ for
$\kappa = 0$, $\kappa > 0$ or $\kappa < 0$, and that the proof of Theorem~\ref{riemannian-geometry:theorem-minding} shows: if $M$ has constant sectional curvature $\kappa$,
$v \neq 0$ lies in the domain of $\exp_p$ and $w = w_\parallel + w_\perp$ is the decomposition of $w \in T_pM$ into a multiple of $v$ and a vector orthogonal to $v$, then
\[
|d_v\exp_p(w)|^2 = |w_\parallel|^2 + \Bigl(\frac{s_\kappa(|v|)}{|v|}\Bigr)^2|w_\perp|^2. \tag{$\dagger$}
\]
Multiplying a metric by a constant $c > 0$ does not change the Levi-Civita connection (the Koszul formula is homogeneous) or $R(X, Y)Z$, and it multiplies the
sectional curvatures by $1/c$, since an orthonormal basis for $cg$ is $(u, v)/\sqrt c$ for an orthonormal basis $(u, v)$ for $g$.

\begin{definition}\label{riemannian-geometry:definition-model-spaces}
For $n \geq 2$ and $\kappa \in \RR$ the \emph{model space} $M^n_\kappa$ is: Euclidean space $\RR^n$ if $\kappa = 0$; the sphere $S^n(r) = \{x \in \RR^{n+1} : |x| = r\}$ of
radius $r = 1/\sqrt\kappa$ with the induced metric if $\kappa > 0$; and the \emph{hyperbolic space} $\{x \in \RR^n : x^n > 0\}$ with the metric
$|\kappa|^{-1}(dx_1^2 + \cdots + dx_n^2)/(x^n)^2$ if $\kappa < 0$.
\end{definition}

All three are complete, connected and simply connected with constant sectional curvature $\kappa$. For $\RR^n$ this is clear. The sphere $S^n(r)$ has curvature
$1/r^2 = \kappa$ (Example~\ref{riemannian-geometry:example-surfaces}(1)), is complete, being compact (Theorem~\ref{riemannian-geometry:theorem-hopf-rinow}), and is simply
connected for $n \geq 2$ (Corollary~\ref{homotopy:corollary-spheres}). Hyperbolic space is simply connected, being a convex subset of $\RR^n$ (for $n = 2$ its curvature is computed in
Example~\ref{riemannian-geometry:example-conformal-metrics}(1)); that it is complete with
curvature $-1$ for $\kappa = -1$ is computed in \cite[Chapter 8]{do-carmo-riemannian}, and the case of general $\kappa < 0$ follows by scaling.

\begin{theorem}[Killing--Hopf]\label{riemannian-geometry:theorem-killing-hopf}
Let $M$ and $N$ be complete, connected and simply connected Riemannian manifolds of the same dimension $n \geq 2$, with the same constant sectional curvature
$\kappa$. For $p \in M$, $q \in N$ and a linear isometry $I \colon T_pM \to T_qN$ there is a unique isometry $F \colon M \to N$ with $F(p) = q$ and $d_pF = I$.
In particular $M$ is isometric to the model space $M^n_\kappa$.
\end{theorem}

\begin{proof}
Uniqueness is Lemma~\ref{riemannian-geometry:lemma-local-isometries}(2). By Theorem~\ref{riemannian-geometry:theorem-hopf-rinow}, $\exp_p$ and $\exp_q$ are defined on
the whole tangent spaces. The key observation is the following. Let $U \subseteq T_pM$ be an open set on which $\exp_p$ is a local diffeomorphism, and
$F = \exp_q \circ I \circ \psi$ for a smooth local inverse $\psi$ of $\exp_p|_U$. For $x = \exp_p(v)$, $v \in U$, and $w \in T_pM$ the chain rule gives
$d_xF(d_v\exp_p(w)) = d_{Iv}\exp_q(Iw)$. The linear isometry $I$ preserves lengths and maps $w_\parallel$ and $w_\perp$ to the components of $Iw$ along and orthogonal to
$Iv$, so by $(\dagger)$ on $M$ and on $N$ the vectors $d_v\exp_p(w)$ and $d_{Iv}\exp_q(Iw)$ have the same length (for $v = 0$ both differentials are the identity). As
$d_v\exp_p$ is onto $T_xM$, $d_xF$ preserves lengths, hence inner products: $F$ is a local isometry.

\emph{The case $\kappa \leq 0$.} By Theorem~\ref{riemannian-geometry:theorem-cartan-hadamard}, $\exp_p$ and $\exp_q$ are diffeomorphisms, so
$F = \exp_q \circ I \circ \exp_p^{-1} \colon M \to N$ is a diffeomorphism, a local isometry by the observation, hence an isometry; and $F(p) = q$, $d_pF = I$ as
$d_0\exp = \id$.

\emph{The case $\kappa > 0$.} Multiplying both metrics by $\kappa$ we may assume $\kappa = 1$. Let $S = S^n$ be the unit sphere and $o \in S$. We show: for every linear
isometry $J \colon T_oS \to T_pM$ there is an isometry $G \colon S \to M$ with $G(o) = p$ and $d_oG = J$. Applying this to $M$ with $J$ and to $N$ with $I \circ J$ gives
isometries $G_M$ and $G_N$, and $F = G_N \circ G_M^{-1}$ has the required properties.

For a unit vector $u \perp o$, the great circle $c(t) = \cos t\,o + \sin t\,u$ has acceleration $c^{\prime\prime} = -c$, which is normal to $S$, so
$D_tc^{\prime} = (c^{\prime\prime})^\top = 0$ (Theorem~\ref{riemannian-geometry:theorem-gauss-formula}): $c$ is the geodesic with initial velocity $u$. Hence
$\exp_o(v) = \cos|v|\,o + \sin|v|\,v/|v|$ for $v \neq 0$. It maps $B = B(0, \pi)$ bijectively onto $S \setminus \{-o\}$:
every $y \neq -o$ is uniquely $\cos\theta\,o + \sin\theta\,u$ with $\theta = \arccos\langle y, o\rangle \in [0, \pi)$ and, if $\theta > 0$, a unit vector $u \perp o$.
By $(\dagger)$, with $s_1(\ell) = \sin\ell > 0$ for $0 < \ell < \pi$, $d_v\exp_o$ is injective for $v \in B$; so $\exp_o \colon B \to S \setminus \{-o\}$ is a bijective local
diffeomorphism, hence a diffeomorphism. By the observation (with the roles of $M$ and $N$ played by $S$ and $M$),
$G_1 = \exp_p \circ J \circ (\exp_o|_B)^{-1} \colon S \setminus \{-o\} \to M$ is a local isometry with $G_1(o) = p$ and $d_oG_1 = J$. Choose $o^{\prime} \in S$ with
$o^{\prime} \neq \pm o$, and let $G_2 = \exp_{p^{\prime}} \circ J^{\prime} \circ (\exp_{o^{\prime}}|_B)^{-1} \colon S \setminus \{-o^{\prime}\} \to M$ with
$p^{\prime} = G_1(o^{\prime})$ and $J^{\prime} = d_{o^{\prime}}G_1$, again a local isometry. The open set $W = S \setminus \{-o, -o^{\prime}\}$ contains $o^{\prime}$, and it
is connected: stereographic projection from $-o$ (Example~\ref{general-topology:example-stereographic}) maps it homeomorphically onto $\RR^n$ minus a point, which is
connected for $n \geq 2$ (Lemma~\ref{topological-manifolds:lemma-punctured-connected}). On $W$ the local isometries $G_1$ and $G_2$ agree at $o^{\prime}$ together with
their differentials, so $G_1 = G_2$ on $W$ by Lemma~\ref{riemannian-geometry:lemma-local-isometries}(2). Since $-o \neq -o^{\prime}$, they glue to a local isometry
$G \colon S \to M$. By Lemma~\ref{riemannian-geometry:lemma-proper-covering}, $G$ is a covering map ($S$ is compact and $M$ connected and Hausdorff). As $M$ is simply
connected and $S$ is path-connected, the monodromy action on a fibre is transitive and $\pi_1(M)$ is trivial, so each fibre is a single point
(Corollary~\ref{homotopy:corollary-lifting-consequences}(3)). So $G$ is a bijective local isometry, that is, an isometry, with $G(o) = p$ and $d_oG = J$.

Finally, the model space $M^n_\kappa$ satisfies the hypotheses, so $M$ is isometric to it.
\end{proof}

\begin{corollary}\label{riemannian-geometry:corollary-model-isometries}
\begin{enumerate}
\item The model spaces are \emph{frame-homogeneous}: for any two points $p, q \in M^n_\kappa$ and any linear isometry $I \colon T_pM^n_\kappa \to T_qM^n_\kappa$ there is a
unique isometry $F$ of $M^n_\kappa$ with $F(p) = q$ and $d_pF = I$.
\item The isometries of $\RR^n$ are the maps $x \mapsto Ax + b$ with $A \in \mathrm{O}(n)$ and $b \in \RR^n$.
\item The isometries of $S^n(r)$ are the restrictions of the maps $A \in \mathrm{O}(n + 1)$.
\end{enumerate}
\end{corollary}

\begin{proof}
(1) is Theorem~\ref{riemannian-geometry:theorem-killing-hopf} with $M = N = M^n_\kappa$. (2) These maps are isometries, since $d(Ax + b) = A$ preserves lengths. Given
an isometry $F$, $A = d_0F$ is an orthogonal matrix and $x \mapsto Ax + F(0)$ agrees with $F$ at $0$ together with its differential, so it equals $F$ by
Lemma~\ref{riemannian-geometry:lemma-local-isometries}(2). (3) Restrictions of orthogonal maps are isometries of $S^n(r)$. Given an isometry $F$ and $o \in S^n(r)$,
let $A$ be the linear map with $A(o) = F(o)$ and $A = d_oF$ on $T_oS^n(r) = o^\perp$; it maps the orthonormal basis formed by $o/r$ and an orthonormal basis of $o^\perp$
to an orthonormal basis, since $d_oF$ is an isometry onto $F(o)^\perp$, so $A \in \mathrm{O}(n + 1)$. Then $A|_{S^n(r)}$ and $F$ agree at $o$ together with their
differentials, so they are equal.
\end{proof}

\begin{corollary}[Space forms]\label{riemannian-geometry:corollary-space-forms}
Let $M$ be a complete connected Riemannian manifold of dimension $n \geq 2$ with constant sectional curvature $\kappa$. Then $M$ is isometric to $M^n_\kappa/\Gamma$ for
a group $\Gamma \cong \pi_1(M)$ of isometries of $M^n_\kappa$ acting freely, such that the projection $M^n_\kappa \to M^n_\kappa/\Gamma$ is a covering map, with the metric for
which this projection is a local isometry.
\end{corollary}

\begin{proof}
Let $\pi \colon \widetilde{M} \to M$ be a universal covering (Theorem~\ref{homotopy:theorem-universal-cover}; manifolds are locally path-connected and locally simply
connected). $\widetilde{M}$ is a connected manifold (Proposition~\ref{topological-manifolds:proposition-constructions}(4)) with a unique smooth structure making $\pi$ a local
diffeomorphism (Example~\ref{smooth-manifolds:example-smooth-quotients}), and $\tilde g = \pi^*g$ makes $\pi$ a local isometry, so $\widetilde{M}$ has constant curvature
$\kappa$. It is complete: let $\tilde\gamma$ be a maximal geodesic of $\widetilde{M}$ with $\tilde\gamma(0) = \tilde x$. Then $\pi \circ \tilde\gamma$ is a geodesic
(Lemma~\ref{riemannian-geometry:lemma-local-isometries}(1)), which extends to a geodesic $\gamma$ defined on $\RR$ by completeness of $M$. Let $\hat\gamma$ be the lift of
$\gamma$ starting at $\tilde x$ (Proposition~\ref{homotopy:proposition-homotopy-lifting}). Near each time, $\hat\gamma$ is $\gamma$ followed by a local inverse of $\pi$, which is a
local isometry, so $\hat\gamma$ is a geodesic; it has the initial velocity of $\tilde\gamma$, so it extends $\tilde\gamma$ to $\RR$, and $\tilde\gamma$ was defined on all of
$\RR$. By Theorem~\ref{riemannian-geometry:theorem-killing-hopf} there is an isometry $\Theta \colon M^n_\kappa \to \widetilde{M}$.

The deck transformations $\tau$ of $\pi$ satisfy $\pi \circ \tau = \pi$, so $\tau^*\tilde g = \tau^*\pi^*g = \pi^*g = \tilde g$: they are isometries. The group
$\mathrm{Deck}(\pi)$ is isomorphic to $\pi_1(M)$ and acts transitively on each fibre of $\pi$ (Proposition~\ref{homotopy:proposition-deck}), and freely, since a deck
transformation with a fixed point is a lift of $\pi$ agreeing with the identity at one point, hence the identity by the uniqueness in
Proposition~\ref{homotopy:proposition-lifting-criterion}.
So the fibres of $\pi$ are the orbits, and $\pi$ induces a bijection $\widetilde{M}/\mathrm{Deck}(\pi) \to M$, a homeomorphism because $\pi$ is open. The group
$\Gamma = \Theta^{-1}\,\mathrm{Deck}(\pi)\,\Theta$ acts on $M^n_\kappa$ by isometries, freely, and $\pi \circ \Theta$ identifies $M^n_\kappa/\Gamma$ with $M$; the projection is the
covering map $\pi \circ \Theta$, a local isometry.
\end{proof}

\begin{example}\label{riemannian-geometry:example-space-forms}
\begin{enumerate}
\item \emph{Flat manifolds.} The flat tori $\RR^n/\ZZ^n$ and the flat Klein bottle (Example~\ref{riemannian-geometry:example-quotient-metrics}) are complete flat manifolds,
quotients of $\RR^n$.
\item \emph{Spherical space forms.} $\RR\PP^n = S^n/\{\pm1\}$ and the lens spaces $L(p, q) = S^3/(\ZZ/p)$ (Example~\ref{topological-manifolds:example-covering-actions})
have constant curvature $1$, because the maps $(z_1, z_2) \mapsto (\zeta^kz_1, \zeta^{qk}z_2)$ are unitary, hence orthogonal on $\RR^4 = \CC^2$, and so isometries of
$S^3$. In particular, in dimension $3$ there are infinitely many spherical space forms, with fundamental groups $\ZZ/p$ for all $p$.
\item \emph{Hyperbolic surfaces.} By the Gauss--Bonnet theorem (Corollary~\ref{riemannian-geometry:corollary-gauss-bonnet}) a compact orientable surface with a metric of
constant curvature $-1$ has genus $g \geq 2$, and its area is $4\pi(g - 1)$; by Corollary~\ref{riemannian-geometry:corollary-space-forms} it is a quotient of the hyperbolic
plane.
\end{enumerate}
\end{example}

In even dimension the positive curvature case is very restricted.

\begin{corollary}\label{riemannian-geometry:corollary-even-spherical}
Let $n$ be even and $M$ complete and connected with constant curvature $\kappa > 0$. Then $M$ is isometric to $S^n(r)$ or to $\RR\PP^n(r) = S^n(r)/\{\pm1\}$, with
$r = 1/\sqrt\kappa$.
\end{corollary}

\begin{proof}
By Corollaries~\ref{riemannian-geometry:corollary-space-forms} and~\ref{riemannian-geometry:corollary-model-isometries}(3), $M = S^n(r)/\Gamma$ for a subgroup
$\Gamma \subseteq \mathrm{O}(n + 1)$ acting freely on the sphere. Every $A \in \mathrm{SO}(n + 1)$ has the eigenvalue $1$, because $n + 1$ is odd:
$\det(A - I) = \det(A - AA^T) = \det A\,\det(I - A^T) = \det(I - A) = (-1)^{n+1}\det(A - I) = -\det(A - I)$. A unit eigenvector, scaled to length $r$, is a fixed point of
$A$ on $S^n(r)$, so $\Gamma \cap \mathrm{SO}(n + 1) = \{I\}$, and $\det \colon \Gamma \to \{\pm1\}$ is injective. If $\Gamma \neq \{I\}$ it is $\{I, A\}$ with
$\det A = -1$ and $A^2 = I$; then $A$ is diagonalisable with eigenvalues $\pm1$ (its minimal polynomial divides $x^2 - 1$), and the eigenvalue $1$ is excluded since the
action is free, so $A = -I$.
\end{proof}

Finally we state two rigidity theorems for positive curvature, whose proofs use comparison geometry beyond this chapter. The first says that the diameter bound of
the Bonnet--Myers theorem (Theorem~\ref{riemannian-geometry:theorem-bonnet-myers}) is attained only by the round sphere.

\begin{theorem}[Cheng's maximal diameter theorem]\label{riemannian-geometry:theorem-maximal-diameter}
Let $M$ be complete and connected of dimension $n$, with $\mathrm{Ric} \geq (n - 1)\kappa g$ for some $\kappa > 0$, and with diameter $\pi/\sqrt\kappa$. Then $M$ is
isometric to the sphere $S^n(1/\sqrt\kappa)$.
\end{theorem}

This is \cite{cheng-1975}. For example $\RR\PP^n$ with the metric of constant curvature $1$ has $\mathrm{Ric} = (n - 1)g$ but diameter $\pi/2$, and a product
$S^2 \times S^2$ of unit spheres has $\mathrm{Ric} = g$ and diameter $\sqrt2\,\pi$, less than the bound $\pi/\sqrt{1/3} = \sqrt3\,\pi$ given by
Theorem~\ref{riemannian-geometry:theorem-bonnet-myers} with $n = 4$ and $3\kappa = 1$.

\begin{theorem}[Sphere theorem]\label{riemannian-geometry:theorem-sphere}
Let $M$ be complete and simply connected with sectional curvatures in $(\frac14, 1]$. Then $M$ is homeomorphic to $S^n$ (Rauch, Berger, Klingenberg), and in fact
diffeomorphic to $S^n$ (Brendle and Schoen).
\end{theorem}

The topological statement is \cite[Chapter 13]{do-carmo-riemannian}; the differentiable one is \cite{brendle-schoen-2009}, which proves more generally that a compact
manifold with pointwise $\frac14$-pinched curvature is diffeomorphic to a spherical space form. The constant $\frac14$ is optimal: the complex projective space $\CC\PP^m$
with the Fubini--Study metric (Chapter~\ref{kahler}) has sectional curvatures filling $[\frac14, 1]$ after scaling, it is compact and simply connected, and for $m \geq 2$
it is not homeomorphic to a sphere, since $H^2(\CC\PP^m; \ZZ) \cong \ZZ$. By Theorem~\ref{riemannian-geometry:theorem-bonnet-myers}, the hypothesis of the sphere theorem
already forces $M$ to be compact.

\section{Exercises}\label{riemannian-geometry:section-exercises}

\begin{exercise}\label{riemannian-geometry:exercise-hodge-star-signs}
Verify that $\delta = -*d*$ on $k$-forms when $n$ is even, and compute $\delta$ on $1$-forms in $\RR^3$, identifying it with minus the divergence.
\end{exercise}

\begin{exercise}\label{riemannian-geometry:exercise-conformal}
Show that in dimension $n = 2k$ the Hodge star on $k$-forms depends only on the conformal class of the metric, and that $*^2 = (-1)^k$ on $\Omega^k$.
\end{exercise}

\begin{solution}
Under $g \mapsto \lambda g$ with $\lambda > 0$, the inner product on $k$-forms scales by $\lambda^{-k}$ and the volume form by $\lambda^{n/2} = \lambda^k$, so $*$ is unchanged by the characterisation
$\alpha \wedge *\beta = \langle\alpha,\beta\rangle\mathrm{vol}$. The sign is $(-1)^{k(n-k)} = (-1)^{k^2} = (-1)^k$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-laplacian-sphere}
Compute the Laplacian on functions on the round sphere $S^2$ in spherical coordinates, and verify that the restrictions of linear functions of $\RR^3$ are eigenfunctions with eigenvalue $2$.
\end{exercise}

\begin{exercise}\label{riemannian-geometry:exercise-bi-invariant}
Let $G$ be a compact Lie group with a bi-invariant metric. Show that the geodesics through $e$ are the one-parameter subgroups, and that $\nabla_XY = \frac12[X, Y]$ for left-invariant fields.
\end{exercise}
\begin{exercise}\label{riemannian-geometry:exercise-great-circles}
Show that the geodesics of the round sphere $S^n \subseteq \RR^{n+1}$ are the great circles $\gamma(t) = \cos(t)p + \sin(t)v$ with $|p| = |v| = 1$ and $v \perp p$ (up to affine reparametrisation).
\end{exercise}

\begin{solution}
The reflection $\rho$ of $\RR^{n+1}$ fixing the plane spanned by $p$ and $v$ pointwise preserves $S^n$ and its metric, so it is an isometry, and it maps geodesics to geodesics. The geodesic $\gamma_v$ with $\gamma_v(0) = p$, $\dot\gamma_v(0) = v$
is mapped to a geodesic with the same initial data, hence to itself by uniqueness; so $\gamma_v$ lies in the fixed set of $\rho$ on $S^n$, the great circle through $p$ and $v$. Having constant speed $1$
(Proposition~\ref{riemannian-geometry:proposition-geodesic-speed}) and starting at $p$ with velocity $v$, it is $\cos(t)p + \sin(t)v$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-holonomy-product}
Show that the holonomy group of a Riemannian product $M_1 \times M_2$ at $(p_1, p_2)$ is $\mathrm{Hol}_{p_1}(M_1) \times \mathrm{Hol}_{p_2}(M_2)$, acting on $T_{p_1}M_1 \oplus T_{p_2}M_2$.
\end{exercise}

\begin{solution}
The Levi-Civita connection of the product metric is the direct sum of those of the factors, since the direct sum connection is metric and torsion-free. So a vector $(v_1, v_2)$ is transported along a curve $(c_1, c_2)$ to
$(P_{c_1}v_1, P_{c_2}v_2)$. Loops at $(p_1, p_2)$ are exactly the pairs of loops at $p_1$ and $p_2$, which gives the claim.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-trivial-holonomy}
Let $(M, g)$ be connected with $\mathrm{Hol}_p(g) = \{1\}$. Show that $M$ is flat and that $TM$ is trivial.
\end{exercise}

\begin{solution}
By Theorem~\ref{riemannian-geometry:theorem-holonomy-principle}, each vector of an orthonormal basis of $T_pM$ extends to a parallel vector field $E_i$; the $E_i$ are orthonormal everywhere (parallel transport is an isometry), so they
trivialise $TM$. For parallel fields $R(X, Y)E_i = \nabla_X\nabla_YE_i - \nabla_Y\nabla_XE_i - \nabla_{[X,Y]}E_i = 0$, so $R = 0$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-surface-revolution-area}
Let $f, h \colon (a, b) \to \RR$ be smooth with $f > 0$ and $(f^{\prime}, h^{\prime})$ nowhere zero, and let $S$ be the surface of revolution
$x(t, \theta) = (f(t)\cos\theta, f(t)\sin\theta, h(t))$, assuming that the profile curve $t \mapsto (f(t), h(t))$ is injective and a homeomorphism onto its image. Show that the area of $S$ is $2\pi\int_a^bf(t)\sqrt{f^{\prime}(t)^2 + h^{\prime}(t)^2}\,dt$.
\end{exercise}

\begin{solution}
Here $x_t = (f^{\prime}\cos\theta, f^{\prime}\sin\theta, h^{\prime})$ and $x_\theta = (-f\sin\theta, f\cos\theta, 0)$, so $E = f^{\prime 2} + h^{\prime 2}$, $F = 0$ and $G = f^2$.
By Proposition~\ref{riemannian-geometry:proposition-parametrised-volume}(2), $dA = f\sqrt{f^{\prime 2} + h^{\prime 2}}\,dt\,d\theta$ on the image of
$(a, b) \times (0, 2\pi)$, and the missing meridian $\theta = 0$ has area $0$ (it is a segment in the chart given by $\theta \in (-\pi, \pi)$). Integrating over $\theta$
gives the formula.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-archimedes}
Show that the area of the part of the unit sphere $S^2$ between the planes $z = c$ and $z = d$, for $-1 \leq c < d \leq 1$, is $2\pi(d - c)$: it depends only on the
distance between the planes (Archimedes).
\end{exercise}

\begin{solution}
Apart from the poles and a meridian, which have area $0$, the sphere is the surface of revolution of Exercise~\ref{riemannian-geometry:exercise-surface-revolution-area}
with $t = z \in (-1, 1)$, $f(z) = \sqrt{1 - z^2}$ and $h(z) = z$. Then $f^{\prime} = -z/\sqrt{1 - z^2}$ and $f\sqrt{f^{\prime 2} + 1} = \sqrt{1 - z^2} \cdot (1 - z^2)^{-1/2} = 1$, so
the area of the zone is $2\pi\int_c^d1\,dz = 2\pi(d - c)$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-killing-hyperbolic}
On the upper half plane $H = \{z = x + iy : y > 0\}$ with the metric $(dx^2 + dy^2)/y^2$, show that the Killing fields form a Lie algebra of dimension $3$, spanned by
$\partial_x$, $x\partial_x + y\partial_y$ and $(x^2 - y^2)\partial_x + 2xy\,\partial_y$. (Show first that the maps $z \mapsto (az + b)/(cz + d)$ with $a, b, c, d$ real and
$ad - bc = 1$ are isometries of $H$.)
\end{exercise}

\begin{solution}
For $\phi(z) = (az + b)/(cz + d)$ one computes $\operatorname{Im}\phi(z) = y/|cz + d|^2$ and $\phi^{\prime}(z) = 1/(cz + d)^2$, so $\phi$ maps $H$ to itself. As a map of
$\RR^2 = \CC$, its differential at $z$ is multiplication by the complex number $\phi^{\prime}(z)$, which multiplies Euclidean lengths by $|\phi^{\prime}(z)| = 1/|cz + d|^2$.
The hyperbolic length of $v \in T_zH$ is $|v|/y$, and that of its image is $|\phi^{\prime}(z)||v|/\operatorname{Im}\phi(z) = |v|/y$: $\phi$ is an isometry. The three fields generate the one-parameter groups
$z \mapsto z + t$, $z \mapsto e^tz$ and $z \mapsto z/(1 - tz)$ (for $z/(1 - tz)$, the derivative in $t$ at $t = 0$ is $z^2 = (x^2 - y^2) + 2ixy$), which are of the form above
(for the last one, with $a = d = 1$, $b = 0$, $c = -t$, defined on all of $H$ since $1 - tz \neq 0$ for $y > 0$); so they are Killing fields. They are linearly independent,
and by Theorem~\ref{riemannian-geometry:theorem-killing-algebra} the Killing algebra has dimension at most $2 \cdot 3/2 = 3$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-catenoid-helicoid}
Show that the helicoid and the catenoid of Example~\ref{riemannian-geometry:example-minimal-surfaces} are locally isometric: in the parameters $(u, w)$, with $v = \sinh w$ on the
helicoid, both metrics are $\cosh^2w\,(du^2 + dw^2)$.
\end{exercise}

\begin{solution}
The helicoid has the metric $(1 + v^2)\,du^2 + dv^2$. Substituting $v = \sinh w$, $dv = \cosh w\,dw$ and $1 + v^2 = \cosh^2w$ gives $\cosh^2w\,(du^2 + dw^2)$. The catenoid,
with $v$ renamed $w$, has $E = G = \cosh^2w$ and $F = 0$. So the map sending the point with parameters $(u, \sinh w)$ on the helicoid to the point with parameters $(u, w)$ on
the catenoid preserves the metric: it is a local isometry from the helicoid onto the catenoid, which winds the helicoid infinitely often around the catenoid, as the catenoid is
$2\pi$-periodic in $u$.
\end{solution}

\begin{exercise}\label{riemannian-geometry:exercise-no-compact-minimal}
Show that a minimal surface in $\RR^3$ has Gaussian curvature $K \leq 0$, and deduce that there is no compact minimal surface in $\RR^3$.
\end{exercise}

\begin{solution}
If $H = 0$, the principal curvatures are $\kappa$ and $-\kappa$, so $K = -\kappa^2 \leq 0$ (Corollary~\ref{riemannian-geometry:corollary-theorema-egregium}). A compact surface in
$\RR^3$ has a point with $K > 0$ by Lemma~\ref{riemannian-geometry:lemma-elliptic-point}.
\end{solution}
