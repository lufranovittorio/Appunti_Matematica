\chapterstatus{complete}

\section{Introduction}\label{fields:section-introduction}

This chapter studies field extensions: algebraic elements, splitting fields and
algebraic closures, separability, the fundamental theorem of Galois theory, finite
fields, cyclotomic fields and solvability by radicals. We give Artin's algebraic proof
of the fundamental theorem of algebra, and we study transcendence bases, which show
that $\CC$ has an enormous group of automorphisms. These automorphisms reappear in the
theory of absolute Hodge classes (Chapter~\ref{absolute-hodge}).

References are \cite{lang-algebra}, \cite{artin-galois} and \cite{stewart-galois}.

\noindent\textbf{Prerequisites.} Chapter~\ref{number-systems} (Number Systems),
Chapter~\ref{groups} (Groups) and Chapter~\ref{rings} (Rings and Modules).

\section{Field extensions}\label{fields:section-extensions}

\begin{definition}\label{fields:definition-extension}
A \emph{field extension} $L/K$ is a field $L$ together with a subfield $K$ (more
generally, a homomorphism of fields $K \to L$, which is injective by
Counterexample~\ref{categories:counterexample-fields-products}). Then $L$ is a
$K$-vector space, and its dimension $[L : K] = \dim_K L$ is the \emph{degree} of the
extension; the extension is \emph{finite} if the degree is finite. For a subset
$S \subseteq L$, $K(S)$ denotes the smallest subfield of $L$ containing $K$ and $S$, and $K[S]$
the smallest subring. A \emph{$K$-homomorphism} between extensions of $K$ is a field
homomorphism that is the identity on $K$; $\Aut(L/K)$ is the group of $K$-automorphisms
of $L$.
\end{definition}

\begin{theorem}[Tower law]\label{fields:theorem-tower}
Let $K \subseteq L \subseteq M$ be fields. If $(l_i)_{i \in I}$ is a basis of $L$ over $K$ and $(m_j)_{j \in J}$ a
basis of $M$ over $L$, then $(l_i m_j)_{(i,j) \in I \times J}$ is a basis of $M$ over $K$. In
particular $[M : K] = [M : L][L : K]$.
\end{theorem}

\begin{proof}
Every $x \in M$ is $\sum_j a_j m_j$ with $a_j \in L$, and each $a_j = \sum_i c_{ij} l_i$ with
$c_{ij} \in K$, so the $l_i m_j$ generate. If $\sum_{i,j} c_{ij} l_i m_j = 0$, then
$\sum_j (\sum_i c_{ij} l_i) m_j = 0$ gives $\sum_i c_{ij} l_i = 0$ for all $j$, hence all $c_{ij} = 0$.
\end{proof}

\begin{definition}\label{fields:definition-algebraic}
Let $L/K$ be an extension. An element $\alpha \in L$ is \emph{algebraic} over $K$ if
$f(\alpha) = 0$ for some nonzero $f \in K[x]$, and \emph{transcendental} otherwise. The
extension is \emph{algebraic} if all elements of $L$ are algebraic over $K$.
\end{definition}

\begin{proposition}\label{fields:proposition-minimal-polynomial}
Let $\alpha \in L$ be algebraic over $K$. The kernel of the evaluation homomorphism
$K[x] \to L$, $f \mapsto f(\alpha)$, is generated by a unique monic irreducible polynomial
$m_\alpha$, the \emph{minimal polynomial} of $\alpha$. The evaluation induces an isomorphism
$K[x]/(m_\alpha) \cong K[\alpha]$, so $K[\alpha] = K(\alpha)$ is a field with basis
$1, \alpha, \ldots, \alpha^{d-1}$ over $K$, where $d = \deg m_\alpha = [K(\alpha) : K]$.
\end{proposition}

\begin{proof}
The kernel is a nonzero ideal of the PID $K[x]$
(Theorem~\ref{rings:theorem-polynomial-division}), hence generated by a unique monic
polynomial. The image $K[\alpha]$ is a subring of a field, hence a domain, so the kernel is
prime, and the generator is irreducible. Nonzero prime ideals of a PID are maximal
(proof of Theorem~\ref{rings:theorem-euclidean-pid-ufd}), so $K[\alpha] \cong K[x]/(m_\alpha)$ is a
field (Proposition~\ref{rings:proposition-prime-maximal-quotients}). By division with
remainder, the classes of $1, x, \ldots, x^{d-1}$ form a basis of $K[x]/(m_\alpha)$.
\end{proof}

\begin{proposition}\label{fields:proposition-algebraic-closure-in-extension}
\begin{enumerate}
\item An element $\alpha$ is algebraic over $K$ if and only if $[K(\alpha) : K] < \infty$.
Every finite extension is algebraic.
\item If $\alpha_1, \ldots, \alpha_n$ are algebraic over $K$, then $K(\alpha_1, \ldots, \alpha_n)/K$ is
finite. The elements of $L$ algebraic over $K$ form a subfield of $L$.
\item If $L/K$ and $M/L$ are algebraic, then $M/K$ is algebraic.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) If $[L : K] = n$ is finite and $\alpha \in L$, the $n + 1$ elements $1, \alpha, \ldots, \alpha^n$ are
linearly dependent, which gives a nonzero polynomial with root $\alpha$. The rest follows
from Proposition~\ref{fields:proposition-minimal-polynomial}.
(2) Each $\alpha_i$ is algebraic over $K(\alpha_1, \ldots, \alpha_{i-1})$, so the claim follows from the
tower law. If $\alpha, \beta$ are algebraic, then $\alpha \pm \beta$, $\alpha\beta$ and $\alpha^{-1}$ lie in the
finite extension $K(\alpha, \beta)$ and are algebraic by (1).
(3) Let $\alpha \in M$ be a root of $c_nx^n + \cdots + c_0$ with $c_i \in L$ not all zero. Then $\alpha$ is
algebraic over $K(c_0, \ldots, c_n)$, which is finite over $K$ by (2); so $K(c_0, \ldots, c_n, \alpha)$
is finite over $K$ and $\alpha$ is algebraic by (1).
\end{proof}

\begin{example}\label{fields:example-degrees}
$[\QQ(\sqrt 2) : \QQ] = 2$, since $x^2 - 2$ is irreducible by
Proposition~\ref{number-systems:proposition-sqrt-two}. The real number $\sqrt[3]{2}$ has
minimal polynomial $x^3 - 2$, irreducible by Eisenstein's criterion
(Proposition~\ref{rings:proposition-eisenstein}), so $[\QQ(\sqrt[3]{2}) : \QQ] = 3$. Hence
$\sqrt[3]{2} \notin \QQ(\sqrt 2)$, by the tower law. Transcendental numbers exist by
Exercise~\ref{number-systems:exercise-algebraic-numbers}.
\end{example}

\begin{proposition}[Kronecker]\label{fields:proposition-kronecker}
Let $f \in K[x]$ be irreducible. Then $L = K[x]/(f)$ is an extension of $K$ of degree
$\deg f$ in which the class of $x$ is a root of $f$.
\end{proposition}

\begin{proof}
$(f)$ is maximal, so $L$ is a field; the composite $K \to K[x] \to L$ is injective, and
$f(\bar x) = \overline{f(x)} = 0$. The degree is computed as in
Proposition~\ref{fields:proposition-minimal-polynomial}.
\end{proof}

\section{Splitting fields and algebraic closures}\label{fields:section-splitting}

\begin{definition}\label{fields:definition-splitting-field}
Let $f \in K[x]$ be nonconstant. An extension $L/K$ is a \emph{splitting field} of $f$ if
$f = c(x - \alpha_1) \cdots (x - \alpha_n)$ in $L[x]$ and $L = K(\alpha_1, \ldots, \alpha_n)$. A field is
\emph{algebraically closed} if every nonconstant polynomial over it has a root, and an
\emph{algebraic closure} of $K$ is an algebraic extension that is algebraically closed.
\end{definition}

\begin{proposition}\label{fields:proposition-splitting-exists}
Every nonconstant $f \in K[x]$ has a splitting field $L$, with $[L : K] \leq (\deg f)!$.
\end{proposition}

\begin{proof}
By induction on $n = \deg f$. Let $g$ be an irreducible factor of $f$ and
$K_1 = K[x]/(g)$, in which $f$ has a root $\alpha_1$ and $[K_1 : K] \leq n$. Over $K_1$,
$f = (x - \alpha_1) f_1$ with $\deg f_1 = n - 1$; a splitting field of $f_1$ over $K_1$ is a
splitting field of $f$ over $K$ of degree at most $n \cdot (n-1)!$.
\end{proof}

\begin{theorem}[Extension of homomorphisms]\label{fields:theorem-extension}
Let $L = K(\alpha_1, \ldots, \alpha_r)$ be a finite extension and $\tau \colon K \to M$ a field
homomorphism. The number of homomorphisms $L \to M$ extending $\tau$ is at most $[L : K]$.
For $i = 1, \ldots, r$ let $g_i$ be the minimal polynomial of $\alpha_i$ over
$K(\alpha_1, \ldots, \alpha_{i-1})$. If for every extension $\tau'$ of $\tau$ to
$K(\alpha_1, \ldots, \alpha_{i-1})$ the polynomial $g_i^{\tau'}$ splits into distinct linear factors
in $M[x]$, then there are exactly $[L : K]$ extensions.
\end{theorem}

\begin{proof}
By induction on $r$, using the tower law, it suffices to treat $r = 1$: $L = K(\alpha)$ with
minimal polynomial $g$. An extension $\sigma$ of $\tau$ is determined by $\sigma(\alpha)$, which is a
root of $g^\tau$ (the polynomial obtained by applying $\tau$ to the coefficients).
Conversely, for each root $\beta \in M$ of $g^\tau$, the homomorphism
$K[x] \to M$ extending $\tau$ with $x \mapsto \beta$ kills $g$ and so induces an extension
$K(\alpha) \cong K[x]/(g) \to M$. Hence the extensions correspond to the roots of $g^\tau$ in $M$,
of which there are at most $\deg g = [K(\alpha) : K]$ by Corollary~\ref{rings:corollary-roots},
with equality if $g^\tau$ splits into distinct linear factors.
\end{proof}

\begin{corollary}\label{fields:corollary-splitting-unique}
Let $L$ and $L'$ be splitting fields of $f \in K[x]$. Then there is a $K$-isomorphism
$L \to L'$. For every finite extension $L/K$, $|\Aut(L/K)| \leq [L : K]$.
\end{corollary}

\begin{proof}
Write $L = K(\alpha_1, \ldots, \alpha_n)$ with the $\alpha_i$ the roots of $f$. At each stage the
minimal polynomial $g_i$ divides $f$, and $f$ splits in $L'$, so $g_i^{\tau'}$ has a root in $L'$
and the proof of Theorem~\ref{fields:theorem-extension} gives at least one extension of
the inclusion $K \to L'$ to $\sigma \colon L \to L'$. It is injective, and its image contains all
roots of $f$ in $L'$ (it contains $n$ roots counted with multiplicity), so it is $L'$. The
second statement is the first part of Theorem~\ref{fields:theorem-extension} with
$M = L$.
\end{proof}

\begin{theorem}\label{fields:theorem-algebraic-closure}
Every field $K$ has an algebraic closure $\bar K$. If $L/K$ is algebraic and $\tau \colon K \to M$ is
a homomorphism into an algebraically closed field, then $\tau$ extends to $L \to M$. Any two
algebraic closures of $K$ are $K$-isomorphic.
\end{theorem}

\begin{proof}
\emph{Existence} (Artin). Let $\mathcal{F}$ be the set of monic nonconstant polynomials in
$K[x]$, let $A = K[x_f : f \in \mathcal{F}]$ be the polynomial ring in variables indexed by
$\mathcal{F}$ (the union of the polynomial rings in finitely many of them), and $I$ the ideal
generated by the $f(x_f)$. Then $I \neq A$: otherwise $1 = \sum_{i=1}^n g_i f_i(x_{f_i})$ for
finitely many $f_i$; in a splitting field of $f_1 \cdots f_n$ choose roots $\beta_i$ of $f_i$,
and apply the homomorphism $x_{f_i} \mapsto \beta_i$, other variables $\mapsto 0$, to get $1 = 0$. Let
$\mathfrak{m} \supseteq I$ be maximal (Theorem~\ref{rings:theorem-maximal-ideals-exist}) and
$E_1 = A/\mathfrak{m}$, a field extension of $K$ in which every $f \in \mathcal{F}$ has a root.
Repeating, we get $K = E_0 \subseteq E_1 \subseteq E_2 \subseteq \cdots$ where every nonconstant polynomial
over $E_n$ has a root in $E_{n+1}$. The union $E$ is algebraically closed, since a
polynomial over $E$ has coefficients in some $E_n$. The elements of $E$ algebraic over
$K$ form a field $\bar K$ (Proposition~\ref{fields:proposition-algebraic-closure-in-extension}).
If $g \in \bar K[x]$ is nonconstant, a root $\alpha \in E$ is algebraic over $\bar K$, hence over
$K$, so $\alpha \in \bar K$. Thus $\bar K$ is an algebraic closure.

\emph{Extension.} Consider pairs $(F, \sigma)$ with $K \subseteq F \subseteq L$ a subfield and
$\sigma \colon F \to M$ extending $\tau$, ordered by extension. Chains have upper bounds (take
unions), so by Zorn's lemma there is a maximal $(F, \sigma)$. If $\alpha \in L \setminus F$, its minimal
polynomial over $F$ maps to a polynomial with a root in $M$, and
Theorem~\ref{fields:theorem-extension} extends $\sigma$ to $F(\alpha)$; contradiction. So $F = L$.

\emph{Uniqueness.} If $\bar K$ and $\bar K'$ are algebraic closures, extend the inclusion
$K \to \bar K'$ to $\sigma \colon \bar K \to \bar K'$. Its image is an algebraically closed field over
which $\bar K'$ is algebraic, so every element of $\bar K'$ is a root of a polynomial that
splits over $\sigma(\bar K)$, and $\sigma$ is surjective.
\end{proof}

\section{Separability}\label{fields:section-separability}

\begin{definition}\label{fields:definition-separable}
A nonzero polynomial $f \in K[x]$ is \emph{separable} if it has no repeated root in a
splitting field. An algebraic element is \emph{separable} if its minimal polynomial is,
and an algebraic extension is separable if all its elements are. The \emph{derivative} of
$f = \sum a_i x^i$ is $f' = \sum i a_i x^{i-1}$; it satisfies $(fg)' = f'g + fg'$.
\end{definition}

\begin{proposition}\label{fields:proposition-separable-criterion}
A nonzero polynomial $f$ is separable if and only if $\gcd(f, f') = 1$ in $K[x]$. An
irreducible polynomial is separable if and only if $f' \neq 0$. If $\operatorname{char} K = 0$,
every irreducible polynomial is separable; if $\operatorname{char} K = p > 0$, an irreducible
$f$ is inseparable if and only if $f = g(x^p)$ for some $g \in K[x]$.
\end{proposition}

\begin{proof}
In a splitting field, if $f = (x - \alpha)^2 h$ then $f'(\alpha) = 0$; if $f = (x - \alpha)h$ with
$h(\alpha) \neq 0$ then $f'(\alpha) = h(\alpha) \neq 0$. So $f$ has a repeated root iff $f$ and $f'$ have a
common root in a splitting field, iff they have a common factor of positive degree in
$K[x]$ (a common root $\alpha$ makes $m_\alpha$ a common factor; conversely a common factor has
a root in the splitting field). If $f$ is irreducible and $f' \neq 0$, the gcd divides $f$ and
has degree $< \deg f$, so it is $1$. If $f' = 0$ then $\gcd(f, f') = f$. Finally $f' = 0$ iff
$i a_i = 0$ for all $i$, which in characteristic $0$ means $f$ is constant, and in
characteristic $p$ means $a_i = 0$ unless $p \mid i$.
\end{proof}

\begin{definition}\label{fields:definition-perfect}
A field $K$ is \emph{perfect} if it has characteristic $0$, or characteristic $p$ and the
\emph{Frobenius} $x \mapsto x^p$ is surjective. (The Frobenius is always an injective ring
homomorphism in characteristic $p$, since $p$ divides the binomial coefficients
$\binom{p}{i}$ for $0 < i < p$.)
\end{definition}

\begin{proposition}\label{fields:proposition-perfect}
Over a perfect field every irreducible polynomial is separable, so every algebraic
extension is separable. Finite fields are perfect.
\end{proposition}

\begin{proof}
If $\operatorname{char} K = p$ and $f = \sum b_i x^{pi}$ with $b_i = c_i^p$, then $f = (\sum c_i x^i)^p$
is not irreducible. A finite field is perfect because an injective map from a finite set
to itself is surjective.
\end{proof}

\begin{counterexample}\label{fields:counterexample-inseparable}
Let $K = \mathbb{F}_p(t)$ be the field of fractions of $\mathbb{F}_p[t]$. The polynomial
$f = x^p - t$ is irreducible, by Eisenstein's criterion for the prime $t$ of the UFD
$\mathbb{F}_p[t]$, but $f' = 0$. In a splitting field, with $\beta^p = t$, we have
$f = (x - \beta)^p$: a single root of multiplicity $p$. So $K$ is not perfect and $K(\beta)/K$ is
not separable.
\end{counterexample}

\section{Galois extensions}\label{fields:section-galois}

\begin{definition}\label{fields:definition-normal}
An algebraic extension $L/K$ is \emph{normal} if every irreducible polynomial in $K[x]$
with a root in $L$ splits into linear factors in $L[x]$. For a group $G$ of automorphisms
of a field $L$, the \emph{fixed field} is $L^G = \{x \in L : \sigma(x) = x \text{ for all } \sigma \in G\}$.
\end{definition}

\begin{lemma}[Artin]\label{fields:lemma-artin}
Let $G$ be a finite group of automorphisms of a field $L$. Then $[L : L^G] \leq |G|$.
\end{lemma}

\begin{proof}
Let $K = L^G$, $G = \{\sigma_1, \ldots, \sigma_n\}$ with $\sigma_1 = \id$, and let
$\alpha_1, \ldots, \alpha_m \in L$ with $m > n$. The $m$ vectors $(\sigma_i(\alpha_j))_i \in L^n$ are linearly
dependent over $L$ by Lemma~\ref{rings:lemma-exchange}, so the system
$\sum_j \sigma_i(\alpha_j) x_j = 0$, $i = 1, \ldots, n$, has a nonzero solution $(x_j) \in L^m$. Choose
one with the fewest nonzero entries and, scaling, with $x_{j_0} = 1$ for some $j_0$. For
$\tau \in G$, applying $\tau$ gives $\sum_j \tau\sigma_i(\alpha_j)\tau(x_j) = 0$ for all $i$, and since
$\tau\sigma_i$ runs over $G$ as $i$ does, $(\tau(x_j))_j$ is also a solution. The difference
$(x_j - \tau(x_j))_j$ is a solution with fewer nonzero entries (the $j_0$ entry vanishes), so it
is zero. Hence all $x_j \in K$, and the equation for $\sigma_1 = \id$ is a nontrivial $K$-linear
relation $\sum_j \alpha_j x_j = 0$. So any $m > n$ elements are linearly dependent over $K$.
\end{proof}

\begin{theorem}\label{fields:theorem-galois-characterisation}
For a finite extension $L/K$ with $G = \Aut(L/K)$, the following are equivalent:
\begin{enumerate}
\item $|G| = [L : K]$;
\item $L^G = K$;
\item $L$ is a splitting field over $K$ of a separable polynomial;
\item $L/K$ is normal and separable.
\end{enumerate}
Such an extension is called \emph{Galois}, and $\mathrm{Gal}(L/K) = G$ its \emph{Galois group}.
Moreover, if $G$ is any finite group of automorphisms of a field $L$, then $L/L^G$ is Galois
with group $G$.
\end{theorem}

\begin{proof}
For the last statement: $G \subseteq \Aut(L/L^G)$, so by Lemma~\ref{fields:lemma-artin} and
Corollary~\ref{fields:corollary-splitting-unique},
$[L : L^G] \leq |G| \leq |\Aut(L/L^G)| \leq [L : L^G]$.

(1) $\Rightarrow$ (2): $K \subseteq L^G$ and by the last statement $[L : L^G] = |G| = [L : K]$, so
$L^G = K$ by the tower law.

(2) $\Rightarrow$ (4): Let $\alpha \in L$ and let $\alpha_1, \ldots, \alpha_r$ be the distinct elements of the
orbit $G\alpha$. The polynomial $g = \prod_i (x - \alpha_i)$ has coefficients fixed by $G$, hence in
$K$, and $g(\alpha) = 0$, so $m_\alpha \mid g$. Every $\alpha_i = \sigma(\alpha)$ is a root of $m_\alpha$, so $g \mid m_\alpha$.
Thus $m_\alpha = g$ splits in $L$ with distinct roots.

(4) $\Rightarrow$ (3): Write $L = K(\beta_1, \ldots, \beta_s)$ and let $f$ be the product of the distinct
minimal polynomials of the $\beta_j$. It is separable (distinct monic irreducible
polynomials have no common root), it splits in $L$ by normality, and its roots
generate $L$.

(3) $\Rightarrow$ (1): Apply Theorem~\ref{fields:theorem-extension} with $M = L$ and $\tau$ the
inclusion $K \to L$: at every stage the minimal polynomial divides $f$, which splits into
distinct linear factors in $L$. So there are exactly $[L : K]$ $K$-homomorphisms $L \to L$.
Each is an injective $K$-linear map of the finite-dimensional $K$-vector space $L$ to itself,
hence bijective by Theorem~\ref{rings:theorem-dimension} (its image is a subspace of the
same finite dimension). So $|G| = [L : K]$.
\end{proof}

\begin{theorem}[Fundamental theorem of Galois theory]\label{fields:theorem-fundamental}
Let $L/K$ be a finite Galois extension with group $G$. The maps $H \mapsto L^H$ and
$F \mapsto \mathrm{Gal}(L/F)$ are mutually inverse, inclusion-reversing bijections between
subgroups $H \leq G$ and intermediate fields $K \subseteq F \subseteq L$. Moreover
$[L : L^H] = |H|$ and $[L^H : K] = [G : H]$. An intermediate field $F$ is Galois over $K$ if
and only if $\mathrm{Gal}(L/F)$ is normal in $G$, and then restriction induces an isomorphism
$G/\mathrm{Gal}(L/F) \cong \mathrm{Gal}(F/K)$.
\end{theorem}

\begin{proof}
If $L$ is the splitting field of a separable $f \in K[x]$, it is also the splitting field of
$f$ over $F$, so $L/F$ is Galois and $L^{\mathrm{Gal}(L/F)} = F$ by
Theorem~\ref{fields:theorem-galois-characterisation}. For $H \leq G$, the last statement of
that theorem gives $\mathrm{Gal}(L/L^H) = H$ and $[L : L^H] = |H|$; then
$[L^H : K] = [L : K]/|H| = [G : H]$ by the tower law. The maps clearly reverse inclusions.

For $\sigma \in G$, $\mathrm{Gal}(L/\sigma(F)) = \sigma\mathrm{Gal}(L/F)\sigma^{-1}$. If $F/K$ is Galois, it is
normal, so $\sigma$ maps each $\alpha \in F$ to a root of $m_\alpha$, which lies in $F$; thus
$\sigma(F) = F$ and $\mathrm{Gal}(L/F)$ is normal. Conversely, if $N = \mathrm{Gal}(L/F)$ is normal, then
$\sigma(F) = L^{\sigma N \sigma^{-1}} = F$ for all $\sigma$, and restriction is a homomorphism
$G \to \Aut(F/K)$ with kernel $N$. Its image has $[G : N] = [F : K]$ elements and lies in a
group with at most $[F : K]$ elements (Corollary~\ref{fields:corollary-splitting-unique}), so
$|\Aut(F/K)| = [F : K]$, $F/K$ is Galois and the restriction map is surjective.
\end{proof}

\begin{example}\label{fields:example-biquadratic}
The field $L = \QQ(\sqrt 2, \sqrt 3)$ is the splitting field of $(x^2 - 2)(x^2 - 3)$, of degree $4$
over $\QQ$: if $\sqrt 3 = a + b\sqrt 2$ with $a, b \in \QQ$, squaring gives $2ab = 0$, and neither
$a = 0$ nor $b = 0$ is possible by the argument of
Proposition~\ref{number-systems:proposition-sqrt-two}. Its Galois group consists of the four
automorphisms $\sqrt 2 \mapsto \pm\sqrt 2$, $\sqrt 3 \mapsto \pm\sqrt 3$, so it is isomorphic to
$\ZZ/2\ZZ \times \ZZ/2\ZZ$. It has three subgroups of order $2$, corresponding to the three
quadratic subfields $\QQ(\sqrt 2)$, $\QQ(\sqrt 3)$ and $\QQ(\sqrt 6)$.
\end{example}

\begin{counterexample}\label{fields:counterexample-not-galois}
The extension $\QQ(\sqrt[3]{2})/\QQ$ of degree $3$ is not Galois. A $\QQ$-automorphism must send
$\sqrt[3]{2}$ to a cube root of $2$ in $\QQ(\sqrt[3]{2}) \subseteq \RR$, and the only real cube root of
$2$ is $\sqrt[3]{2}$ (Proposition~\ref{number-systems:proposition-roots}); so the automorphism
group is trivial. The polynomial $x^3 - 2$ has a root in $\QQ(\sqrt[3]{2})$ but does not split
there, so the extension is not normal.
\end{counterexample}

\section{Finite fields}\label{fields:section-finite-fields}

\begin{lemma}\label{fields:lemma-cyclic}
Every finite subgroup of the multiplicative group of a field is cyclic.
\end{lemma}

\begin{proof}
Let $G$ be such a subgroup. By Theorem~\ref{groups:theorem-structure-abelian},
$G \cong \ZZ/d_1\ZZ \oplus \cdots \oplus \ZZ/d_k\ZZ$ with $d_1 \mid \cdots \mid d_k$, so every element satisfies
$x^{d_k} = 1$. The polynomial $x^{d_k} - 1$ has at most $d_k$ roots, so $|G| \leq d_k$ and
$k \leq 1$.
\end{proof}

\begin{theorem}\label{fields:theorem-finite-fields}
\begin{enumerate}
\item A finite field has $q = p^n$ elements, where $p$ is its characteristic and $n$ its
degree over $\mathbb{F}_p$.
\item For every prime power $q = p^n$ there is a field $\mathbb{F}_q$ with $q$ elements, unique up to
isomorphism: a splitting field of $x^q - x$ over $\mathbb{F}_p$, consisting of the roots of this
polynomial.
\item $\mathbb{F}_{q^n}/\mathbb{F}_q$ is Galois with cyclic group of order $n$, generated by the
Frobenius $x \mapsto x^q$. The subfields of $\mathbb{F}_{p^n}$ are the $\mathbb{F}_{p^m}$ with $m \mid n$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) The characteristic is prime by Lemma~\ref{rings:lemma-characteristic-domain}, and $F$ is
a vector space of some finite dimension $n$ over $\mathbb{F}_p$.
(2) Let $L$ be a splitting field of $f = x^q - x$. Since $f' = -1$, $f$ has $q$ distinct roots. The
set of roots is closed under sums, products and inverses because $x \mapsto x^q$ is a ring
homomorphism (a power of the Frobenius), so it is a subfield, and it equals $L$. If $F$ has $q$
elements, then $F^\times$ has order $q - 1$, so $x^q = x$ for all $x \in F$ by
Theorem~\ref{groups:theorem-lagrange}; hence $F$ is a splitting field of $f$ and $F \cong L$ by
Corollary~\ref{fields:corollary-splitting-unique}.
(3) $\mathbb{F}_{q^n}$ is the splitting field of the separable polynomial $x^{q^n} - x$ over
$\mathbb{F}_q$, hence Galois of degree $n$. The Frobenius $\phi(x) = x^q$ fixes $\mathbb{F}_q$, and
$\phi^k = \id$ means all $q^n$ elements are roots of $x^{q^k} - x$, which requires $k \geq n$. So $\phi$
has order $n$ and generates the Galois group. By the fundamental theorem, subfields
correspond to subgroups of $\ZZ/n\ZZ$, one for each divisor.
\end{proof}

\begin{theorem}[Primitive element theorem]\label{fields:theorem-primitive-element}
Every finite separable extension $L/K$ is \emph{simple}: $L = K(\gamma)$ for some $\gamma \in L$.
\end{theorem}

\begin{proof}
If $K$ is finite, so is $L$, and a generator $\gamma$ of the cyclic group $L^\times$
(Lemma~\ref{fields:lemma-cyclic}) works. Let $K$ be infinite. By induction on the number of
generators it suffices to treat $L = K(\alpha, \beta)$. Let $M$ be a splitting field of
$m_\alpha m_\beta$ over $L$. The minimal polynomials over intermediate fields divide $m_\alpha$ or
$m_\beta$, which are separable, so by Theorem~\ref{fields:theorem-extension} there are exactly
$n = [L : K]$ distinct $K$-homomorphisms $\sigma_1, \ldots, \sigma_n \colon L \to M$. The polynomial
$P(t) = \prod_{i < j}\bigl((\sigma_i(\alpha) - \sigma_j(\alpha)) + t(\sigma_i(\beta) - \sigma_j(\beta))\bigr) \in M[t]$ is
nonzero, because $\sigma_i \neq \sigma_j$ differ on $\alpha$ or on $\beta$. It has finitely many roots, so
there is $c \in K$ with $P(c) \neq 0$. Then $\gamma = \alpha + c\beta$ has $n$ distinct images
$\sigma_i(\gamma)$. The restrictions of the $\sigma_i$ to $K(\gamma)$ are therefore distinct, so
$[K(\gamma) : K] \geq n$ by Theorem~\ref{fields:theorem-extension}, and $K(\gamma) = L$.
\end{proof}

\section{The fundamental theorem of algebra}\label{fields:section-fta}

\begin{lemma}\label{fields:lemma-odd-degree}
Every polynomial of odd degree in $\RR[x]$ has a root in $\RR$.
\end{lemma}

\begin{proof}
We may assume $p = x^n + a_{n-1}x^{n-1} + \cdots + a_0$ is monic, with $n$ odd. Let
$M = 1 + \sum_i |a_i|$. For $|x| \geq M$ we have
$|a_{n-1}x^{n-1} + \cdots + a_0| \leq (M - 1)|x|^{n-1} < |x|^n$, so $p(x)$ has the sign of $x^n$, that is,
of $x$. On $[-M, M]$, the identity $x^i - y^i = (x - y)\sum_{k} x^k y^{i-1-k}$ gives
$|p(x) - p(y)| \leq C|x - y|$ with $C = \sum_{i=1}^n i|a_i| M^{i-1}$ (with $a_n = 1$). Let
$S = \{x \in [-M, M] : p(x) < 0\}$, which contains $-M$, and $c = \sup S$. If $p(c) < 0$, then
$c < M$ and $p < 0$ on $[c, c + |p(c)|/C) \cap [-M, M]$, contradicting $c = \sup S$. If $p(c) > 0$, then
$c > -M$ and $p > 0$ on $(c - p(c)/C, c]$, so $\sup S \leq c - p(c)/C$, again a contradiction. Hence
$p(c) = 0$.
\end{proof}

\begin{lemma}\label{fields:lemma-p-group-index-p}
Every group of order $p^n$ with $n \geq 1$ has a subgroup of index $p$.
\end{lemma}

\begin{proof}
By induction on $n$. By Proposition~\ref{groups:proposition-p-group-center} and
Theorem~\ref{groups:theorem-cauchy} there is $z \in Z(G)$ of order $p$. If $n = 1$ take $\{e\}$.
Otherwise $G/\langle z \rangle$ has order $p^{n-1}$ and, by induction, a subgroup of index $p$,
whose preimage in $G$ has index $p$.
\end{proof}

\begin{theorem}[Fundamental theorem of algebra]\label{fields:theorem-fta}
The field $\CC$ is algebraically closed.
\end{theorem}

\begin{proof}
Let $f \in \CC[x]$ be nonconstant, and let $\bar f$ be obtained by conjugating the coefficients.
Then $g = (x^2 + 1) f \bar f$ lies in $\RR[x]$. Let $L$ be a splitting field of $g$ over $\CC$; it is also
a splitting field of $g$ over $\RR$ (since $\CC = \RR(i)$ and $i$ is a root of $g$), hence Galois over
$\RR$ as $\operatorname{char} \RR = 0$. Let $G = \mathrm{Gal}(L/\RR)$ and let $P$ be a Sylow $2$-subgroup.
The fixed field $L^P$ has odd degree $[G : P]$ over $\RR$; by
Theorem~\ref{fields:theorem-primitive-element} it is $\RR(\gamma)$, and the minimal polynomial of
$\gamma$ is irreducible of odd degree, hence of degree $1$ by Lemma~\ref{fields:lemma-odd-degree}.
So $P = G$ is a $2$-group, and so is $H = \mathrm{Gal}(L/\CC)$. If $H \neq \{e\}$, it has a subgroup of
index $2$ (Lemma~\ref{fields:lemma-p-group-index-p}), whose fixed field is an extension of
$\CC$ of degree $2$, generated by a root of an irreducible quadratic polynomial over $\CC$. This
contradicts Proposition~\ref{number-systems:proposition-complex-square-roots}. So
$L = \CC$, and $f$ has its roots in $\CC$.
\end{proof}

\begin{corollary}\label{fields:corollary-real-irreducibles}
Every nonconstant polynomial over $\CC$ is a product of linear factors, and $\CC$ is an
algebraic closure of $\RR$. The irreducible polynomials in $\RR[x]$ are the linear ones and the
quadratic ones without real roots.
\end{corollary}

\begin{proof}
The first statement follows by induction on the degree. Since $[\CC : \RR] = 2$, the
extension is algebraic. An irreducible real polynomial of degree $\geq 2$ has a non-real root
$\alpha$; then $\bar\alpha$ is also a root, and $(x - \alpha)(x - \bar\alpha) \in \RR[x]$ divides it.
\end{proof}

\section{Cyclotomic fields}\label{fields:section-cyclotomic}

\begin{definition}\label{fields:definition-roots-of-unity}
For $n \geq 1$, the $n$-th roots of unity in $\CC$ form a group $\mu_n$ of order $n$ (the polynomial
$x^n - 1$ has derivative $nx^{n-1}$, so its $n$ roots are distinct), cyclic by
Lemma~\ref{fields:lemma-cyclic}. Its generators are the \emph{primitive} $n$-th roots of unity.
The $n$-th \emph{cyclotomic polynomial} is $\Phi_n = \prod_{\zeta} (x - \zeta)$ over the primitive
$n$-th roots of unity $\zeta$; it has degree $\varphi(n) = |(\ZZ/n\ZZ)^\times|$.
\end{definition}

\begin{theorem}\label{fields:theorem-cyclotomic}
$\Phi_n$ lies in $\ZZ[x]$ and is irreducible over $\QQ$. The extension $\QQ(\zeta)/\QQ$ for a primitive
$n$-th root of unity $\zeta$ is Galois, and $\sigma \mapsto a_\sigma$, where $\sigma(\zeta) = \zeta^{a_\sigma}$, is an
isomorphism $\mathrm{Gal}(\QQ(\zeta)/\QQ) \cong (\ZZ/n\ZZ)^\times$.
\end{theorem}

\begin{proof}
Every $n$-th root of unity is a primitive $d$-th root of unity for exactly one $d \mid n$, so
$x^n - 1 = \prod_{d \mid n}\Phi_d$. By induction on $n$, $\Phi_n$ is the quotient of $x^n - 1$ by a
monic polynomial in $\ZZ[x]$, hence lies in $\ZZ[x]$ by division with remainder.

Let $f$ be the minimal polynomial of $\zeta$ over $\QQ$. Since $f \mid \Phi_n$ in $\QQ[x]$, write
$\Phi_n = fg$; by Lemma~\ref{rings:lemma-gauss} (and since $f$, $g$ are monic) $f, g \in \ZZ[x]$. Let $p$
be a prime with $p \nmid n$. We claim $f(\zeta^p) = 0$. Otherwise $g(\zeta^p) = 0$, so $f(x) \mid g(x^p)$ in
$\ZZ[x]$. Reducing modulo $p$ and using $\bar g(x^p) = \bar g(x)^p$ in $\mathbb{F}_p[x]$, we find that $\bar f$
and $\bar g$ have a common irreducible factor, so $\bar\Phi_n = \bar f \bar g$, and hence $x^n - 1$, has a
repeated root in a splitting field over $\mathbb{F}_p$. But the derivative $nx^{n-1}$ is coprime to
$x^n - 1$ in $\mathbb{F}_p[x]$, a contradiction. Every primitive root of unity is $\zeta^a$ with
$\gcd(a, n) = 1$, and writing $a$ as a product of primes not dividing $n$, repeated
application of the claim shows that $\zeta^a$ is a root of $f$. So $f = \Phi_n$.

$\QQ(\zeta)$ contains all $n$-th roots of unity (powers of $\zeta$), so it is the splitting field of
$x^n - 1$, hence Galois. Each $\sigma$ maps $\zeta$ to a primitive root $\zeta^{a_\sigma}$, and
$\sigma \mapsto a_\sigma$ is an injective homomorphism into $(\ZZ/n\ZZ)^\times$, surjective since
$|\mathrm{Gal}| = \deg\Phi_n = \varphi(n)$.
\end{proof}

\section{Solvability by radicals}\label{fields:section-radicals}

In this section all fields have characteristic $0$.

\begin{definition}\label{fields:definition-radical}
A \emph{radical extension} of $K$ is an extension $E/K$ with a chain
$K = E_0 \subseteq E_1 \subseteq \cdots \subseteq E_r = E$ where $E_{i+1} = E_i(a_i)$ and $a_i^{n_i} \in E_i$ for some
$n_i \geq 1$. A polynomial $f \in K[x]$ is \emph{solvable by radicals} if a splitting field of $f$ is
contained in a radical extension of $K$.
\end{definition}

\begin{lemma}\label{fields:lemma-kummer}
Let $K$ contain a primitive $n$-th root of unity $\omega$ and let $b$ in an extension satisfy
$b^n = a \in K$. Then $K(b)/K$ is Galois with cyclic Galois group. For any $K$ and $m \geq 1$, the
splitting field of $x^m - 1$ over $K$ is Galois with abelian Galois group.
\end{lemma}

\begin{proof}
If $b = 0$ there is nothing to prove. Otherwise the roots of $x^n - a$ are the distinct
$\omega^i b$, all in $K(b)$, so $K(b)$ is a splitting field of a separable polynomial. Each $\sigma$
satisfies $\sigma(b) = \omega^{k_\sigma}b$, and since $\sigma$ fixes $\omega$, $\sigma \mapsto \omega^{k_\sigma}$ is an injective
homomorphism into the cyclic group generated by $\omega$. For the second statement, the roots
of $x^m - 1$ form a cyclic group generated by some $\zeta$ (Lemma~\ref{fields:lemma-cyclic};
they are distinct in characteristic $0$), and $\sigma(\zeta) = \zeta^{a_\sigma}$ defines an injective
homomorphism into the abelian group $(\ZZ/m\ZZ)^\times$.
\end{proof}

\begin{theorem}[Galois]\label{fields:theorem-galois-radicals}
If $f \in K[x]$ is solvable by radicals, then the Galois group of a splitting field of $f$ over $K$
is solvable.
\end{theorem}

\begin{proof}
Let $L \subseteq E$ be a splitting field of $f$ inside a radical extension with data $a_i$, $n_i$ as in
Definition~\ref{fields:definition-radical}. Let $m = n_0 \cdots n_{r-1}$, $h = (x^m - 1)\prod_i m_{a_i}$, and
let $N$ be a splitting field of $h$ over $E$ (Proposition~\ref{fields:proposition-splitting-exists}).
Since $E$ is generated over $K$ by roots of $h$, $N$ is also a splitting field of $h$ over $K$.
Then $N/K$ is Galois with group $G$, and $N$ is generated over $K$ by a primitive $m$-th root of unity $\zeta$ and the elements
$\sigma(a_i)$, $\sigma \in G$, since the roots of $m_{a_i}$ are the $\sigma(a_i)$ (proof of
Theorem~\ref{fields:theorem-galois-characterisation}). Adjoin these generators in the order
$\zeta$, then all $\sigma(a_0)$, then all $\sigma(a_1)$, and so on, obtaining a chain
$K = F_0 \subseteq F_1 = K(\zeta) \subseteq F_2 \subseteq \cdots \subseteq F_s = N$. Each $F_{j+1} = F_j(b)$ with $F_1/F_0$
abelian Galois, and for $j \geq 1$, $b = \sigma(a_i)$ with $b^{n_i} = \sigma(a_i^{n_i})$ in $F_j$ (because
$a_i^{n_i} \in E_i = K(a_0, \ldots, a_{i-1})$) and $F_j$ containing a primitive $n_i$-th root of unity. By
Lemma~\ref{fields:lemma-kummer} each $F_{j+1}/F_j$ is Galois with abelian group. By the fundamental
theorem, $G_j = \mathrm{Gal}(N/F_j)$ satisfies $G_{j+1} \trianglelefteq G_j$ and
$G_j/G_{j+1} \cong \mathrm{Gal}(F_{j+1}/F_j)$ is abelian; so $G$ is solvable
(Proposition~\ref{groups:proposition-solvable}), and so is its quotient
$\mathrm{Gal}(L/K) \cong G/\mathrm{Gal}(N/L)$.
\end{proof}

\begin{remark}\label{fields:remark-converse-radicals}
The converse also holds: if the Galois group is solvable, then $f$ is solvable by radicals.
The proof uses Hilbert's Theorem 90 to show that cyclic extensions of fields containing
enough roots of unity are generated by radicals; see \cite[Chapter VI, \S 7]{lang-algebra}.
\end{remark}

\begin{lemma}\label{fields:lemma-generate-sp}
Let $p$ be prime and $G \leq S_p$ a subgroup acting transitively on $\{1, \ldots, p\}$ and containing a
transposition. Then $G = S_p$.
\end{lemma}

\begin{proof}
The orbit has size $p$, so $p$ divides $|G|$ (Theorem~\ref{groups:theorem-orbit-stabiliser}) and $G$
contains an element of order $p$ (Theorem~\ref{groups:theorem-cauchy}), necessarily a $p$-cycle
$\sigma$. Let $(a\,b) \in G$. Some power of $\sigma$ sends $a$ to $b$ and is again a $p$-cycle; renumbering,
we may assume it is $(1\,2\,\cdots\,p)$ and the transposition is $(1\,2)$. Conjugating $(1\,2)$ by powers
of the $p$-cycle gives all $(i\,\,i{+}1)$, and these generate $S_p$ since
$(i\,\,j) = (j{-}1\,\,j)(i\,\,j{-}1)(j{-}1\,\,j)$ recursively.
\end{proof}

\begin{example}\label{fields:example-quintic}
The polynomial $f = x^5 - 4x + 2$ is not solvable by radicals over $\QQ$. It is irreducible by
Eisenstein's criterion at $p = 2$. Its roots in $\CC$ (Theorem~\ref{fields:theorem-fta}) satisfy
$\sum \alpha_i = 0$ and $\sum_{i<j}\alpha_i\alpha_j = 0$ (the coefficients of $x^4$ and $x^3$ vanish), so
$\sum \alpha_i^2 = 0$; hence not all roots are real. Since $f(-2) < 0 < f(0)$, $f(1) < 0 < f(2)$, the proof
of Lemma~\ref{fields:lemma-odd-degree} gives real roots in $(-2, 0)$, $(0, 1)$ and $(1, 2)$. Non-real
roots come in conjugate pairs, so there are exactly $3$ real and $2$ non-real roots. Complex
conjugation restricts to an element of the Galois group $G$ of the splitting field in $\CC$ that
exchanges the two non-real roots and fixes the others: a transposition. $G$ acts transitively
on the roots (they are roots of one irreducible polynomial, and
Theorem~\ref{fields:theorem-extension} moves any root to any other). By
Lemma~\ref{fields:lemma-generate-sp}, $G \cong S_5$, which is not solvable
(Proposition~\ref{groups:proposition-solvable}).
\end{example}

\section{Transcendence bases}\label{fields:section-transcendence}

\begin{definition}\label{fields:definition-transcendence-basis}
Let $L/K$ be an extension. A subset $S \subseteq L$ is \emph{algebraically independent} over $K$ if
for all distinct $s_1, \ldots, s_n \in S$ the homomorphism $K[x_1, \ldots, x_n] \to L$, $x_i \mapsto s_i$, is
injective. A \emph{transcendence basis} is an algebraically independent set $S$ such that $L$ is
algebraic over $K(S)$. An extension is \emph{purely transcendental} if $L = K(S)$ for an
algebraically independent $S$.
\end{definition}

\begin{lemma}\label{fields:lemma-transcendence-exchange}
Let $S \subseteq L$ be algebraically independent over $K$ and $\alpha \in L$. Then $S \cup \{\alpha\}$ is
algebraically independent if and only if $\alpha \notin S$ and $\alpha$ is transcendental over $K(S)$.
\end{lemma}

\begin{proof}
A nonzero polynomial relation $P(s_1, \ldots, s_n, \alpha) = 0$ with $P \notin K[x_1, \ldots, x_n]$ is, after
collecting powers of $\alpha$, a nonzero polynomial over $K[s_1, \ldots, s_n] \cong K[x_1, \ldots, x_n]$ with root
$\alpha$; and conversely a nonzero polynomial over $K(S)$ with root $\alpha$ can be cleared of
denominators to give such a relation.
\end{proof}

\begin{theorem}\label{fields:theorem-transcendence-degree}
Every extension $L/K$ has a transcendence basis, and any two have the same cardinality, the
\emph{transcendence degree} $\operatorname{trdeg}_K L$. More precisely, if $T \subseteq L$ is such that $L$ is
algebraic over $K(T)$ and $S_0 \subseteq T$ is algebraically independent, then there is a
transcendence basis $S$ with $S_0 \subseteq S \subseteq T$.
\end{theorem}

\begin{proof}
By Zorn's lemma choose a maximal algebraically independent subset $S$ with
$S_0 \subseteq S \subseteq T$. By Lemma~\ref{fields:lemma-transcendence-exchange} every $t \in T$ is
algebraic over $K(S)$, so $K(T)$ is algebraic over $K(S)$ and $L$ is algebraic over $K(S)$ by
Proposition~\ref{fields:proposition-algebraic-closure-in-extension}.

We use the following fact: if $F$ is a field, $s$ is algebraic over $F(t)$ but not over $F$, then
$t$ is algebraic over $F(s)$. Indeed, clearing denominators gives a nonzero
$P = \sum_k P_k(X)Y^k \in F[X, Y]$ with $P(s, t) = 0$; not all $P_k(s)$ vanish, since otherwise all $P_k = 0$
as $s$ is transcendental over $F$; so $t$ is a root of the nonzero polynomial $\sum_k P_k(s)Y^k$.

Let $S$ and $T$ be transcendence bases. Suppose first $T = \{t_1, \ldots, t_n\}$ is finite and let
$s_1, \ldots, s_m \in S$ be distinct. Suppose that after $i < m$ steps $L$ is algebraic over
$K(s_1, \ldots, s_i, T_i)$ with $T_i \subseteq T$ of size $n - i$. Then $s_{i+1}$ is algebraic over this field;
let $T' \subseteq T_i$ be minimal such that $s_{i+1}$ is algebraic over $K(s_1, \ldots, s_i, T')$. Since $s_{i+1}$
is not algebraic over $K(s_1, \ldots, s_i)$, $T'$ is nonempty; pick $t \in T'$ and let
$F = K(s_1, \ldots, s_i, T' \setminus \{t\})$. By the fact above, $t$ is algebraic over $F(s_{i+1})$, so with
$T_{i+1} = T_i \setminus \{t\}$ the field $K(s_1, \ldots, s_i, T_i)$ is algebraic over
$K(s_1, \ldots, s_{i+1}, T_{i+1})$, and so is $L$. In particular $T_i \neq \emptyset$ at every step $i < m$,
so $m \leq n$. Hence $S$ is finite with $|S| \leq |T|$, and by symmetry $|S| = |T|$.

If both are infinite, each $t \in T$ is algebraic over $K(S_t)$ for a finite $S_t \subseteq S$, and
$S' = \bigcup_t S_t$ satisfies $|S'| \leq |T|$ by Proposition~\ref{sets:proposition-union-bound}. Then
$L$ is algebraic over $K(S')$, so every $s \in S$ is algebraic over $K(S')$, which forces $S' = S$
by Lemma~\ref{fields:lemma-transcendence-exchange}. So $|S| \leq |T|$, and by symmetry the
cardinalities agree.
\end{proof}

\begin{theorem}\label{fields:theorem-algebraically-closed-isomorphism}
Let $\Omega$ and $\Omega'$ be algebraically closed fields of the same characteristic, with prime fields
$P \subseteq \Omega$ and $P' \subseteq \Omega'$, and $\operatorname{trdeg}_P \Omega = \operatorname{trdeg}_{P'} \Omega'$. Then every bijection
between transcendence bases $B \to B'$ extends to an isomorphism $\Omega \to \Omega'$.
\end{theorem}

\begin{proof}
The prime fields are isomorphic ($\QQ$ or $\mathbb{F}_p$). A bijection $B \to B'$ induces an isomorphism
of polynomial rings $P[B] \cong P'[B']$ and hence of their fields of fractions
$P(B) \cong P'(B')$. The field $\Omega$ is an algebraic closure of $P(B)$ and $\Omega'$ one of $P'(B')$, so
by Theorem~\ref{fields:theorem-algebraic-closure} the isomorphism extends to an isomorphism
$\Omega \to \Omega'$.
\end{proof}

\begin{corollary}\label{fields:corollary-automorphisms-complex}
$\operatorname{trdeg}_\QQ \CC = 2^{\aleph_0}$. If $\alpha, \beta \in \CC$ are both transcendental over $\QQ$, there is a
field automorphism $\sigma$ of $\CC$ with $\sigma(\alpha) = \beta$. Every automorphism of the algebraic
closure $\bar{\QQ}$ of $\QQ$ in $\CC$ extends to an automorphism of $\CC$.
\end{corollary}

\begin{proof}
If $B$ is a transcendence basis, $\CC$ is algebraic over $\QQ(B)$, and a field algebraic over a field
of infinite cardinality $\kappa$ has cardinality $\kappa$ (each element is a root of one of
$\kappa$ polynomials, each with finitely many roots); $\QQ(B)$ has cardinality
$\max(|B|, \aleph_0)$. Since $|\CC| = 2^{\aleph_0}$ (Proposition~\ref{number-systems:proposition-reals-cardinality})
is uncountable, $|B| = 2^{\aleph_0}$. Extend $\{\alpha\}$ and $\{\beta\}$ to transcendence bases $B$, $B'$
(Theorem~\ref{fields:theorem-transcendence-degree} with $T = \CC$ and $S_0 = \{\alpha\}$, respectively
$S_0 = \{\beta\}$); both have cardinality
$2^{\aleph_0}$, and removing $\alpha$ and $\beta$ leaves sets of the same cardinality, so there is a
bijection $B \to B'$ with $\alpha \mapsto \beta$. Apply
Theorem~\ref{fields:theorem-algebraically-closed-isomorphism}. For the last statement, choose a
transcendence basis $B$ of $\CC$ over $\bar{\QQ}$ (it is also one over $\QQ$), extend the automorphism
to $\bar{\QQ}(B)$ by fixing $B$, and then to $\CC$ by Theorem~\ref{fields:theorem-algebraic-closure}.
\end{proof}

\begin{remark}\label{fields:remark-wild-automorphisms}
The automorphisms of Corollary~\ref{fields:corollary-automorphisms-complex} are constructed with
the axiom of choice, and apart from the identity and complex conjugation they are not
continuous (Exercise~\ref{number-systems:exercise-automorphisms-complex} shows that the only ones
fixing $\RR$ are these two). They nevertheless act on algebraic varieties defined over $\CC$ by
acting on coefficients, and the question of how they interact with Hodge theory is the subject
of Chapter~\ref{absolute-hodge}.
\end{remark}

\begin{remark}\label{fields:remark-infinite-galois}
For an algebraic extension $L/K$ that is normal and separable but infinite, $\mathrm{Gal}(L/K)$ is
the inverse limit of the finite groups $\mathrm{Gal}(F/K)$ over finite Galois subextensions $F$, and the
fundamental theorem holds for closed subgroups with respect to the resulting profinite
topology; see \cite[Chapter VI, \S 14]{lang-algebra}. Topological groups are introduced in
Chapter~\ref{general-topology}.
\end{remark}

\section{Exercises}\label{fields:section-exercises}

\begin{exercise}\label{fields:exercise-splitting-x3-2}
Show that the splitting field of $x^3 - 2$ over $\QQ$ is $\QQ(\sqrt[3]{2}, \omega)$ with
$\omega = (-1 + \sqrt{-3})/2$, that it has degree $6$, and that its Galois group is isomorphic to
$S_3$. Describe all intermediate fields.
\end{exercise}

\begin{exercise}\label{fields:exercise-constructible}
Show that if $\alpha \in \RR$ lies in a tower $\QQ = K_0 \subseteq \cdots \subseteq K_r$ with $[K_{i+1} : K_i] = 2$, then
$[\QQ(\alpha) : \QQ]$ is a power of $2$. Deduce that $\sqrt[3]{2}$ is not in any such tower
(impossibility of doubling the cube).
\end{exercise}

\begin{exercise}\label{fields:exercise-irreducible-finite-field}
Show that for every $n \geq 1$ there is an irreducible polynomial of degree $n$ over $\mathbb{F}_p$, and
that $x^{p^n} - x$ is the product of all monic irreducible polynomials over $\mathbb{F}_p$ of degree
dividing $n$.
\end{exercise}

\begin{exercise}\label{fields:exercise-dedekind-independence}
Let $G$ be a group and $\chi_1, \ldots, \chi_n \colon G \to L^\times$ distinct homomorphisms into the
multiplicative group of a field. Show that the $\chi_i$ are linearly independent over $L$ as maps
$G \to L$.
\end{exercise}

\begin{exercise}\label{fields:exercise-rational-function-field}
Let $k$ be a field and $G \leq \Aut(k(t)/k)$ the group of order $2$ generated by $t \mapsto 1/t$.
Show that $k(t)^G = k(t + 1/t)$.
\end{exercise}

\begin{exercise}\label{fields:exercise-ahlfors-countable-algebraic}
Show that $\bar{\QQ}$ is countable and that $\operatorname{trdeg}_{\bar{\QQ}} \CC = 2^{\aleph_0}$.
\end{exercise}
