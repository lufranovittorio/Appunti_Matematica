\chapter{Cohomology of Coherent Sheaves}\label{coherent-cohomology}
\chapterstatus{planned}

Cohomology of coherent sheaves on projective schemes.

\noindent\textbf{Prerequisites.} Chapter~\ref{coherent-sheaves} (Quasi-coherent and Coherent Sheaves) and Chapter~\ref{sheaf-cohomology} (Sheaf Cohomology).

\noindent\textbf{Planned contents.}
\begin{itemize}
\item Vanishing of higher cohomology on affine schemes.
\item \v{C}ech computations; the cohomology of $\mathcal{O}(n)$ on $\PP^n$.
\item Finiteness and Serre vanishing.
\item Euler characteristic and the Hilbert polynomial.
\item Serre duality.
\item Semicontinuity, and cohomology and base change.
\end{itemize}
